function radiusAt(y){
  let i=0;
  while(i<profile.length-2 && y>profile[i+1][0]) i++;
  const a=profile[i],b=profile[i+1];
  const h=b[0]-a[0],t=THREE.MathUtils.clamp((y-a[0])/h,0,1);
  const before=profile[Math.max(0,i-1)];
  const after=profile[Math.min(profile.length-1,i+2)];
  const m0=(b[1]-before[1])/(b[0]-before[0]);
  const m1=(after[1]-a[1])/(after[0]-a[0]);
  const t2=t*t,t3=t2*t;
  const r=(2*t3-3*t2+1)*a[1]+(t3-2*t2+t)*h*m0
         +(-2*t3+3*t2)*b[1]+(t3-t2)*h*m1;
  const d=((6*t2-6*t)*a[1]+(3*t2-4*t+1)*h*m0
         +(-6*t2+6*t)*b[1]+(3*t2-2*t)*h*m1)/h;
  return [Math.max(0,r),d];
}

function bodySection(y0,y1,rows,offset){
  const positions=[],normals=[],colors=[],indices=[];
  const segments=96;
  for(let j=0;j<=rows;j++){
    const y=THREE.MathUtils.lerp(y0,y1,j/rows);
    const [r,dr]=radiusAt(y);
    for(let i=0;i<=segments;i++){
      const angle=i/segments*Math.PI*2;
      const s=Math.sin(angle),c=Math.cos(angle);
      const x=r*s;
      positions.push(x,y-offset,r*.755*c);
      const n=new THREE.Vector3(s,-dr,c/.755).normalize();
      normals.push(n.x,n.y,n.z);
      const color=skinColor(x,y,c);
      colors.push(color.r,color.g,color.b);
      if(j<rows&&i<segments){
        const a=j*(segments+1)+i,b=a+segments+1;
        indices.push(a,a+1,b,a+1,b+1,b);
      }
    }
  }
  const geometry=new THREE.BufferGeometry();
  geometry.setAttribute('position',new THREE.Float32BufferAttribute(positions,3));
  geometry.setAttribute('normal',new THREE.Float32BufferAttribute(normals,3));
  geometry.setAttribute('color',new THREE.Float32BufferAttribute(colors,3));
  geometry.setIndex(indices);
  return geometry;
}

function catmull(a,b,c,d,t){
  return .5*((2*b)+(-a+c)*t+(2*a-5*b+4*c-d)*t*t+(-a+3*b-3*c+d)*t*t*t);
}

function organicForearm(parent,name,points){
  const pos=[],indices=[],colors=[];
  const rows=60,sides=40;
  const bright=new THREE.Color('#ffc015');
  const shade=new THREE.Color('#ed9d13');
  const color=new THREE.Color();
  for(let j=0;j<=rows;j++){
    const u=j/rows*(points.length-1);
    const k=Math.min(points.length-2,Math.floor(u));
    const t=u-k;
    const p0=points[Math.max(0,k-1)],p1=points[k];
    const p2=points[k+1],p3=points[Math.min(points.length-1,k+2)];
    const v=p1.map((_,c)=>catmull(p0[c],p1[c],p2[c],p3[c],t));
    const [y,x,z,r0]=v,r=Math.max(0,r0);
    for(let i=0;i<=sides;i++){
      const a=i/sides*Math.PI*2;
      pos.push(x+Math.sin(a)*r,y,z+Math.cos(a)*r*.90);
      color.copy(bright).lerp(shade,.10*(1-Math.cos(a)));
      colors.push(color.r,color.g,color.b);
      if(j<rows&&i<sides){
        const q=j*(sides+1)+i;
        indices.push(q,q+1,q+sides+1,q+1,q+sides+2,q+sides+1);
      }
    }
  }
  const geometry=new THREE.BufferGeometry();
  geometry.setAttribute('position',new THREE.Float32BufferAttribute(pos,3));
  geometry.setAttribute('color',new THREE.Float32BufferAttribute(colors,3));
  geometry.setIndex(indices);
  geometry.computeVertexNormals();
  const mesh=new THREE.Mesh(geometry,skin);
  mesh.name=name;
  mesh.castShadow=true;
  mesh.receiveShadow=true;
  parent.add(mesh);
  return mesh;
}

const upAxis=new THREE.Vector3(0,1,0);

function makeSegment(parent,name,start,end){
  const group=new THREE.Group();
  group.name=name;
  parent.add(group);
  const delta=new THREE.Vector3().subVectors(end,start);
  const restLength=delta.length();
  const rotation=new THREE.Quaternion().setFromUnitVectors(upAxis,delta.normalize());
  const unturn=new THREE.Group();
  unturn.name=name+'-rest-orientation';
  unturn.quaternion.copy(rotation).invert();
  group.add(unturn);
  const content=new THREE.Group();
  content.name=name+'-volume';
  content.position.copy(start).negate();
  unturn.add(content);
  const segment={group,content,restLength,delta:new THREE.Vector3()};
  setSegment(segment,start,end,1,1);
  return segment;
}

function setSegment(segment,start,end,width,depth){
  segment.delta.subVectors(end,start);
  const length=segment.delta.length();
  segment.group.position.copy(start);
  segment.group.quaternion.setFromUnitVectors(upAxis,segment.delta.divideScalar(length));
  segment.group.scale.set(width,length/segment.restLength,depth);
}

const period=1200;
const count=frames.length;
const channels=frames[0].v.length;

/* Periodic, shape-preserving cubic interpolation. */
for(let i=0;i<count;i++){
  const previous=frames[(i+count-1)%count];
  const current=frames[i];
  const next=frames[(i+1)%count];
  const h0=(current.t-previous.t+period)%period;
  const h1=(next.t-current.t+period)%period;
  current.m=new Float64Array(channels);
  for(let j=0;j<channels;j++){
    const d0=(current.v[j]-previous.v[j])/h0;
    const d1=(next.v[j]-current.v[j])/h1;
    if(d0*d1>0){
      const w0=2*h1+h0,w1=h1+2*h0;
      current.m[j]=(w0+w1)/(w0/d0+w1/d1);
    }
  }
}
const sampled=new Float64Array(channels);

function sample(time){
  let i=0;
  while(i<count-1 && time>=frames[i+1].t) i++;
  const a=frames[i],b=frames[(i+1)%count];
  const duration=(b.t-a.t+period)%period;
  const u=(time-a.t)/duration;
  const u2=u*u,u3=u2*u;
  const h00=2*u3-3*u2+1;
  const h10=u3-2*u2+u;
  const h01=-2*u3+3*u2;
  const h11=u3-u2;
  for(let j=0;j<channels;j++){
    sampled[j]=h00*a.v[j]+h10*duration*a.m[j]
      +h01*b.v[j]+h11*duration*b.m[j];
  }
}

function smoothWeight(y,low,high){
  const u=THREE.MathUtils.clamp((y-low)/(high-low),0,1);
  return u*u*(3-2*u);
}
function weightDerivative(y,low,high){
  const u=THREE.MathUtils.clamp((y-low)/(high-low),0,1);
  return 6*u*(1-u)/(high-low);
}

let bendCos=1,bendSin=0,headHeight=1,headTX=0,headTY=0,bodyDip=0;

function warpPoint(source,target){
  const x=source.x,y=source.y;
  const w=smoothWeight(y,-.34,.82);
  const bob=smoothWeight(y,-2.08,.82);
  const rx=bendCos*x-bendSin*headHeight*(y-.82)+headTX;
  const ry=bendSin*x+bendCos*headHeight*(y-.82)+.82+headTY;
  target.set(
    x+(rx-x)*w,
    y+(ry-y)*w+bodyDip*bob,
    source.z
  );
  return target;
}

function poseBody(){
  bodyDip=sampled[0];
  const roll=sampled[1];
  headHeight=sampled[3];
  bendCos=Math.cos(roll);
  bendSin=Math.sin(roll);
  headTX=sampled[2]/pixelsPerUnit+.562*headHeight*bendSin;
  headTY=.562*(1-headHeight*bendCos);

  head.position.set(headTX,.82+headTY+bodyDip,0);
  head.rotation.set(0,0,roll);
  head.scale.set(1,headHeight,1);

  const positions=torsoPositions.array;
  const normals=torsoNormals.array;
  for(let i=0;i<restPositions.length;i+=3){
    const x=restPositions[i],y=restPositions[i+1];
    const w=smoothWeight(y,-.34,.82);
    const dw=weightDerivative(y,-.34,.82);
    const bob=smoothWeight(y,-2.08,.82);
    const dbob=weightDerivative(y,-2.08,.82);
    const rx=bendCos*x-bendSin*headHeight*(y-.82)+headTX;
    const ry=bendSin*x+bendCos*headHeight*(y-.82)+.82+headTY;
    positions[i]=x+(rx-x)*w;
    positions[i+1]=y+(ry-y)*w+bodyDip*bob;
    positions[i+2]=restPositions[i+2];

    const a=1+w*(bendCos-1);
    const b=-w*bendSin*headHeight+dw*(rx-x);
    const c=w*bendSin;
    const d=1+w*(bendCos*headHeight-1)+dw*(ry-y)+bodyDip*dbob;
    const determinant=a*d-b*c;
    const nx=(d*restNormals[i]-c*restNormals[i+1])/determinant;
    const ny=(-b*restNormals[i]+a*restNormals[i+1])/determinant;
    const nz=restNormals[i+2];
    const inverseLength=1/Math.hypot(nx,ny,nz);
    normals[i]=nx*inverseLength;
    normals[i+1]=ny*inverseLength;
    normals[i+2]=nz*inverseLength;
  }
  torsoPositions.needsUpdate=true;
  torsoNormals.needsUpdate=true;
}

  elbow:new THREE.Vector3(),
  palm:new THREE.Vector3()
};
const rightLimb={
  upper:rightUpper,forearm:rightForearm,hand:rightHand,
  restShoulder:rightShoulderRest,
  shoulder:new THREE.Vector3(),
  elbow:new THREE.Vector3(),
  palm:new THREE.Vector3()
};

function poseLimb(limb,offset){
  screenPoint(sampled[offset],sampled[offset+1],sampled[offset+4],limb.palm);
  screenPoint(sampled[offset+2],sampled[offset+3],sampled[offset+5],limb.elbow);
  warpPoint(limb.restShoulder,limb.shoulder);

  const width=sampled[offset+6];
  setSegment(limb.upper,limb.shoulder,limb.elbow,1,1);
  setSegment(limb.forearm,limb.elbow,limb.palm,width,1+(width-1)*.5);
  limb.hand.position.copy(limb.palm);
  limb.hand.rotation.set(
    sampled[offset+11],
    sampled[offset+10],
    sampled[offset+9]
  );
  const palmWidth=sampled[offset+7];
  limb.hand.scale.set(
    palmWidth,
    sampled[offset+8],
    1+(palmWidth-1)*.35
  );
}

function render(t){
  const numericTime=Number(t);
  const time=Number.isFinite(numericTime)
    ? ((numericTime%period)+period)%period : 0;
  sample(time);
  poseBody();
  poseLimb(leftLimb,4);
  poseLimb(rightLimb,16);
  renderer.render(scene,ref);
}

window.nwb={
  scene,
  renderer,
  cameras:{ref},
  characters:[character],
  render,
  ready:true
};
render(0);
