const byId=id=>document.getElementById(id);
const nodes={};
[
 "torso-outline","head-outline","neck-shape","head-surface",

function mix(a,b,u){
  return a.map((v,i)=>v+(b[i]-v)*u);
}

function cubicPath(a){
  let d="M"+a[0]+" "+a[1];
  for(let i=2;i<a.length;i+=6)d+="C"+a.slice(i,i+6).join(" ");
  return d+"Z";
}

function ellipse(id,a){
  const e=nodes[id]||byId(id);
  e.setAttribute("cx",a[0]);
  e.setAttribute("cy",a[1]);
  e.setAttribute("rx",a[2]);
  e.setAttribute("ry",a[3]);
}

function limbData(s,e,w){
  const ux=e[0]-s[0],uy=e[1]-s[1];
  const vx=w[0]-e[0],vy=w[1]-e[1];
  const ul=Math.hypot(ux,uy)||1;
  const vl=Math.hypot(vx,vy)||1;
  const ns=[-uy/ul,ux/ul];
  const nw=[-vy/vl,vx/vl];
  let nx=ns[0]+nw[0],ny=ns[1]+nw[1];
  let nl=Math.hypot(nx,ny);
  if(nl<.01){nx=ns[0];ny=ns[1];nl=1;}
  let tx=ux/ul+vx/vl,ty=uy/ul+vy/vl;
  let tl=Math.hypot(tx,ty);
  if(tl<.01){tx=vx/vl;ty=vy/vl;tl=1;}
  const reach=Math.min(ul,vl)*.31;
  return {
    ux:ux,uy:uy,vx:vx,vy:vy,ul:ul,vl:vl,
    ns:ns,nw:nw,ne:[nx/nl,ny/nl],
    jt:[tx/tl*reach,ty/tl*reach]
  };
}

function tube(s,e,w,sw,ew,ww){
  const a=limbData(s,e,w);
  const p=(v,n,r)=>[v[0]+n[0]*r,v[1]+n[1]*r];
  const f=v=>v.join(" ");
  const sp=p(s,a.ns,sw),ep=p(e,a.ne,ew),wp=p(w,a.nw,ww);
  const sm=p(s,a.ns,-sw),em=p(e,a.ne,-ew),wm=p(w,a.nw,-ww);
  return "M"+f(sp)+
    "C"+f([sp[0]+a.ux*.46,sp[1]+a.uy*.46])+" "+
          f([ep[0]-a.jt[0],ep[1]-a.jt[1]])+" "+f(ep)+
    "C"+f([ep[0]+a.jt[0],ep[1]+a.jt[1]])+" "+
          f([wp[0]-a.vx*.27,wp[1]-a.vy*.27])+" "+f(wp)+
    "Q"+f([w[0]+a.vx/a.vl*ww,w[1]+a.vy/a.vl*ww])+" "+f(wm)+
    "C"+f([wm[0]-a.vx*.27,wm[1]-a.vy*.27])+" "+
          f([em[0]+a.jt[0],em[1]+a.jt[1]])+" "+f(em)+
    "C"+f([em[0]-a.jt[0],em[1]-a.jt[1]])+" "+
          f([sm[0]+a.ux*.46,sm[1]+a.uy*.46])+" "+f(sm)+
    "Q"+f([s[0]-a.ux/a.ul*sw,s[1]-a.uy/a.ul*sw])+" "+f(sp)+"Z";
}

function centerPath(s,e,w){
  const a=limbData(s,e,w);
  return "M"+s.join(" ")+
    "C"+[s[0]+a.ux*.46,s[1]+a.uy*.46,
          e[0]-a.jt[0],e[1]-a.jt[1],e[0],e[1]].join(" ")+
    "C"+[e[0]+a.jt[0],e[1]+a.jt[1],
          w[0]-a.vx*.27,w[1]-a.vy*.27,w[0],w[1]].join(" ");
}

function drawHand(side,a){
  const hand=byId("hand-"+side);
  const spread=a[11],curl=a[12];
  const sign=side==="right"?-1:1;
  hand.setAttribute("transform",
    "translate("+a[4]+" "+a[5]+") rotate("+a[9]+") scale("+(a[10]*sign)+" "+a[10]+")");
  const digits=[
    [[-5,7],[-9-5*spread,12],[-10-9*spread,23-22*spread]],
    [[-5,14],[-6-3*spread,23],[-7-5*spread,31]],
    [[-1,17],[-1,25],[-1,35]],
    [[3,16],[4+2*spread,24],[5+4*spread,33]],
    [[6,13],[8+4*spread,20],[10+7*spread,28-2*spread]]
  ];
  digits.forEach((d,i)=>{
    const base=d[0],joint=d[1].slice(),tip=d[2].slice();
    joint[1]=base[1]+(joint[1]-base[1])*(1-.2*curl);
    tip[0]*=1-.4*curl;
    tip[1]=base[1]+(tip[1]-base[1])*(1-.79*curl);
    const width=i===0?2.95:(i===4?2.3:2.7);
    fingers[side][i].setAttribute("d",
      tube(base,joint,tip,width,width*.87,width*.61));
  });
}

function drawArm(side,a,depth,profile){
  const s=[a[0],a[1]],e=[a[2],a[3]],w=[a[4],a[5]];
  byId(side+"-arm-shape").setAttribute("d",tube(s,e,w,a[6],a[7],a[8]));
  const center=centerPath(s,e,w);
  byId(side+"-arm-highlight").setAttribute("d",center);
  byId(side+"-arm-highlight").setAttribute("transform","translate(-2 -1)");
  byId(side+"-contact-shadow").setAttribute("d",center);
  byId(side+"-contact-shadow").setAttribute("opacity",
    depth*(.17+.04*profile));
  drawHand(side,a);
}

function drawLeg(side,a){
  const x=a[0],y=a[1],r=a[2],kx=a[3],ky=a[4],kr=a[5],ax=a[6],ay=a[7],ar=a[8];
  const d="M"+(x-r)+" "+y+
    "C"+(x-r+1)+" "+(y+17)+" "+(kx-kr)+" "+(ky-14)+" "+(kx-kr)+" "+ky+
    "C"+(kx-kr+2)+" "+(ky+17)+" "+(ax-ar)+" "+(ay-10)+" "+(ax-ar)+" "+ay+
    "L"+(ax+ar)+" "+ay+
    "C"+(ax+ar)+" "+(ay-14)+" "+(kx+kr)+" "+(ky+15)+" "+(kx+kr)+" "+ky+
    "C"+(kx+kr)+" "+(ky-15)+" "+(x+r-1)+" "+(y+15)+" "+(x+r)+" "+y+"Z";
  byId(side+"-leg-shape").setAttribute("d",d);
  byId(side+"-leg-shadow").setAttribute("d",d);
  byId(side+"-leg-light").setAttribute("d",
    "M"+x+" "+(y+8)+"Q"+kx+" "+ky+" "+ax+" "+ay);
  if(side==="left"){
    byId("left-leg-seam").setAttribute("d",
      "M"+(x+r-2)+" "+(y+8)+"Q"+(kx+kr-1)+" "+ky+" "+(ax+ar-1)+" "+ay);
  }
}

function eye(side,a){
  const g=byId("eye-"+side);
  g.setAttribute("transform",
    "translate("+a[0]+" "+a[1]+") scale("+a[2]+" "+a[3]+")");
  g.setAttribute("opacity",a[4]);
  const open=a[5];
  const rgb=[
    Math.round(139+(12-139)*open),
    Math.round(139+(24-139)*open),
    Math.round(48+(9-48)*open)
  ];
  byId(side+"-pupil").setAttribute("fill","rgb("+rgb.join(",")+")");
  byId(side+"-pupil").setAttribute("opacity",.65+.35*open);
  byId(side+"-eye-glint").setAttribute("opacity",.13+.52*open);
  byId(side+"-eyelid").setAttribute("opacity",1-open);
}

function render(t){
  let ms=Number(t);
  if(!Number.isFinite(ms))ms=0;
  ms=((ms%5540)+5540)%5540;
  let k=0;
  while(k<7&&ms>=times[k+1])k++;
  const j=Math.min(k+1,7);
  let u=k===7?0:(ms-times[k])/(times[j]-times[k]);
  u=u*u*(3-2*u);
  const q={};
  Object.keys(frames[0]).forEach(key=>{
    q[key]=mix(frames[k][key],frames[j][key],u);
  });

  nodes["torso-outline"].setAttribute("d",cubicPath(q.b));
  nodes["neck-shape"].setAttribute("d",cubicPath(q.n));
  nodes["head-outline"].setAttribute("d",
    cubicPath(mix(frontHead,sideHead,q.h[4])));
  nodes["head-surface"].setAttribute("transform",
    "translate("+q.h[0]+" "+q.h[1]+") scale("+q.h[2]+" "+q.h[3]+")");

  nodes["front-body-color"].setAttribute("opacity",1-.91*q.h[4]);
  ellipse("belly-patch",q.bl);
  ellipse("chest-light",q.ch);
  nodes["chest-light"].setAttribute("opacity",.3-.12*q.h[4]);
  ellipse("lower-belly-light",
    [q.bl[0]+4,q.bl[1]+23,q.bl[2]*.66,q.bl[3]*.36]);
  ellipse("back-highlight",
    [q.bl[0]+q.bl[2]*1.5,q.bl[1]-33,q.bl[2]*.38,q.bl[3]*.72]);
  nodes["back-highlight"].setAttribute("opacity",.38*q.h[4]);
  ellipse("body-edge-shade",
    [q.bl[0]+q.bl[2]*.93,q.bl[1]+15,q.bl[2]*.17,q.bl[3]*.82]);

  drawLeg("right",q.rl);
  drawLeg("left",q.ll);
  drawArm("left",q.la,q.dep[0],q.h[4]);
  drawArm("right",q.ra,q.dep[1],q.h[4]);

  ["left","right"].forEach((side,i)=>{
    const parent=q.dep[i]>.5?nodes["front-arms"]:nodes["rear-arms"];
    parent.appendChild(nodes["arm-"+side]);
  });

  eye("left",q.el);
  eye("right",q.er);
  ellipse("muzzle",q.mz);
  nodes["muzzle"].setAttribute("transform",
    "rotate("+q.mz[4]+" "+q.mz[0]+" "+q.mz[1]+")");
  nodes["muzzle"].setAttribute("opacity",q.mz[5]);

  const m=q.m;
  nodes["mouth-line"].setAttribute("d",
    "M"+m[0]+" "+m[1]+"C"+m.slice(2).join(" "));
  nodes["lower-lip"].setAttribute("d",
    "M"+(m[0]+2)+" "+(m[1]+2.2)+
    "C"+m[2]+" "+(m[3]+2.4)+" "+m[4]+" "+(m[5]+2.4)+" "+(m[6]-2)+" "+(m[7]+2.2));
}

window.nwb={render:render};
