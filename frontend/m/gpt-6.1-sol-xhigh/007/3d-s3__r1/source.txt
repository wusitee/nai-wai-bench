function sample(rows,y,k){
  let i=0;
  while(i<rows.length-2&&y>rows[i+1][0])i++;
  const a=rows[i],b=rows[i+1];
  const p=rows[Math.max(0,i-1)],n=rows[Math.min(rows.length-1,i+2)];
  const d=b[0]-a[0],t=THREE.MathUtils.clamp((y-a[0])/d,0,1);
  const m0=(b[k]-p[k])/(b[0]-p[0]);
  const m1=(n[k]-a[k])/(n[0]-a[0]);
  return (2*t*t*t-3*t*t+1)*a[k]
    +(t*t*t-2*t*t+t)*d*m0
    +(-2*t*t*t+3*t*t)*b[k]
    +(t*t*t-t*t)*d*m1;
}
function profileMesh(name,parent,rows,mat,colorFn){
  const heightSteps=88,sides=64;
  const pos=[],uv=[],colors=[],indices=[];
  const y0=rows[0][0],y1=rows[rows.length-1][0];
  for(let i=0;i<=heightSteps;i++){
    const y=THREE.MathUtils.lerp(y0,y1,i/heightSteps);
    const cx=sample(rows,y,1),rx=Math.max(0,sample(rows,y,2));
    const rz=Math.max(0,sample(rows,y,3)),cz=sample(rows,y,4);
    for(let j=0;j<=sides;j++){
      const a=j/sides*Math.PI*2;
      const x=cx+rx*Math.cos(a),z=cz+rz*Math.sin(a);
      pos.push(x,y,z);
      uv.push(j/sides,i/heightSteps);
      if(colorFn){
        const c=colorFn(x,y,z,a);
        colors.push(c.r,c.g,c.b);
      }
      if(i<heightSteps&&j<sides){
        const a0=i*(sides+1)+j,b=a0+sides+1;
        indices.push(a0,b,b+1,a0,b+1,a0+1);
      }
    }
  }
  const geo=new THREE.BufferGeometry();
  geo.setAttribute('position',new THREE.Float32BufferAttribute(pos,3));
  geo.setAttribute('uv',new THREE.Float32BufferAttribute(uv,2));
  if(colorFn)geo.setAttribute('color',new THREE.Float32BufferAttribute(colors,3));
  geo.setIndex(indices);
  geo.computeVertexNormals();
  const mesh=new THREE.Mesh(geo,mat);
  mesh.name=name;
  mesh.castShadow=true;
  mesh.receiveShadow=true;
  parent.add(mesh);
  return mesh;
}
function taperedCurve(name,parent,points,r0,r1,mat){
  const curve=new THREE.CatmullRomCurve3(points.map(p=>new THREE.Vector3(...p)));
  const count=20,sides=10;
  const frames=curve.computeFrenetFrames(count,false);
  const positions=[],uvs=[],indices=[];
  for(let i=0;i<=count;i++){
    const t=i/count,p=curve.getPointAt(t);
    const r=THREE.MathUtils.lerp(r0,r1,t);
    for(let j=0;j<=sides;j++){
      const a=j/sides*Math.PI*2;
      const v=p.clone()
        .addScaledVector(frames.normals[i],Math.cos(a)*r)
        .addScaledVector(frames.binormals[i],Math.sin(a)*r);
      positions.push(v.x,v.y,v.z);
      uvs.push(j/sides,t);
      if(i<count&&j<sides){
        const k=i*(sides+1)+j,l=k+sides+1;
        indices.push(k,k+1,l+1,k,l+1,l);
      }
    }
  }
  const start=positions.length/3;
  const p0=curve.getPointAt(0),p1=curve.getPointAt(1);
  positions.push(p0.x,p0.y,p0.z,p1.x,p1.y,p1.z);
  uvs.push(.5,0,.5,1);
  for(let j=0;j<sides;j++){
    indices.push(start,j+1,j);
    const k=count*(sides+1)+j;
    indices.push(start+1,k,k+1);
  }
  const geo=new THREE.BufferGeometry();
  geo.setAttribute('position',new THREE.Float32BufferAttribute(positions,3));
  geo.setAttribute('uv',new THREE.Float32BufferAttribute(uvs,2));
  geo.setIndex(indices);
  geo.computeVertexNormals();
  const mesh=new THREE.Mesh(geo,mat);
  mesh.name=name;
  mesh.castShadow=true;
  mesh.receiveShadow=true;
  parent.add(mesh);
  return mesh;
}

function makeHand(parent,side){
  const hand=group(side<0?'left-hand':'right-hand',parent);
  const palm=group('palm',hand);
  const p=ellipsoid('olive-palm',palm,handMat,side*.020,-.105,.025,.105,.14,.071);
  p.rotation.z=side*.10;
  const fingers=[];
  const lengths=[.18,.235,.25,.20];
  for(let i=0;i<4;i++){
    const x=(i-1.5)*.048+side*.020;
    const dx=(i-1.5)*.021+side*.012;
    const len=lengths[i];
    const f=group('finger-'+(i+1),hand,x,-.16,.035);
    taperedCurve('proximal-finger',f,[
      [0,0,0],[dx*.18,-len*.24,.014],[dx*.4,-len*.57,.024]
    ],.033,.026,fingerMat);
    const tipJoint=group('finger-tip-joint',f,dx*.4,-len*.57,.024);
    taperedCurve('distal-finger',tipJoint,[
      [0,0,0],[dx*.32,-len*.21,.006],[dx*.6,-len*.43,-.006]
    ],.026,.017,fingerMat);
    ellipsoid('rounded-fingertip',tipJoint,fingerMat,
      dx*.6,-len*.43,-.006,.018,.021,.018);
    fingers.push({root:f,tip:tipJoint});
  }
  const thumb=group('thumb',hand);
  taperedCurve('bent-thumb',thumb,[
    [-side*.062,-.055,.045],
    [-side*.12,-.10,.073],
    [-side*.14,-.17,.081],
    [-side*.13,-.225,.07]
  ],.038,.021,handMat);
  ellipsoid('thumb-tip',thumb,handMat,-side*.13,-.225,.07,.022,.025,.022);
  hand.userData={fingers,thumb,side};
  return hand;
}

function quaternionFrom(a){
  return new THREE.Quaternion().setFromEuler(new THREE.Euler(a[0],a[1],a[2],'ZYX'));
}
for(const k of keys){
  k.qL=quaternionFrom(k.Lrot);
  k.qR=quaternionFrom(k.Rrot);
  k.qH=quaternionFrom(k.Hrot);
}

const down=new THREE.Vector3(0,-1,0);
function poseArm(arm,elbow,wrist){
  const e=new THREE.Vector3(...elbow);
  const w=new THREE.Vector3(...wrist);
  const upperDirection=e.clone().sub(arm.shoulder);
  const foreDirection=w.clone().sub(e);
  const upperLength=upperDirection.length();
  const foreLength=foreDirection.length();
  arm.upper.quaternion.setFromUnitVectors(down,upperDirection.clone().normalize());
  arm.upper.scale.set(1,upperLength,1);
  arm.elbow.position.copy(upperDirection);
  arm.fore.quaternion.setFromUnitVectors(down,foreDirection.normalize());
  arm.fore.scale.set(1,foreLength,1);
  arm.hand.position.copy(w.sub(arm.shoulder));
}
function poseFingers(hand,curl,spread,size){
  hand.scale.setScalar(size);
  const d=hand.userData;
  for(let i=0;i<d.fingers.length;i++){
    const f=d.fingers[i];
    f.root.rotation.set(curl*.25,0,(i-1.5)/1.5*spread);
    f.tip.rotation.x=curl*1.75;
  }
  d.thumb.rotation.set(curl*.22,0,-d.side*spread*.8);
}

  const a=keys[index],b=keys[index+1];

  for(const e of eyeParts){
