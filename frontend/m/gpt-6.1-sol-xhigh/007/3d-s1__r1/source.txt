function ellipsoid(name,parent,mat,x,y,z,sx,sy,sz){

}

// Profiles contain [height, horizontal center, width, depth, depth center].
function sample(rows,y,k){
  let i=0;
  while(i<rows.length-2 && y>rows[i+1][0])i++;
  const a=rows[i], b=rows[i+1];
  const p=rows[Math.max(0,i-1)], n=rows[Math.min(rows.length-1,i+2)];
  const d=b[0]-a[0], t=Math.max(0,Math.min(1,(y-a[0])/d));
  const m0=(b[k]-p[k])/(b[0]-p[0]);
  const m1=(n[k]-a[k])/(n[0]-a[0]);
  return (2*t*t*t-3*t*t+1)*a[k]
    +(t*t*t-2*t*t+t)*d*m0
    +(-2*t*t*t+3*t*t)*b[k]
    +(t*t*t-t*t)*d*m1;
}
function profileMesh(name,parent,rows,mat,colorFn){
  const heightSteps=88, sides=64;
  const pos=[],uv=[],colors=[],indices=[];
  const y0=rows[0][0], y1=rows[rows.length-1][0];
  for(let i=0;i<=heightSteps;i++){
    const y=THREE.MathUtils.lerp(y0,y1,i/heightSteps);
    const cx=sample(rows,y,1), rx=Math.max(0,sample(rows,y,2));
    const rz=Math.max(0,sample(rows,y,3)), cz=sample(rows,y,4);
    for(let j=0;j<=sides;j++){
      const angle=j/sides*Math.PI*2;
      const x=cx+rx*Math.cos(angle), z=cz+rz*Math.sin(angle);
      pos.push(x,y,z);
      uv.push(j/sides,i/heightSteps);
      if(colorFn){
        const c=colorFn(x,y,z,angle);
        colors.push(c.r,c.g,c.b);
      }
      if(i<heightSteps && j<sides){
        const a=i*(sides+1)+j,b=a+sides+1;
        indices.push(a,b,b+1,a,b+1,a+1);
      }
    }
  }
  const geo=new THREE.BufferGeometry();
  geo.setAttribute('position',new THREE.Float32BufferAttribute(pos,3));
  geo.setAttribute('uv',new THREE.Float32BufferAttribute(uv,2));
  if(colorFn)geo.setAttribute('color',new THREE.Float32BufferAttribute(colors,3));
  geo.setIndex(indices);
  geo.computeVertexNormals();
  const mesh=new THREE.Mesh(geo,mat);
  mesh.name=name;
  mesh.castShadow=true;
  mesh.receiveShadow=true;
  parent.add(mesh);
  return mesh;
}
function taperedCurve(name,parent,points,r0,r1,mat){
  const curve=new THREE.CatmullRomCurve3(points.map(p=>new THREE.Vector3(...p)));
  const count=20,sides=9;
  const frames=curve.computeFrenetFrames(count,false);
  const positions=[],uvs=[],indices=[];
  for(let i=0;i<=count;i++){
    const t=i/count,p=curve.getPointAt(t);
    const r=THREE.MathUtils.lerp(r0,r1,t);
    for(let j=0;j<=sides;j++){
      const a=j/sides*Math.PI*2;
      const v=p.clone()
        .addScaledVector(frames.normals[i],Math.cos(a)*r)
        .addScaledVector(frames.binormals[i],Math.sin(a)*r);
      positions.push(v.x,v.y,v.z);
      uvs.push(j/sides,t);
      if(i<count && j<sides){
        const k=i*(sides+1)+j,l=k+sides+1;
        indices.push(k,k+1,l+1,k,l+1,l);
      }
    }
  }
  const start=positions.length/3;
  const p0=curve.getPointAt(0),p1=curve.getPointAt(1);
  positions.push(p0.x,p0.y,p0.z,p1.x,p1.y,p1.z);
  uvs.push(.5,0,.5,1);
  for(let j=0;j<sides;j++){
    indices.push(start,j+1,j);
    const k=count*(sides+1)+j;
    indices.push(start+1,k,k+1);
  }
  const geometry=new THREE.BufferGeometry();
  geometry.setAttribute('position',new THREE.Float32BufferAttribute(positions,3));
  geometry.setAttribute('uv',new THREE.Float32BufferAttribute(uvs,2));
  geometry.setIndex(indices);
  geometry.computeVertexNormals();
  const mesh=new THREE.Mesh(geometry,mat);
  mesh.name=name;
  mesh.castShadow=true;
  parent.add(mesh);
  return mesh;
}

function makeLeg(side){
  const left=side<0;
  const leg=group(left?'left-leg':'right-leg',bird);
  const topX=left?-.36:.31;
  const ankleX=left?-.22:.32;
  const legOrange=new THREE.Color(left?0xeaa51a:0xe8ac1e);
  const legGold=new THREE.Color(0xf4bb29);
  profileMesh('tapered-leg',leg,[
    [-.36,ankleX,0,0,.02],
    [-.20,ankleX,.185,.205,.02],
    [.10,ankleX,.205,.235,.02],
    [.46,left?-.27:.31,.235,.27,.015],
    [.86,left?-.33:.31,.275,.305,0],
    [1.18,topX,left?.315:.285,.345,-.025],
    [1.46,topX,.30,.32,-.035],
    [1.65,topX,0,0,-.04]
  ],plumage,(x,y,z,a)=>{
    const f=Math.max(0,Math.sin(a));
    return legOrange.clone().lerp(legGold,.28*f);
  });
  const foot=group(left?'left-foot':'right-foot',leg,ankleX,-.30,.095);
  ellipsoid('rounded-foot',foot,footMat,0,-.035,.085,.235,.105,.32);
  for(let i=0;i<3;i++){
    const toe=group('toe-'+(i+1),foot);
    ellipsoid('toe',toe,footMat,(i-1)*.115,-.045,.30,.065,.065,.19);
  }
}

function makeHand(parent,side){
  const hand=group(side<0?'left-hand':'right-hand',parent);
  const palm=group('palm',hand);
  const palmMesh=ellipsoid('olive-palm',palm,handMat,side*.025,-.105,.025,.105,.14,.068);
  palmMesh.rotation.z=side*.10;
  const lengths=[.20,.265,.28,.22];
  for(let i=0;i<4;i++){
    const f=group('finger-'+(i+1),hand);
    const x=(i-1.5)*.052+side*.025;
    const dx=(i-1.5)*.025+side*.014;
    const len=lengths[i];
    const end=[x+dx,-.17-len,.05];
    taperedCurve('curved-finger',f,[
      [x,-.16,.035],
      [x+dx*.4,-.22,.057],
      [x+dx*.85,-.17-len*.72,.065],
      end
    ],.029,.012,fingerMat);
    ellipsoid('rounded-fingertip',f,fingerMat,...end,.013,.017,.014);
  }
  const thumb=group('thumb',hand);
  taperedCurve('bent-thumb',thumb,[
    [-side*.062,-.055,.045],
    [-side*.12,-.10,.073],
    [-side*.145,-.18,.081],
    [-side*.13,-.25,.07]
  ],.036,.017,handMat);
  ellipsoid('thumb-tip',thumb,handMat,-side*.13,-.25,.07,.017,.022,.018);
  return hand;
}
function makeArm(side){
  const left=side<0;
  const sy=left?3.25:3.27;
  const sx=left?-.65:.62;
  const length=left?1.43:1.55;
  const dx=left?-.40:.39;
  const arm=group(left?'left-arm':'right-arm',bird,sx,sy,left?-.045:-.12);
  const wing=group('wing',arm);
  const wingGold=new THREE.Color(0xeeb919);
  const wingLight=new THREE.Color(0xf7c523);
  const wristColor=new THREE.Color(0x9f9135);
  profileMesh('long-tapering-wing',wing,[
    [-length-.045,dx,0,0,.15],
    [-length,dx,.058,.067,.15],
    [-length+.19,dx*.94,.072,.083,.12],
    [-length+.46,dx*.75,.092,.105,.075],
    [-.65,dx*.57,.117,.135,.025],
    [-.27,dx*.26,.15,.16,0],
    [.04,0,.15,.155,-.015],
    [.17,-side*.015,0,0,-.025]
  ],plumage,(x,y,z,a)=>{
    const front=Math.max(0,Math.sin(a));
    const c=wingGold.clone().lerp(wingLight,front*.25);
    return c.lerp(wristColor,1-THREE.MathUtils.smoothstep(y,-length,-length+.22));
  });
  const hand=makeHand(arm,side);
  hand.position.set(dx,-length,.15);
  hand.rotation.z=left?-.11:.12;
}
