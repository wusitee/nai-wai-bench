const furMats=new Map();
function fur(mesh,count,color,length=.025){
  const geometry=mesh.geometry;
  const p=geometry.attributes.position;
  const n=geometry.attributes.normal;
  const ix=geometry.index;
  const triangleCount=Math.floor((ix?ix.count:p.count)/3);
  const positions=[],colors=[];
  const col=new THREE.Color(color);
  const pa=v(0,0,0),pb=v(0,0,0),pc=v(0,0,0);
  const na=v(0,0,0),nb=v(0,0,0),nc=v(0,0,0);
  for(let i=0;i<count;i++){
    const tri=Math.floor(random()*triangleCount)*3;
    const a=ix?ix.getX(tri):tri;
    const b=ix?ix.getX(tri+1):tri+1;
    const c=ix?ix.getX(tri+2):tri+2;
    pa.fromBufferAttribute(p,a); pb.fromBufferAttribute(p,b); pc.fromBufferAttribute(p,c);
    na.fromBufferAttribute(n,a); nb.fromBufferAttribute(n,b); nc.fromBufferAttribute(n,c);
    let u=random(),w=random();
    if(u+w>1){u=1-u;w=1-w;}
    const pt=pa.clone().multiplyScalar(1-u-w).addScaledVector(pb,u).addScaledVector(pc,w);
    const normal=na.clone().multiplyScalar(1-u-w).addScaledVector(nb,u).addScaledVector(nc,w).normalize();
    const len=length*(.35+random()*.8);
    const start=pt.clone().addScaledVector(normal,.0025);
    const end=pt.clone().addScaledVector(normal,len);
    end.x+=(random()-.5)*len*.45;
    end.y+=(random()-.5)*len*.45;
    end.z+=(random()-.5)*len*.45;
    positions.push(start.x,start.y,start.z,end.x,end.y,end.z);
    const tint=.78+random()*.35;
    for(let j=0;j<2;j++) colors.push(col.r*tint,col.g*tint,col.b*tint);
  }
  const geo=new THREE.BufferGeometry();
  geo.setAttribute('position',new THREE.Float32BufferAttribute(positions,3));
  geo.setAttribute('color',new THREE.Float32BufferAttribute(colors,3));
  let mat=furMats.get(color);
  if(!mat){
    mat=new THREE.LineBasicMaterial({vertexColors:true,transparent:true,opacity:.22,depthWrite:false});
    furMats.set(color,mat);
  }
  const hairs=new THREE.LineSegments(geo,mat);
  hairs.name=mesh.name+'-fine-fur';
  mesh.add(hairs);
}

function cubic(a,b,c,d,t){
  return .5*((2*b)+(-a+c)*t+(2*a-5*b+4*c-d)*t*t+(-a+3*b-3*c+d)*t*t*t);
}
function sampleProfile(profile,y){
  let i=0;
  while(i<profile.length-2 && y>profile[i+1][0]) i++;
  const b=profile[i],c=profile[i+1];
  const t=THREE.MathUtils.clamp((y-b[0])/(c[0]-b[0]),0,1);
  const a=profile[Math.max(0,i-1)],d=profile[Math.min(profile.length-1,i+2)];
  const result=[y];
  for(let k=1;k<5;k++) result[k]=cubic(a[k],b[k],c[k],d[k],t);
  result[1]=Math.max(.0001,result[1]);
  result[2]=Math.max(.0001,result[2]);
  return result;
}
function bodyMesh(name,parent,profile){
  const positions=[],uvs=[],indices=[];
  const rows=112,sides=80;
  const lo=profile[0][0],hi=profile[profile.length-1][0];
  for(let i=0;i<=rows;i++){
    const t=i/rows;
    const y=lo+(hi-lo)*(.5-.5*Math.cos(t*Math.PI));
    const s=sampleProfile(profile,y);
    for(let j=0;j<=sides;j++){
      const a=j/sides*Math.PI*2;
      positions.push(s[3]+s[1]*Math.cos(a),y,s[4]+s[2]*Math.sin(a));
      uvs.push(j/sides,t);
      if(i<rows&&j<sides){
        const a0=i*(sides+1)+j,b0=a0+sides+1;
        indices.push(a0,b0,a0+1,b0,b0+1,a0+1);
      }
    }
  }
  const geo=new THREE.BufferGeometry();
  geo.setAttribute('position',new THREE.Float32BufferAttribute(positions,3));
  geo.setAttribute('uv',new THREE.Float32BufferAttribute(uvs,2));
  geo.setIndex(indices); geo.computeVertexNormals();
  const mesh=new THREE.Mesh(geo,yellow);
  mesh.name=name; mesh.castShadow=true; mesh.receiveShadow=true;
  parent.add(mesh);
  fur(mesh,10500,'#f5cb44',.027);
  return mesh;
}
function bellyPatch(name,parent,profile,cx,cy,rx,ry){
  const positions=[],uvs=[],indices=[];
  const rings=25,sides=80;
  for(let r=0;r<=rings;r++){
    for(let j=0;j<=sides;j++){
      const angle=j/sides*Math.PI*2;
      const q=r/rings;
      const x=cx+Math.cos(angle)*q*rx;
      const y=cy+Math.sin(angle)*q*ry;
      const s=sampleProfile(profile,y);
      const a=THREE.MathUtils.clamp((x-s[3])/s[1],-.999,.999);
      const z=s[4]+s[2]*Math.sqrt(1-a*a)+.012;
      positions.push(x,y,z);
      uvs.push(.5+q*Math.cos(angle)*.5,.5+q*Math.sin(angle)*.5);
      if(r<rings&&j<sides){
        const a0=r*(sides+1)+j,b0=a0+sides+1;
        indices.push(a0,a0+1,b0,a0+1,b0+1,b0);
      }
    }
  }
  const geo=new THREE.BufferGeometry();

function tube(name,parent,points,radii,material,segments=48,sides=14){
  const curve=new THREE.CatmullRomCurve3(points.map(a=>v(...a)));
  const geo=new THREE.TubeGeometry(curve,segments,1,sides,false);
  const pos=geo.attributes.position;
  for(let i=0;i<=segments;i++){
    const t=i/segments;
    const f=t*(radii.length-1);
    const a=Math.min(radii.length-2,Math.floor(f));
    const r=THREE.MathUtils.lerp(radii[a],radii[a+1],f-a);
    const center=curve.getPointAt(t);
    for(let j=0;j<=sides;j++){
      const index=i*(sides+1)+j;
      const p=v(pos.getX(index),pos.getY(index),pos.getZ(index));
      p.sub(center).multiplyScalar(r).add(center);
      pos.setXYZ(index,p.x,p.y,p.z);
    }
  }
  geo.computeVertexNormals();
  const mesh=new THREE.Mesh(geo,material);
  mesh.name=name;mesh.castShadow=true;mesh.receiveShadow=true;
  parent.add(mesh);
  ellipsoid(name+'-rounded-start',parent,material,v(...points[0]),v(radii[0],radii[0],radii[0]));
  const end=points[points.length-1],r=radii[radii.length-1];
  ellipsoid(name+'-rounded-end',parent,material,v(...end),v(r,r,r));
  return mesh;
}

    const notes=new THREE.LineSegments(geo,pageLineMat);
    notes.name='small-pencil-handwriting';g.add(notes);
  }
  return g;
}
