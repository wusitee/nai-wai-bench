const furMats=new Map();
function fur(mesh,count,color,length=.025){
  const geometry=mesh.geometry;
  const p=geometry.attributes.position;
  const n=geometry.attributes.normal;
  const ix=geometry.index;
  const motion=geometry.getAttribute('motionWeight');
  const triangleCount=Math.floor((ix?ix.count:p.count)/3);
  const positions=[],colors=[],weights=[];
  const col=new THREE.Color(color);
  const pa=v(0,0,0),pb=v(0,0,0),pc=v(0,0,0);
  const na=v(0,0,0),nb=v(0,0,0),nc=v(0,0,0);
  for(let i=0;i<count;i++){
    const tri=Math.floor(random()*triangleCount)*3;
    const a=ix?ix.getX(tri):tri;
    const b=ix?ix.getX(tri+1):tri+1;
    const c=ix?ix.getX(tri+2):tri+2;
    pa.fromBufferAttribute(p,a);
    pb.fromBufferAttribute(p,b);
    pc.fromBufferAttribute(p,c);
    na.fromBufferAttribute(n,a);
    nb.fromBufferAttribute(n,b);
    nc.fromBufferAttribute(n,c);
    let u=random(),w=random();
    if(u+w>1){u=1-u;w=1-w;}
    const pt=pa.clone().multiplyScalar(1-u-w).addScaledVector(pb,u).addScaledVector(pc,w);
    const normal=na.clone().multiplyScalar(1-u-w).addScaledVector(nb,u).addScaledVector(nc,w).normalize();
    const len=length*(.35+random()*.8);
    const start=pt.clone().addScaledVector(normal,.0025);
    const end=pt.clone().addScaledVector(normal,len);
    end.x+=(random()-.5)*len*.45;
    end.y+=(random()-.5)*len*.45;
    end.z+=(random()-.5)*len*.45;
    positions.push(start.x,start.y,start.z,end.x,end.y,end.z);
    const tint=.86+random()*.23;
    for(let j=0;j<2;j++) colors.push(col.r*tint,col.g*tint,col.b*tint);
    if(motion){
      const weight=motion.getX(a)*(1-u-w)+motion.getX(b)*u+motion.getX(c)*w;
      weights.push(weight,weight);
    }
  }
  const geo=new THREE.BufferGeometry();
  geo.setAttribute('position',new THREE.Float32BufferAttribute(positions,3));
  geo.setAttribute('color',new THREE.Float32BufferAttribute(colors,3));
  if(motion) geo.setAttribute('motionWeight',new THREE.Float32BufferAttribute(weights,1));
  let mat=furMats.get(color);
  if(!mat){
    mat=new THREE.LineBasicMaterial({
      vertexColors:true,transparent:true,opacity:.14,depthWrite:false
    });
    furMats.set(color,mat);
  }
  const hairs=new THREE.LineSegments(geo,mat);
  hairs.name=mesh.name+'-fine-fur';
  mesh.add(hairs);
}

function cubic(a,b,c,d,t){
  return .5*((2*b)+(-a+c)*t+(2*a-5*b+4*c-d)*t*t+(-a+3*b-3*c+d)*t*t*t);
}
function sampleProfile(profile,y){
  let i=0;
  while(i<profile.length-2&&y>profile[i+1][0]) i++;
  const b=profile[i],c=profile[i+1];
  const t=clamp((y-b[0])/(c[0]-b[0]),0,1);
  const a=profile[Math.max(0,i-1)],d=profile[Math.min(profile.length-1,i+2)];
  const result=[y];
  for(let k=1;k<5;k++) result[k]=cubic(a[k],b[k],c[k],d[k],t);
  result[1]=Math.max(.0001,result[1]);
  result[2]=Math.max(.0001,result[2]);
  return result;
}
function bodyMesh(name,parent,profile){
  const positions=[],uvs=[],indices=[];
  const rows=112,sides=80;
  const lo=profile[0][0],hi=profile[profile.length-1][0];
  for(let i=0;i<=rows;i++){
    const t=i/rows;
    const y=lo+(hi-lo)*(.5-.5*Math.cos(t*Math.PI));
    const s=sampleProfile(profile,y);
    for(let j=0;j<=sides;j++){
      const a=j/sides*Math.PI*2;
      positions.push(s[3]+s[1]*Math.cos(a),y,s[4]+s[2]*Math.sin(a));
      uvs.push(j/sides,t);
      if(i<rows&&j<sides){
        const a0=i*(sides+1)+j,b0=a0+sides+1;
        indices.push(a0,b0,a0+1,b0,b0+1,a0+1);
      }
    }
  }
  const geo=new THREE.BufferGeometry();
  geo.setAttribute('position',new THREE.Float32BufferAttribute(positions,3));
  geo.setAttribute('uv',new THREE.Float32BufferAttribute(uvs,2));
  geo.setIndex(indices);
  geo.computeVertexNormals();
  const mesh=new THREE.Mesh(geo,yellow);
  mesh.name=name;
  mesh.castShadow=true;
  mesh.receiveShadow=true;
  parent.add(mesh);
  fur(mesh,10500,'#f9ca4c',.023);
  return mesh;
}
function bellyPatch(name,parent,profile,cx,cy,rx,ry){
  const positions=[],uvs=[],indices=[],colors=[];
  const rings=32,sides=80;
  const pale=new THREE.Color('#f3dfb7');
  const gold=new THREE.Color('#ffc134');
  for(let r=0;r<=rings;r++){
    for(let j=0;j<=sides;j++){
      const angle=j/sides*Math.PI*2;
      const q=r/rings;
      const x=cx+Math.cos(angle)*q*rx;
      const y=cy+Math.sin(angle)*q*ry;
      const s=sampleProfile(profile,y);
      const a=clamp((x-s[3])/s[1],-.999,.999);
      const z=s[4]+s[2]*Math.sqrt(1-a*a)+.009;
      positions.push(x,y,z);
      uvs.push(.5+q*Math.cos(angle)*.5,.5+q*Math.sin(angle)*.5);
      const color=pale.clone().lerp(gold,smooth((q-.88)/.12));
      colors.push(color.r,color.g,color.b);
      if(r<rings&&j<sides){
        const a0=r*(sides+1)+j,b0=a0+sides+1;
        indices.push(a0,b0,a0+1,a0+1,b0,b0+1);
      }
    }
  }
  const geo=new THREE.BufferGeometry();
  geo.setAttribute('position',new THREE.Float32BufferAttribute(positions,3));
  geo.setAttribute('uv',new THREE.Float32BufferAttribute(uvs,2));
  geo.setAttribute('color',new THREE.Float32BufferAttribute(colors,3));
  geo.setIndex(indices);
  geo.computeVertexNormals();
  const material=cream.clone();
  material.color.set('#ffffff');
  material.vertexColors=true;
  const mesh=new THREE.Mesh(geo,material);
  mesh.name=name;
  mesh.castShadow=false;
  mesh.receiveShadow=true;
  parent.add(mesh);
  fur(mesh,2400,'#ead8b6',.017);
  return mesh;
}
function tube(name,parent,points,radii,material,segments=48,sides=14,motionStart=null){
  const curve=new THREE.CatmullRomCurve3(points.map(a=>v(...a)));
  const geo=new THREE.TubeGeometry(curve,segments,1,sides,false);
  const pos=geo.attributes.position;
  const weights=[];
  for(let i=0;i<=segments;i++){
    const t=i/segments;
    const f=t*(radii.length-1);
    const a=Math.min(radii.length-2,Math.floor(f));
    const r=THREE.MathUtils.lerp(radii[a],radii[a+1],f-a);
    const center=curve.getPointAt(t);
    for(let j=0;j<=sides;j++){
      const index=i*(sides+1)+j;
      const p=v(pos.getX(index),pos.getY(index),pos.getZ(index));
      p.sub(center).multiplyScalar(r).add(center);
      pos.setXYZ(index,p.x,p.y,p.z);
      if(motionStart!==null) weights.push(smooth((t-motionStart)/(1-motionStart)));
    }
  }
  if(motionStart!==null){
    geo.setAttribute('motionWeight',new THREE.Float32BufferAttribute(weights,1));
  }
  geo.computeVertexNormals();
  const mesh=new THREE.Mesh(geo,material);
  mesh.name=name;
  mesh.castShadow=true;
  mesh.receiveShadow=true;
  parent.add(mesh);
  const first=ellipsoid(name+'-rounded-start',parent,material,
    v(...points[0]),v(radii[0],radii[0],radii[0]));
  const end=points[points.length-1],r=radii[radii.length-1];
  const last=ellipsoid(name+'-rounded-end',parent,material,v(...end),v(r,r,r));
  mesh.userData.caps=[first,last];
  mesh.userData.capBases=[first.position.clone(),last.position.clone()];
  return mesh;
}

function mouth(name,parent,points,r=.014,openingWidth=.16,faceAngle=.2){
  const g=group(name,parent);
  const center=v(0,0,0);
  for(const p of points) center.add(v(...p));
  center.multiplyScalar(1/points.length);
  g.position.copy(center);
  const path=new THREE.CatmullRomCurve3(points.map(a=>v(...a).sub(center)));
  const lineMaterial=smileMat.clone();
  lineMaterial.transparent=true;
  lineMaterial.depthWrite=false;
  const line=new THREE.Mesh(new THREE.TubeGeometry(path,32,r,8,false),lineMaterial);
  line.name='quiet-smile';
  g.add(line);
  const openingMaterial=new THREE.MeshStandardMaterial({
    color:'#392e1d',roughness:1,transparent:true,opacity:0,depthWrite:false
  });
  const opening=ellipsoid('small-mouth-opening',g,openingMaterial,
    v(0,0,.008),v(openingWidth,.024,.021));
  opening.rotation.y=faceAngle;
  opening.visible=false;
  g.userData={line,opening,openingWidth};
  return g;
}
function poseMouth(g,amount){

  const {line,opening,openingWidth}=g.userData;
  line.visible=amount<.999;
  line.material.opacity=1-amount;
  line.scale.y=1-.75*amount;
  opening.visible=amount>.001;
  opening.material.opacity=amount;
  opening.scale.set(openingWidth*(1+.35*amount),.007+.020*amount,.021);
}

function book(name,x,z,w,d,angle,writing){
  const g=group(name,desk);
  g.position.set(x,.729,z);
  g.rotation.y=angle;
  box('thin-brown-cover',g,coverMat,v(0,.014,0),v(w+.055,.027,d+.035));
  box('stack-of-paper',g,pageSideMat,v(0,.035,0),v(w,.037,d));
  for(const side of [-1,1]){
    const pos=[],uv=[],indices=[];
    const nx=24,nz=12;
    for(let j=0;j<=nz;j++){
      for(let i=0;i<=nx;i++){
        const xx=side*(i/nx)*w*.5;
        const zz=(j/nz-.5)*d;
        const yy=pageHeight(xx,w)+.006*Math.sin(j/nz*Math.PI);
        pos.push(xx,yy,zz);
        uv.push(i/nx,j/nz);
        if(i<nx&&j<nz){
          const a=j*(nx+1)+i,b=a+nx+1;
          if(side<0) indices.push(a,b,a+1,a+1,b,b+1);
          else indices.push(a,a+1,b,a+1,b+1,b);
        }
      }
    }
    const geo=new THREE.BufferGeometry();
    geo.setAttribute('position',new THREE.Float32BufferAttribute(pos,3));
    geo.setAttribute('uv',new THREE.Float32BufferAttribute(uv,2));
    geo.setIndex(indices);
    geo.computeVertexNormals();
    const m=new THREE.Mesh(geo,pageMat);
    m.name=side<0?'left-open-page':'right-open-page';
    m.receiveShadow=true;
    m.castShadow=true;
    g.add(m);
  }
  const creaseGeo=new THREE.BufferGeometry().setFromPoints([
    v(0,.050,-d*.5),v(0,.050,d*.5)
  ]);
  const crease=new THREE.Line(creaseGeo,new THREE.LineBasicMaterial({
    color:'#bfb49b',transparent:true,opacity:.55
  }));
  crease.name='center-page-fold';
  g.add(crease);
  const edgePositions=[];
  for(let j=0;j<4;j++){
    const y=.025+j*.008;
    edgePositions.push(-w/2,y,d/2,w/2,y,d/2);
  }
  const edges=new THREE.BufferGeometry();
  edges.setAttribute('position',new THREE.Float32BufferAttribute(edgePositions,3));
  g.add(new THREE.LineSegments(edges,new THREE.LineBasicMaterial({
    color:'#b8a98d',transparent:true,opacity:.35
  })));
  if(writing){
    const strokes=[];
    for(let row=0;row<5;row++){
      const zz=-d*.22+row*.095;
      let xx=-w*.39+random()*.035;
      const end=-w*.11+random()*.10;
      while(xx<end){
        const length=.035+random()*.075;
        const segments=3+Math.floor(random()*4);
        for(let k=0;k<segments;k++){
          const xa=xx+length*k/segments,xb=xx+length*(k+1)/segments;
          const za=zz+(random()-.5)*.032,zb=zz+(random()-.5)*.032;
          strokes.push(xa,pageHeight(xa,w)+.009,za,xb,pageHeight(xb,w)+.009,zb);
        }
        xx+=length+.025+random()*.025;
      }
    }
    const geo=new THREE.BufferGeometry();
    geo.setAttribute('position',new THREE.Float32BufferAttribute(strokes,3));
    const notes=new THREE.LineSegments(geo,pageLineMat);
    notes.name='small-pencil-handwriting';
    g.add(notes);
  }
  return g;
}

}

function captureDeformation(mesh,weightFunction=null){
  const targets=[mesh,...mesh.children.filter(child=>child.isLineSegments)];
  return targets.map(object=>{
    object.frustumCulled=false;
    const geometry=object.geometry;
    const position=geometry.getAttribute('position');
    const normal=geometry.getAttribute('normal');
    const motion=geometry.getAttribute('motionWeight');
    const base=new Float32Array(position.array);
    const weights=new Float32Array(position.count);
    for(let i=0;i<position.count;i++){
      weights[i]=weightFunction
        ?weightFunction(base[i*3+1])
        :(motion?motion.getX(i):0);
    }
    return {
      geometry,position,normal,base,weights,
      baseNormals:normal?new Float32Array(normal.array):null
    };
  });
}

function poseHeadSurface(records,pivot,quaternion,translation){
  const e=rotationMatrix.makeRotationFromQuaternion(quaternion).elements;
  for(const record of records){
    const {geometry,position,normal,base,weights,baseNormals}=record;
    const dst=position.array;
    const nd=normal?normal.array:null;
    for(let i=0;i<weights.length;i++){
      const j=i*3,w=weights[i];
      const x=base[j]-pivot.x;
      const y=base[j+1]-pivot.y;
      const z=base[j+2]-pivot.z;
      const rx=e[0]*x+e[4]*y+e[8]*z;
      const ry=e[1]*x+e[5]*y+e[9]*z;
      const rz=e[2]*x+e[6]*y+e[10]*z;
      dst[j]=base[j]+w*(rx-x+translation.x);
      dst[j+1]=base[j+1]+w*(ry-y+translation.y);
      dst[j+2]=base[j+2]+w*(rz-z+translation.z);
      if(nd){
        const nx=baseNormals[j],ny=baseNormals[j+1],nz=baseNormals[j+2];
        const ax=nx+w*(e[0]*nx+e[4]*ny+e[8]*nz-nx);
        const ay=ny+w*(e[1]*nx+e[5]*ny+e[9]*nz-ny);
        const az=nz+w*(e[2]*nx+e[6]*ny+e[10]*nz-nz);
        const inv=1/(Math.hypot(ax,ay,az)||1);
        nd[j]=ax*inv;
        nd[j+1]=ay*inv;
        nd[j+2]=az*inv;
      }
    }
    position.needsUpdate=true;
    if(normal) normal.needsUpdate=true;
    geometry.boundingBox=null;
    geometry.boundingSphere=null;
  }
}
function poseArmSurface(mesh,records,translation){
  for(const record of records){
    const {geometry,position,base,weights,normal}=record;
    const dst=position.array;
    for(let i=0;i<weights.length;i++){
      const j=i*3,w=weights[i];
      dst[j]=base[j]+w*translation.x;
      dst[j+1]=base[j+1]+w*translation.y;
      dst[j+2]=base[j+2]+w*translation.z;
    }
    position.needsUpdate=true;
    if(normal) geometry.computeVertexNormals();
    geometry.boundingBox=null;
    geometry.boundingSphere=null;
  }
  mesh.userData.caps[0].position.copy(mesh.userData.capBases[0]);
  mesh.userData.caps[1].position.copy(mesh.userData.capBases[1]).add(translation);
}

function sampleTrack(values,time){
  if(time>=1400) return values[7];
  const i=Math.floor(time/200);
  const u=(time-i*200)/200;
  return cubic(
    values[Math.max(0,i-1)],
    values[i],
    values[Math.min(7,i+1)],
    values[Math.min(7,i+2)],
    u
  );
}
