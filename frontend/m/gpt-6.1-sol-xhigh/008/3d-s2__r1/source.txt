function cubic(a,b,c,d,t){
  return b+0.5*t*(c-a+t*(2*a-5*b+4*c-d+t*(3*(b-c)+d-a)));
}
function interpolate(values,t){
  const u=THREE.MathUtils.clamp(t,0,1)*(values.length-1);
  const i=Math.min(values.length-2,Math.floor(u)),f=u-i;
  return cubic(values[Math.max(0,i-1)],values[i],values[i+1],
    values[Math.min(values.length-1,i+2)],f);
}
function closeSeamNormals(geometry,rows,columns){
  const n=geometry.attributes.normal;
  for(let j=0;j<rows;j++){
    const a=j*(columns+1),b=a+columns;
    const v=V(n.getX(a)+n.getX(b),n.getY(a)+n.getY(b),n.getZ(a)+n.getZ(b)).normalize();
    n.setXYZ(a,v.x,v.y,v.z);
    n.setXYZ(b,v.x,v.y,v.z);
  }
  n.needsUpdate=true;
}

function profile(y){
  y=THREE.MathUtils.clamp(y,bodyProfile[0][0],bodyProfile.at(-1)[0]);
  let i=0;
  while(i<bodyProfile.length-2&&y>bodyProfile[i+1][0])i++;
  const t=(y-bodyProfile[i][0])/(bodyProfile[i+1][0]-bodyProfile[i][0]);
  const out=[];
  for(let k=1;k<5;k++){
    out.push(cubic(
      bodyProfile[Math.max(0,i-1)][k],bodyProfile[i][k],
      bodyProfile[i+1][k],bodyProfile[Math.min(bodyProfile.length-1,i+2)][k],t
    ));
  }
  out[0]=Math.max(0.012,out[0]);
  out[1]=Math.max(0.012,out[1]);
  return out;
}
function bodyPoint(y,a){
  const [rx,rz,cx,cz]=profile(y);
  return V(cx+rx*Math.sin(a),y,cz+rz*Math.cos(a));
}
function bodyNormal(y,a){
  const p=profile(y),p0=profile(y-0.002),p1=profile(y+0.002);
  const da=V(p[0]*Math.cos(a),0,-p[1]*Math.sin(a));
  const dy=V(
    (p1[2]-p0[2]+(p1[0]-p0[0])*Math.sin(a))/0.004,
    1,
    (p1[3]-p0[3]+(p1[1]-p0[1])*Math.cos(a))/0.004
  );
  return da.cross(dy).normalize();
}

function addFur(parent,name,count,sampler,baseColor){
  const positions=[],normals=[],colors=[];
  for(let i=0;i<count;i++){
    const sample=sampler();
    if(!sample)continue;
    const p=sample.p,n=sample.n.normalize();
    const color=(sample.color||baseColor).clone().multiplyScalar(0.92+random()*0.13);
    let tangent=new THREE.Vector3().crossVectors(n,Math.abs(n.y)<0.9?V(0,1,0):V(1,0,0)).normalize();
    const bitangent=new THREE.Vector3().crossVectors(n,tangent).normalize();
    const a=random()*Math.PI*2;
    tangent.multiplyScalar(Math.cos(a)).addScaledVector(bitangent,Math.sin(a)).normalize();
    const cross=new THREE.Vector3().crossVectors(n,tangent).normalize();
    const length=0.012+random()*0.016,width=0.002+random()*0.0015;
    const base=p.clone().addScaledVector(n,-0.003);
    const tip=p.clone().addScaledVector(n,length).addScaledVector(tangent,(random()-0.5)*0.009);
    for(const direction of [tangent,cross]){
      const a=base.clone().addScaledVector(direction,-width);
      const b=base.clone().addScaledVector(direction,width);
      for(const v of [a,b,tip]){
        positions.push(v.x,v.y,v.z);
        normals.push(n.x,n.y,n.z);
        colors.push(color.r,color.g,color.b);
      }
    }
  }
  const geo=new THREE.BufferGeometry();
  geo.setAttribute('position',new THREE.Float32BufferAttribute(positions,3));
  geo.setAttribute('normal',new THREE.Float32BufferAttribute(normals,3));
  geo.setAttribute('color',new THREE.Float32BufferAttribute(colors,3));
  const m=mesh(geo,furMat,parent,name);
  m.castShadow=false;
}

function tubeFrame(curve,t){
  const tangent=curve.getTangent(t).normalize();
  let a=new THREE.Vector3().crossVectors(tangent,V(0,0,1));
  if(a.lengthSq()<0.0001)a.crossVectors(tangent,V(1,0,0));
  a.normalize();
  const b=new THREE.Vector3().crossVectors(tangent,a).normalize();
  return {a,b,tangent};
}
function tubeGeometry(curve,radii,t0,t1,origin,segments=44,sides=36,depth=1){
  const pos=[],uv=[],idx=[];
  for(let j=0;j<=segments;j++){
    const t=t0+(t1-t0)*j/segments;
    const p=curve.getPoint(t).sub(origin),r=interpolate(radii,t);
    const f=tubeFrame(curve,t);
    for(let i=0;i<=sides;i++){
      const a=2*Math.PI*i/sides;
      const q=p.clone().addScaledVector(f.a,r*Math.cos(a))
        .addScaledVector(f.b,r*depth*Math.sin(a));
      pos.push(q.x,q.y,q.z);uv.push(i/sides,t);
      if(j<segments&&i<sides){
        const k=j*(sides+1)+i;
        idx.push(k,k+sides+1,k+1,k+1,k+sides+1,k+sides+2);
      }
    }
  }
  const cap0=pos.length/3,p0=curve.getPoint(t0).sub(origin);
  pos.push(p0.x,p0.y,p0.z);uv.push(0.5,t0);
  const cap1=pos.length/3,p1=curve.getPoint(t1).sub(origin);
  pos.push(p1.x,p1.y,p1.z);uv.push(0.5,t1);
  for(let i=0;i<sides;i++){
    idx.push(cap0,i,i+1);
    const a=segments*(sides+1)+i;
    idx.push(cap1,a+1,a);
  }
  const geo=new THREE.BufferGeometry();
  geo.setAttribute('position',new THREE.Float32BufferAttribute(pos,3));
  geo.setAttribute('uv',new THREE.Float32BufferAttribute(uv,2));
  geo.setIndex(idx);
  geo.computeVertexNormals();
  closeSeamNormals(geo,segments+1,sides);
  geo.userData={segments,sides,depth};
  return geo;
}
function deformTube(geo,curve,radii,t0,t1,origin){
  const {segments,sides,depth}=geo.userData;
  const pos=geo.attributes.position;
  let k=0;
  for(let j=0;j<=segments;j++){
    const t=t0+(t1-t0)*j/segments;
    const p=curve.getPoint(t).sub(origin),r=interpolate(radii,t);
    const f=tubeFrame(curve,t);
    for(let i=0;i<=sides;i++){
      const a=2*Math.PI*i/sides,c=r*Math.cos(a),s=r*depth*Math.sin(a);
      pos.setXYZ(k++,p.x+f.a.x*c+f.b.x*s,p.y+f.a.y*c+f.b.y*s,p.z+f.a.z*c+f.b.z*s);
    }
  }
  const p0=curve.getPoint(t0).sub(origin),p1=curve.getPoint(t1).sub(origin);
  pos.setXYZ(k++,p0.x,p0.y,p0.z);
  pos.setXYZ(k,p1.x,p1.y,p1.z);
  pos.needsUpdate=true;
  geo.computeVertexNormals();
  closeSeamNormals(geo,segments+1,sides);
}
function limbFur(parent,count,curve,radii,t0,t1,origin,color){
  const samples=[],colors=[];
  for(let i=0;i<count;i++){
    samples.push({
      t:t0+(t1-t0)*random(),a:random()*Math.PI*2,
      length:0.012+random()*0.016,width:0.002+random()*0.0015,
      turn:random()*Math.PI*2,bend:(random()-0.5)*0.009
    });
    const c=color.clone().multiplyScalar(0.92+random()*0.13);
    for(let j=0;j<6;j++)colors.push(c.r,c.g,c.b);
  }
  const geo=new THREE.BufferGeometry();
  geo.setAttribute('position',new THREE.Float32BufferAttribute(new Float32Array(count*18),3));
  geo.setAttribute('normal',new THREE.Float32BufferAttribute(new Float32Array(count*18),3));
  geo.setAttribute('color',new THREE.Float32BufferAttribute(colors,3));
  const m=mesh(geo,furMat,parent,'short-limb-fuzz');
  m.castShadow=false;
  function update(){
    const positions=geo.attributes.position,normals=geo.attributes.normal;
    let index=0;
    for(const s of samples){
      const f=tubeFrame(curve,s.t);
      const n=f.a.clone().multiplyScalar(Math.cos(s.a)).addScaledVector(f.b,Math.sin(s.a)).normalize();
      const circ=new THREE.Vector3().crossVectors(n,f.tangent).normalize();
      const tangent=f.tangent.clone().multiplyScalar(Math.cos(s.turn)).addScaledVector(circ,Math.sin(s.turn)).normalize();
      const cross=new THREE.Vector3().crossVectors(n,tangent).normalize();
      const p=curve.getPoint(s.t).sub(origin).addScaledVector(n,interpolate(radii,s.t));
      const base=p.clone().addScaledVector(n,-0.003);
      const tip=p.clone().addScaledVector(n,s.length).addScaledVector(tangent,s.bend);
      for(const direction of [tangent,cross]){
        const a=base.clone().addScaledVector(direction,-s.width);
        const b=base.clone().addScaledVector(direction,s.width);
        for(const v of [a,b,tip]){
          positions.setXYZ(index,v.x,v.y,v.z);
          normals.setXYZ(index,n.x,n.y,n.z);
          index++;
        }
      }
    }
    positions.needsUpdate=true;
    normals.needsUpdate=true;
  }
  update();
  return update;
}
function makeLimb(name,points,radii,material,upperName,lowerName,furCount){
  const start=points[0].clone();
  const root=group(name,character,start);
  const curve=new THREE.CatmullRomCurve3(points.map(p=>p.clone().sub(start)),false,'catmullrom',0.35);
  const split=0.54,elbow=curve.getPoint(split),zero=V(0,0,0);
  const upper=group(upperName,root);
  const lower=group(lowerName,root,elbow);
  const upMesh=mesh(tubeGeometry(curve,radii,0,split,zero),material,upper,upperName+'-volume');
  const lowMesh=mesh(tubeGeometry(curve,radii,split,1,elbow),material,lower,lowerName+'-volume');
  const updateUpperFur=limbFur(upper,Math.floor(furCount/2),curve,radii,0,split,zero,material.color);
  const updateLowerFur=limbFur(lower,Math.floor(furCount/2),curve,radii,split,1,elbow,material.color);
  const end=curve.getPoint(1).sub(elbow);
  function pose(newPoints){
    start.copy(newPoints[0]);
    root.position.copy(start);
    for(let i=0;i<curve.points.length;i++)curve.points[i].copy(newPoints[i]).sub(start);
    elbow.copy(curve.getPoint(split));
    lower.position.copy(elbow);
    end.copy(curve.getPoint(1)).sub(elbow);
    deformTube(upMesh.geometry,curve,radii,0,split,zero);
    deformTube(lowMesh.geometry,curve,radii,split,1,elbow);
    updateUpperFur();
    updateLowerFur();
  }
  return {root,upper,lower,end,pose};
}

function headZ(x,y){
  return RZ*Math.sqrt(Math.max(0,1-(x/RX)**2-(y/RY)**2));
}
function mouthRim(a){
  const c=Math.cos(a),s=Math.sin(a);
  const x=mouthX+mouthRX*mouthWidth*Math.sign(c)*Math.pow(Math.abs(c),0.57)*(0.88+0.12*s);
  const y=mouthY+mouthRY*mouthOpening*(s>=0?Math.pow(s,0.57):s);
  return V(x,y,headZ(x,y));
}
function headSurface(front){
  const positions=[],normals=[],uv=[],idx=[],rows=34,cols=128;
  for(let j=0;j<=rows;j++){
    const u=j/rows;
    for(let i=0;i<=cols;i++){
      const a=2*Math.PI*i/cols;
      let x,y,z;
      if(front){
        const inner=mouthRim(a);
        x=THREE.MathUtils.lerp(inner.x,RX*Math.cos(a),u);
        y=THREE.MathUtils.lerp(inner.y,RY*Math.sin(a),u);
        z=headZ(x,y);
      }else{
        x=RX*u*Math.cos(a);y=RY*u*Math.sin(a);
        z=-RZ*Math.sqrt(Math.max(0,1-u*u));
      }
      positions.push(x,y,z);
      const n=V(x/(RX*RX),y/(RY*RY),z/(RZ*RZ)).normalize();
      normals.push(n.x,n.y,n.z);
      uv.push(0.5+Math.atan2(x/RX,z/RZ)/(2*Math.PI),
        0.5+Math.asin(THREE.MathUtils.clamp(y/RY,-1,1))/Math.PI);
      if(j<rows&&i<cols){
        const a0=j*(cols+1)+i,b=a0+cols+1,c=b+1,d=a0+1;
        if(front)idx.push(a0,b,c,a0,c,d);
        else idx.push(a0,c,b,a0,d,c);
      }
    }
  }
  const geo=new THREE.BufferGeometry();
  geo.setAttribute('position',new THREE.Float32BufferAttribute(positions,3));
  geo.setAttribute('normal',new THREE.Float32BufferAttribute(normals,3));
  geo.setAttribute('uv',new THREE.Float32BufferAttribute(uv,2));
  geo.setIndex(idx);
  return geo;
}

function cavityGeometry(){
  const positions=[],colors=[],idx=[],rings=30,cols=128;
  const dark=new THREE.Color(0x090603),edge=new THREE.Color(0x30200e);
  for(let j=0;j<=rings;j++){
    const r=j/rings;
    for(let i=0;i<=cols;i++){
      const a=2*Math.PI*i/cols,p=mouthRim(a);
      const x=mouthX+(p.x-mouthX)*r,y=mouthY+(p.y-mouthY)*r;
      const z=0.065+(p.z-0.073)*Math.pow(r,1.65);
      positions.push(x,y,z);
      const tint=dark.clone().lerp(edge,Math.pow(r,4)*0.7);
      colors.push(tint.r,tint.g,tint.b);
      if(j<rings&&i<cols){
        const q=j*(cols+1)+i,b=q+cols+1;
        idx.push(q,b,b+1,q,b+1,q+1);
      }
    }
  }
  const geo=new THREE.BufferGeometry();
  geo.setAttribute('position',new THREE.Float32BufferAttribute(positions,3));
  geo.setAttribute('color',new THREE.Float32BufferAttribute(colors,3));
  geo.setIndex(idx);geo.computeVertexNormals();
  return geo;
}

function insideMouth(x,y){
  const dy=(y-mouthY)/mouthRY;
  if(dy<=-1||dy>=1)return false;
  const s=dy>=0?Math.pow(dy,1/0.57):dy;
  const c=Math.sqrt(Math.max(0,1-s*s));
  const w=mouthRX*Math.pow(c,0.57)*(0.88+0.12*s);
  return Math.abs(x-mouthX)<w+0.025;
}
addFur(head,'fine-head-fuzz',850,()=>{

  const p=V(RX*r*Math.cos(a),RY*y,RZ*r*Math.sin(a));
  if(p.z>0&&insideMouth(p.x,p.y))return null;
  return {p,n:V(p.x/(RX*RX),p.y/(RY*RY),p.z/(RZ*RZ)).normalize()};
},headGold.color);

  mouthWidth=width;
  replaceGeometry(faceMesh,headSurface(true));
  replaceGeometry(cavity,cavityGeometry());
  replaceGeometry(lip,pathGeometry(lipPoints(),0.026,true,112));
  replaceGeometry(innerLip,pathGeometry(lowerLipPoints(),0.013,false,60));
  for(const tooth of toothMeshes)replaceGeometry(tooth.mesh,toothGeometry(tooth.a0,tooth.a1));
  replaceGeometry(toothCrescent,pathGeometry(lowerTeethPoints(),0.016,false,32));
  const tx=mouthX+(-0.035-mouthX)*width;
  const ty=mouthY+(-0.485-mouthY)*opening;
  tongue.position.set(tx,ty,0.215+headZ(tx,ty)-headZ(-0.035,-0.485));
  tongue.scale.set(0.145*width,0.055*opening,0.052);
  const lx=mouthX+(-0.207-mouthX)*width;
  const ly=mouthY+(-0.478-mouthY)*opening;
  littleTooth.position.set(lx,ly,0.329+headZ(lx,ly)-headZ(-0.207,-0.478));
  littleTooth.scale.set(0.025,0.055*(0.85+0.15*opening),0.024);
  lastOpening=opening;
  lastWidth=width;
}

function samplePose(t){
  const phase=((t%1220)+1220)%1220;
  let i=0;
  while(i<poseKeys.length-2&&phase>=poseKeys[i+1].t)i++;
  const a=poseKeys[i],b=poseKeys[i+1];
  const u=(phase-a.t)/(b.t-a.t);
  const s=u*u*(3-2*u),p={};
  for(const k of poseProperties)p[k]=a[k]+(b[k]-a[k])*s;
  return p;
}

function blendPoint(a,b,u){return a.clone().lerp(b,u);}
function poseArms(p){
  const c=p.c,sway=p.hx*0.12;
  const sl=leftArmRest[0].clone().applyMatrix4(torso.matrix);
  const sr=rightArmRest[0].clone().applyMatrix4(torso.matrix);
  const dl=sl.clone().sub(leftArmRest[0]),dr=sr.clone().sub(rightArmRest[0]);
  const follow=[1,0.9,0.7,0.4,0.2,0];
  const lRest=leftArmRest.map((v,i)=>v.clone().addScaledVector(dl,follow[i]));
  const rRest=rightArmRest.map((v,i)=>v.clone().addScaledVector(dr,follow[i]));
  const lSquat=[
    sl,sl.clone().add(V(-0.27,-0.24,0.02)),
    V(-1.17+sway,1.58,0.18),V(-1.18+sway,1.36,0.28),
    V(-1.05+sway,1.04,0.48),V(-0.94+sway,0.89,0.54)
  ];
  const rSquat=[
    sr,sr.clone().add(V(0.39,-0.20,0.02)),
    V(1.37+sway,1.67,0.13),V(1.40+sway,1.40,0.25),
    V(1.06+sway,1.04,0.48),V(0.82+sway,0.85,0.58)
  ];
  leftArm.pose(lRest.map((v,i)=>blendPoint(v,lSquat[i],c)));
  rightArm.pose(rRest.map((v,i)=>blendPoint(v,rSquat[i],c)));
  leftHand.position.copy(leftArm.end);
  rightHand.position.copy(rightArm.end);
  leftHand.rotation.set(-0.07+0.10*c,0,-0.035-0.10*c);
  rightHand.rotation.set(-0.07+0.10*c,0,0.09+0.02*c);
}

function render(t){
  const time=Number.isFinite(t)?t:0;
  const p=samplePose(time);
  character.rotation.set(0,p.yaw,0);

  torso.scale.set(1+0.07*p.c,p.sy,1+0.05*p.c);
  torso.rotation.set(p.pitch,0,p.roll);
  const scaledPivot=V(0,p.sy,0).applyQuaternion(torso.quaternion);
  torso.position.set(-scaledPivot.x,1+p.drop-scaledPivot.y,-scaledPivot.z);
  torso.updateMatrix();

  head.position.copy(headBase).applyMatrix4(torso.matrix);
  head.position.x+=p.hx;
  head.position.y+=p.lift;
  headEuler.set(p.rx,p.ry,p.rz,'XYZ');
  headLocalRotation.setFromEuler(headEuler);
  head.quaternion.copy(torso.quaternion).multiply(headLocalRotation);
  head.scale.set(1,1-0.08*p.c,1);
  poseMouth(p.open,p.width);

  poseArms(p);
  poseLegs(p);

  renderer.setRenderTarget(null);
  renderer.setViewport(0,0,216,216);
  renderer.setScissorTest(false);
  renderer.render(scene,ref);
}
