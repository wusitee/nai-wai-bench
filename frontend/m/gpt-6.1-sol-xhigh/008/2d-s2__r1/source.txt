  poses.push(poses[0]);

  const nodes = {
    body:el("torso-shell"),
    belly:el("belly"),
    leftLeg:el("left-leg"),
    rightLeg:el("right-leg"),
    head:el("head"),
    eyes:el("eyes"),
    nose:el("nose"),
    mouth:el("laughing-mouth"),
    leftHand:el("left-hand"),
    rightHand:el("right-hand"),
    shadow:el("ground-shadow")
  };

  const numberPattern = /-?(?:\d*\.)?\d+(?:e[-+]?\d+)?/gi;

  function captureArm(id){
    return Array.from(el(id).querySelectorAll("path")).map(path => {
      const source = path.getAttribute("d");
      const coordinates = source.match(numberPattern).map(Number);
      return {path,source,coordinates};
    });
  }

  const leftArmPaths = captureArm("left-arm");
  const rightArmPaths = captureArm("right-arm");

  function affine(node, values, ax, ay){
    const [cx,cy,sx,sy,shear] = values;
    node.setAttribute("transform",
      "matrix("+sx+" 0 "+shear+" "+sy+" "+
      (cx-sx*ax-shear*ay)+" "+(cy-sy*ay)+")");
  }

  function localTransform(node, values, ax, ay){
    const [dx,dy,sx,sy,angle] = values;
    node.setAttribute("transform",
      "translate("+dx+" "+dy+") translate("+ax+" "+ay+") "+
      "rotate("+angle+") scale("+sx+" "+sy+") "+
      "translate("+(-ax)+" "+(-ay)+")");
  }

  function warpArm(records, values, shoulderY, elbowY, wristY){
    for(const record of records){
      const mapped = [];
      const c = record.coordinates;
      for(let i=0;i<c.length;i+=2){
        const x=c[i], y=c[i+1];
        let a,b,u;
        if(y<=elbowY){
          a=0;
          b=2;
          u=Math.max(0,Math.min(1,(y-shoulderY)/(elbowY-shoulderY)));
        } else {
          a=2;
          b=4;
          u=Math.max(0,Math.min(1,(y-elbowY)/(wristY-elbowY)));
        }
        const dx=values[a]+(values[b]-values[a])*u;
        const dy=values[a+1]+(values[b+1]-values[a+1])*u;
        mapped.push(x+dx,y+dy);
      }
      let cursor=0;
      record.path.setAttribute("d",
        record.source.replace(numberPattern,()=>String(+mapped[cursor++].toFixed(3))));
    }
  }

  function handTransform(node, px, py, dx, dy, angle, sx, sy){
    node.setAttribute("transform",
      "translate("+(px+dx)+" "+(py+dy)+") rotate("+angle+") "+
      "scale("+sx+" "+sy+") translate("+(-px)+" "+(-py)+")");
  }

  function render(t){
    let time=Number(t);
    if(!Number.isFinite(time)) time=0;
    time=((time%1220)+1220)%1220;

    let index=0;
    while(index<times.length-2 && time>=times[index+1]) index++;

    let u=(time-times[index])/(times[index+1]-times[index]);
    u=u*u*(3-2*u);

    const p={};
    const first=poses[index], next=poses[index+1];
    for(const name of Object.keys(first)){
      p[name]=first[name].map((value,j)=>value+(next[name][j]-value)*u);
    }

    affine(nodes.body,p.body,124,186);
    affine(nodes.belly,p.belly,118,145);
    affine(nodes.leftLeg,p.leftLeg,98,194);
    affine(nodes.rightLeg,p.rightLeg,153,194);

    const h=p.head;
    nodes.head.setAttribute("transform",
      "translate("+h[0]+" "+h[1]+") rotate("+h[2]+") "+
      "scale("+h[3]+" "+h[4]+") translate(-124 -50)");

    localTransform(nodes.eyes,p.eyes,121,27);
    localTransform(nodes.mouth,p.mouth,119,36);

    const m=p.mouth;
    nodes.nose.setAttribute("transform",
      "translate("+(m[0]*.6)+" "+(m[1]*.65)+") translate(124 32) "+
      "scale("+(.55+.45*m[2])+" "+(.6+.4*m[3])+") translate(-124 -32)");

    warpArm(leftArmPaths,p.leftArm,70,123,174);
    warpArm(rightArmPaths,p.rightArm,70,128,179);

    const hands=p.hands;
    handTransform(nodes.leftHand,87,174,p.leftArm[4],p.leftArm[5],
      hands[0],hands[1],hands[2]);
    handTransform(nodes.rightHand,160,179,p.rightArm[4],p.rightArm[5],
      hands[3],hands[4],hands[5]);

    const crouch=Math.max(0,Math.min(1,(p.body[1]-186)/24));
    nodes.shadow.setAttribute("rx",String(72+8*crouch));
    nodes.shadow.setAttribute("opacity",String(.19+.025*crouch));
  }

  window.nwb={render:render};
  render(0);
})();
