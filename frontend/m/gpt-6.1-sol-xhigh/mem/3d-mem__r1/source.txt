function group(name,parent,position){
  const g=new THREE.Group();
  g.name=name;
  if(position)g.position.copy(position);
  parent.add(g);
  return g;
}
function mesh(geometry,material,parent,name){
  const m=new THREE.Mesh(geometry,material);
  m.name=name;
  m.castShadow=true;
  m.receiveShadow=true;
  parent.add(m);
  return m;
}
const sphereGeo=new THREE.SphereGeometry(1,40,28);
function ellipsoid(parent,name,pos,scale,material=yellow){
  const m=mesh(sphereGeo,material,parent,name);
  m.position.set(...pos);
  m.scale.set(...scale);
  return m;
}
function tube(parent,name,points,radius,material,closed=false){
  const curve=new THREE.CatmullRomCurve3(points,closed,'centripetal');
  return mesh(new THREE.TubeGeometry(curve,Math.max(32,points.length*5),radius,10,closed),material,parent,name);
}
function spike(parent,name,position,direction,height,radius){
  const g=group(name,parent,new THREE.Vector3(...position));
  g.quaternion.setFromUnitVectors(new THREE.Vector3(0,1,0),new THREE.Vector3(...direction).normalize());
  const profile=[
    new THREE.Vector2(0,0),
    new THREE.Vector2(radius,0),
    new THREE.Vector2(radius*.98,height*.12),
    new THREE.Vector2(radius*.75,height*.40),
    new THREE.Vector2(radius*.36,height*.75),
    new THREE.Vector2(radius*.12,height*.94),
    new THREE.Vector2(0,height)
  ];
  mesh(new THREE.LatheGeometry(profile,24),crestMat,g,'rounded-spine');
  return g;
}

function bodyRadius(y){
  for(let i=1;i<bodySamples.length;i++){
    const a=bodySamples[i-1],b=bodySamples[i];
    if(y>=a.y&&y<=b.y){
      const t=(y-a.y)/(b.y-a.y);
      return THREE.MathUtils.lerp(a.x,b.x,t);
    }
  }
  return .3;
}

function taperedTube(curve,radiusAt,segments=50,sides=24){
  const frames=curve.computeFrenetFrames(segments,false);
  const vertices=[],indices=[];
  for(let j=0;j<=segments;j++){
    const t=j/segments,p=curve.getPointAt(t),r=radiusAt(t);
    for(let i=0;i<sides;i++){
      const a=i/sides*Math.PI*2;
      const q=p.clone()
        .addScaledVector(frames.normals[j],Math.cos(a)*r)
        .addScaledVector(frames.binormals[j],Math.sin(a)*r);
      vertices.push(q.x,q.y,q.z);
    }
  }
  for(let j=0;j<segments;j++)for(let i=0;i<sides;i++){
    const a=j*sides+i,b=j*sides+(i+1)%sides,c=(j+1)*sides+i,d=(j+1)*sides+(i+1)%sides;
    indices.push(a,b,d,a,d,c);
  }
  const start=curve.getPointAt(0),end=curve.getPointAt(1);
  const si=vertices.length/3;
  vertices.push(start.x,start.y,start.z,end.x,end.y,end.z);
  for(let i=0;i<sides;i++){
    indices.push(si,(i+1)%sides,i);
    indices.push(si+1,segments*sides+i,segments*sides+(i+1)%sides);
  }
  const geo=new THREE.BufferGeometry();
  geo.setAttribute('position',new THREE.Float32BufferAttribute(vertices,3));
  geo.setIndex(indices);
  geo.computeVertexNormals();
  return geo;
}

const HY=2.46,HW=1.075,HH=.79,HD=.73;
const head=group('head',character,new THREE.Vector3(0,HY,0));
function faceZ(x,y){
  const n=Math.sqrt(Math.max(0,1-(x/HW)**2-((y-HY)/HH)**2));
  const muzzle=.185*Math.exp(-((y-2.39)/.34)**2)*(1-.40*(x/HW)**2);
  const cheeks=.055*Math.exp(-((Math.abs(x)-.79)/.24)**2-((y-2.37)/.29)**2);
  const brows=.045*Math.exp(-((Math.abs(x)-.48)/.28)**2-((y-2.88)/.20)**2);
  return n*(HD+muzzle+cheeks+brows);
}
function rimPoint(a){
  const s=Math.sin(a);
  const x=.83*Math.cos(a);
  const y=s>=0?2.43-.12*Math.pow(s,.85):2.43-.64*Math.pow(-s,.82);
  return new THREE.Vector3(x,y-HY,faceZ(x,y));
}
function onFace(x,y,offset=.025){
  return new THREE.Vector3(x,y-HY,faceZ(x,y)+offset);
}

/* The head is a closed volume with a real opening and recessed oral cavity. */
{
  const vertices=[],indices=[];
  const S=128,F=38,B=28;
  for(let j=0;j<=F;j++){
    const u=Math.sin(j/F*Math.PI/2);
    for(let i=0;i<S;i++){
      const a=i/S*Math.PI*2,p=rimPoint(a);
      const x=THREE.MathUtils.lerp(p.x,HW*Math.cos(a),u);
      const y=THREE.MathUtils.lerp(p.y+HY,HY+HH*Math.sin(a),u);
      vertices.push(x,y-HY,faceZ(x,y));
    }
  }
  for(let j=1;j<B;j++){
    const a=j/B*Math.PI/2,r=Math.cos(a);
    for(let i=0;i<S;i++){
      const t=i/S*Math.PI*2;
      vertices.push(HW*r*Math.cos(t),HH*r*Math.sin(t),-HD*Math.sin(a));
    }
  }
  const rings=F+B;
  for(let j=0;j<rings-1;j++)for(let i=0;i<S;i++){
    const a=j*S+i,b=(j+1)*S+i,c=(j+1)*S+(i+1)%S,d=j*S+(i+1)%S;
    indices.push(a,b,c,a,c,d);
  }
  const pole=vertices.length/3;
  vertices.push(0,0,-HD);
  for(let i=0;i<S;i++)indices.push((rings-1)*S+i,pole,(rings-1)*S+(i+1)%S);
  const geo=new THREE.BufferGeometry();
  geo.setAttribute('position',new THREE.Float32BufferAttribute(vertices,3));
  geo.setIndex(indices);
  geo.computeVertexNormals();
  mesh(geo,yellow,head,'sculpted-wide-head');
}
