<script>
(function(){
  var T = [0,200,400,600,800,1000,1200,1400,1500];
  // parameter tracks (last value = loop back to start)
  var K = {
    lRot:  [0,0.5,2,5,8,10,11,12,0],
    lEyeX: [0,1,3,6,8,10,10,11,0],
    lEyeY: [0,0,-1,-2,-3,-3,-3,-3,0],
    lHead: [0,0,1,2,3,3,4,4,0],
    rRot:  [0,0,-1,-3,-4,-5,-5,-5.5,0],
    rDY:   [0,0,1,3,4,5,5,6,0],
    rDX:   [0,0,-1,-2,-3,-4,-4,-5,0],
    rEyeX: [0,0,-1,-2,-2,-3,-3,-3,0],
    rArm:  [0,0,-1,-2,-3,-3,-4,-4,0]
  };
  function val(arr, t){
    for (var i=0;i<T.length-1;i++){
      if (t>=T[i] && t<=T[i+1]){
        var u=(t-T[i])/(T[i+1]-T[i]);
        u=u*u*(3-2*u);
        return arr[i]+(arr[i+1]-arr[i])*u;
      }
    }
    return arr[0];
  }
  var el = function(id){return document.getElementById(id);};
  var lUpper=el('left-upper'), lHead=el('left-head'), lEyes=el('left-eyes'),
      rUpper=el('right-upper'), rEyes=el('right-eyes'), rArm=el('right-arm-front'),
      lMouth=el('left-mouth');
  function render(t){
    t = ((t % 1500) + 1500) % 1500;
    var lr=val(K.lRot,t), ex=val(K.lEyeX,t), ey=val(K.lEyeY,t), lh=val(K.lHead,t),
        rr=val(K.rRot,t), rdy=val(K.rDY,t), rdx=val(K.rDX,t), rex=val(K.rEyeX,t), ra=val(K.rArm,t);
    lUpper.setAttribute('transform','rotate('+(lr*0.35).toFixed(3)+' 65 225)');
    lHead.setAttribute('transform','rotate('+(lr*0.65).toFixed(3)+' 67 120) translate('+(lh*0.5).toFixed(3)+' 0)');
    lEyes.setAttribute('transform','translate('+ex.toFixed(3)+' '+ey.toFixed(3)+')');
    lMouth.setAttribute('transform','translate('+(ex*0.8).toFixed(3)+' '+(ey*0.6).toFixed(3)+')');
    rUpper.setAttribute('transform','translate('+rdx.toFixed(3)+' '+rdy.toFixed(3)+') rotate('+rr.toFixed(3)+' 190 225)');
    rEyes.setAttribute('transform','translate('+rex.toFixed(3)+' 0)');
    rArm.setAttribute('transform','rotate('+ra.toFixed(3)+' 103 182) translate('+(rdx*0.3).toFixed(3)+' '+(rdy*0.3).toFixed(3)+')');
  }
  window.nwb = { render: render };
  render(0);
})();
</script>
