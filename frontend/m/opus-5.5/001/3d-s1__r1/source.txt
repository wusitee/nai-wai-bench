const sphereGeo = new THREE.SphereGeometry(1, 40, 28);
function ell(rx, ry, rz, mat, name) {
  const m = new THREE.Mesh(sphereGeo, mat);
  m.scale.set(rx, ry, rz);
  if (name) m.name = name;
  return m;
}
function limb(a, b, r, mat, name) {
  const A = new THREE.Vector3(...a), B = new THREE.Vector3(...b);
  const dir = B.clone().sub(A);
  const len = dir.length();
  const g = new THREE.CapsuleGeometry(r, len, 8, 20);
  const m = new THREE.Mesh(g, mat);
  m.position.copy(A).add(B).multiplyScalar(0.5);
  m.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), dir.normalize());
  if (name) m.name = name;
  return m;
}
function makeEye(name, r) {
  const g = new THREE.Group(); g.name = name;
  const e = ell(r, r * 1.08, r * 0.8, black); g.add(e);
  const h = ell(r * 0.28, r * 0.28, r * 0.2, white);
  h.position.set(-r * 0.35, r * 0.4, r * 0.7); g.add(h);
  return g;
}
function makeMouth(name, radius, tube) {
  const g = new THREE.Group(); g.name = name;
  const t = new THREE.Mesh(new THREE.TorusGeometry(radius, tube, 8, 20, Math.PI * 0.8), mouthMat);
  t.rotation.z = Math.PI + Math.PI * 0.1;
  g.add(t);
  return g;
}

function pageTex(seed) {
  const c = document.createElement('canvas'); c.width = 64; c.height = 64;
  const x = c.getContext('2d');
  x.fillStyle = '#f7f5ef'; x.fillRect(0, 0, 64, 64);
  x.strokeStyle = '#b9c8dc'; x.lineWidth = 1;
  for (let i = 6; i < 64; i += 5) { x.beginPath(); x.moveTo(2, i); x.lineTo(62, i); x.stroke(); }
  x.strokeStyle = '#555'; 
  let s = seed;
  const rnd = () => { s = (s * 9301 + 49297) % 233280; return s / 233280; };
  for (let i = 6; i < 50; i += 5) {
    x.beginPath(); x.moveTo(6, i - 1);
    x.lineTo(10 + rnd() * 45, i - 1); x.stroke();
  }
  const t = new THREE.CanvasTexture(c);
  t.colorSpace = THREE.SRGBColorSpace;
  return t;
}

function openBook(name, x, z, rotY, w, d, matL, matR) {
  const g = new THREE.Group(); g.name = name;
  const geo = new THREE.BoxGeometry(w / 2, 0.03, d);
  const mats = m => [paperSide, paperSide, m, paperSide, paperSide, paperSide];
  const L = new THREE.Mesh(geo, mats(matL)); L.position.set(-w / 4, 0.02, 0); L.rotation.z = 0.06; g.add(L);
  const R = new THREE.Mesh(geo, mats(matR)); R.position.set(w / 4, 0.02, 0); R.rotation.z = -0.06; g.add(R);
  g.position.set(x, 0, z); g.rotation.y = rotY;
  props.add(g);
  return g;
}

function pencil(name, a, b, parent) {
  const g = new THREE.Group(); g.name = name;
  const A = new THREE.Vector3(...a), B = new THREE.Vector3(...b);
  const dir = B.clone().sub(A); const len = dir.length();
  const body = new THREE.Mesh(new THREE.CylinderGeometry(0.025, 0.025, len * 0.85, 8), new THREE.MeshStandardMaterial({ color: 0xe8b04a }));
  body.position.y = len * 0.075; g.add(body);
  const tip = new THREE.Mesh(new THREE.ConeGeometry(0.025, len * 0.15, 8), new THREE.MeshStandardMaterial({ color: 0x4a4a4a }));
  tip.rotation.x = Math.PI; tip.position.y = -len * 0.425; g.add(tip);
  g.position.copy(A).add(B).multiplyScalar(0.5);
  g.quaternion.setFromUnitVectors(new THREE.Vector3(0, -1, 0), dir.normalize());
  (parent || props).add(g);
  return g;
}
