function limb(a, b, r, mat, name) {
  const A = new THREE.Vector3(...a), B = new THREE.Vector3(...b);
  const d = B.clone().sub(A);
  const len = d.length();
  const m = new THREE.Mesh(new THREE.CapsuleGeometry(r, len, 6, 14), mat);
  m.position.copy(A).add(B).multiplyScalar(0.5);
  m.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), d.normalize());
  if (name) m.name = name;
  return m;
}

const SEG = 0.62;
function makeArm(name, shoulder, side) {

  return { arm, upper, fore, hand };
}

const DOWN = new THREE.Vector3(0, -1, 0);
const _T = new THREE.Vector3(), _dir = new THREE.Vector3(), _pole = new THREE.Vector3(),
  _E = new THREE.Vector3(), _Tc = new THREE.Vector3(), _tmp = new THREE.Vector3();
function poseArm(rig, w, p, he) {
  const S = rig.arm.position;
  _T.set(w[0] - S.x, w[1] - S.y, w[2] - S.z);
  let d = _T.length();
  _dir.copy(_T).divideScalar(d > 1e-6 ? d : 1);
  if (d < 1e-6) _dir.set(0, -1, 0);
  d = Math.min(Math.max(d, 0.15), 2 * SEG * 0.999);
  const half = d / 2, h = Math.sqrt(Math.max(0, SEG * SEG - half * half));
  _pole.set(p[0], p[1], p[2]);
  _pole.addScaledVector(_dir, -_pole.dot(_dir));
  if (_pole.lengthSq() < 1e-8) { _pole.set(0, 0, -1).cross(_dir); }
  _pole.normalize();
  _E.copy(_dir).multiplyScalar(half).addScaledVector(_pole, h);
  _Tc.copy(_dir).multiplyScalar(d);
  rig.upper.position.set(0, 0, 0);
  rig.upper.quaternion.setFromUnitVectors(DOWN, _tmp.copy(_E).normalize());
  rig.fore.position.copy(_E);
  rig.fore.quaternion.setFromUnitVectors(DOWN, _tmp.copy(_Tc).sub(_E).normalize());
  rig.hand.position.copy(_Tc);
  rig.hand.rotation.set(he[0], he[1], he[2]);
}

function lerpV(a, b, u) {
  if (Array.isArray(a)) return a.map((v, i) => v + (b[i] - v) * u);
  return a + (b - a) * u;
}
function sample(t) {
  t = ((t % P) + P) % P;
  let i = K.length - 1;
  for (let k = 0; k < K.length - 1; k++) { if (t < K[k + 1].t) { i = k; break; } }
  const a = K[i];
  const last = i === K.length - 1;
  const b = last ? K[0] : K[i + 1];
  const t1 = last ? P : b.t;
  let u = (t - a.t) / (t1 - a.t);
  u = Math.min(1, Math.max(0, u));
  u = u * u * (3 - 2 * u);
  const out = {};
  for (const k in a) { if (k !== 't') out[k] = lerpV(a[k], b[k], u); }
  return out;
}

function render(t) {
  const s = sample(t);
  camera.position.set(s.cam[0], s.cam[1], s.cam[2]);
  camera.rotation.set(0, 0, 0);
  camera.updateMatrixWorld();

  bird.rotation.set(0, s.ry, 0);
  head.scale.set(s.hs[0], s.hs[1], s.hs[2]);

  const eo = Math.min(1, Math.max(0, s.eo));
  eyesClosed.visible = eo < 0.5;
  eyesOpen.visible = eo >= 0.5;
  for (const g of openEyes) g.scale.set(1 / s.hs[0], Math.max(0.05, eo) / s.hs[1], 1 / s.hs[2]);

  const sm = Math.min(1, Math.max(0, s.sm));
  smile.visible = sm > 0.02;
  mouth.scale.set(Math.max(sm, 0.001) / s.hs[0], Math.max(sm, 0.001) / s.hs[1], 1 / s.hs[2]);

  const sn = Math.min(1, Math.max(0, s.sn));
  snack.visible = sn > 0.02;
  snack.scale.setScalar(Math.max(sn, 0.001));

  poseArm(armL, s.lw, s.lp, s.lh);
  poseArm(armR, s.rw, s.rp, s.rh);

  scene.updateMatrixWorld(true);
  renderer.render(scene, camera);
}
