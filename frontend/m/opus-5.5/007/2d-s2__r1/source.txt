<script>
(function(){
  var P=5540;
  function $(id){return document.getElementById(id);}
  function ease(u){return u*u*(3-2*u);}
  function clone(o,t){var r={};for(var k in o)r[k]=o[k];r.t=t;return r;}
  var F0={t:0,s:1,tx:0,ty:0,lu:0,lf:0,ru:0,rf:0,eye:0,smile:0,obj:0};
  var F1={t:800,s:0.97,tx:-16,ty:21,lu:-45,lf:-146,ru:45,rf:146,eye:1,smile:0,obj:1};
  var F2={t:1530,s:1.05,tx:-14,ty:8,lu:30,lf:-175,ru:-10,rf:-130,eye:1,smile:1,obj:0};
  var F3={t:2330,s:1.3,tx:-15,ty:-42,lu:0,lf:0,ru:0,rf:0,eye:1,smile:0,obj:0};
  var F4={t:3130,s:1.15,tx:-25,ty:-13,lu:-20,lf:-143,ru:0,rf:0,eye:1,smile:1,obj:0};
  var FK=[F0,F1,F2,F3,F4,clone(F4,4300),clone(F0,4301),clone(F0,5540)];
  var S6={dx:0,dy:0,a1:107,a2:117};
  var S7={dx:-3,dy:3,a1:95,a2:99};
  var S8={dx:-5,dy:-3,a1:-5,a2:-5};
  var SK=[clone(S6,0),clone(S6,3930),clone(S7,4670),clone(S8,5470),clone(S8,5540)];
  function interp(keys,t){
    for(var i=0;i<keys.length-1;i++){
      var a=keys[i],b=keys[i+1];
      if(t>=a.t&&t<=b.t){
        var u=b.t>a.t?ease((t-a.t)/(b.t-a.t)):0,r={};
        for(var k in a){if(k!=='t')r[k]=a[k]+(b[k]-a[k])*u;}
        return r;
      }
    }
    return keys[keys.length-1];
  }
  function frontW(t){
    if(t<3480)return 1;
    if(t<3580)return 1-(t-3480)/100;
    if(t<5480)return 0;
    if(t<5530)return (t-5480)/50;
    return 1;
  }
  function render(t){
    t=((t%P)+P)%P;
    var f=interp(FK,t),s=interp(SK,t),w=frontW(t);
    $('character-front').setAttribute('transform','translate('+f.tx+' '+f.ty+') scale('+f.s+')');
    $('character-front').setAttribute('opacity',w);
    $('arm-left').setAttribute('transform','rotate('+f.lu+' 66 85)');
    $('forearm-left').setAttribute('transform','rotate('+f.lf+' 50 140)');
    $('arm-right').setAttribute('transform','rotate('+f.ru+' 140 85)');
    $('forearm-right').setAttribute('transform','rotate('+f.rf+' 155 140)');
    $('eyes-open').setAttribute('opacity',f.eye);
    $('eyes-closed').setAttribute('opacity',1-f.eye);
    $('mouth').setAttribute('opacity',f.smile);
    $('held-object').setAttribute('opacity',f.obj);
    $('character-side').setAttribute('opacity',1-w);
    $('character-side').setAttribute('transform','translate('+s.dx+' '+s.dy+')');
    $('side-arm-near').setAttribute('transform','rotate('+s.a1+' 118 95)');
    $('side-arm-far').setAttribute('transform','rotate('+s.a2+' 112 90)');
  }
  window.nwb={render:render};
  render(0);
})();
