(function () {
  var P = 1220;
  var T    = [0, 180, 340, 520, 680, 860, 1020, 1200, 1220];
  var C    = [0, 1,   0,   0.1, 1,   0,   0.05, 1,    0];
  var TILT = [0, 20,  -12, -2,  22,  -12, 0,    24,   0];
  var TX   = [0, 5,   -2,  0,   6,   -2,  0,    7,    0];
  var ch = document.getElementById('character-creature');
  var head = document.getElementById('head');
  var mouth = document.getElementById('mouth');

  function interp(arr, t) {
    for (var i = 0; i < T.length - 1; i++) {
      if (t >= T[i] && t <= T[i + 1]) {
        var u = (t - T[i]) / (T[i + 1] - T[i]);
        u = u * u * (3 - 2 * u);
        return arr[i] + (arr[i + 1] - arr[i]) * u;
      }
    }
    return arr[0];
  }

  function render(t) {
    t = Number(t) || 0;
    t = ((t % P) + P) % P;
    var c = interp(C, t), tilt = interp(TILT, t), tx = interp(TX, t);
    var sx = 1 + 0.12 * c, sy = 1 - 0.2 * c;
    ch.setAttribute('transform',
      'translate(' + (120 + tx) + ' 216) scale(' + sx + ' ' + sy + ') translate(-120 -216)');
    head.setAttribute('transform',
      'translate(122 74) scale(' + (1 / sx) + ' ' + (1 / sy) + ') rotate(' + tilt + ') translate(-122 -74)');
    var my = 1 - 0.15 * c;
    mouth.setAttribute('transform',
      'translate(119 34) scale(1 ' + my + ') translate(-119 -34)');
  }

  window.nwb = { render: render };
  render(0);
})();
