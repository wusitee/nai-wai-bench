<script>
(function(){
  // [t, bodyDx, bodyDy, headDx, headDy, LhX, LhY, RhX, RhY]
  var K = [
    [0,    0,0, 0,0,  70,90,  212,80],
    [160,  2,3, 0,4,  18,72,  158,95],
    [340,  0,4, 2,8,  58,102, 210,92],
    [500,  4,6, 6,12, 20,103, 185,124],
    [680,  3,4, 4,6,  18,95,  178,108],
    [840,  0,1, 0,0,  65,95,  212,92],
    [1020, 3,2, 4,2,  15,85,  170,110],
    [1180, 0,4, 0,6,  50,126, 210,110],
    [1200, 0,0, 0,0,  70,90,  212,80]
  ];
  var body = document.getElementById('body');
  var head = document.getElementById('head');
  var limbL = document.getElementById('arm-left-limb');
  var limbR = document.getElementById('arm-right-limb');
  var handL = document.getElementById('hand-left');
  var handR = document.getElementById('hand-right');

  function f(n){ return (Math.round(n*100)/100).toString(); }

  function sample(t){
    var i = 0;
    while (i < K.length - 2 && t >= K[i+1][0]) i++;
    var a = K[i], b = K[i+1];
    var u = (t - a[0]) / (b[0] - a[0]);
    if (u < 0) u = 0; if (u > 1) u = 1;
    u = u*u*(3 - 2*u);
    var out = [];
    for (var j = 1; j < a.length; j++) out.push(a[j] + (b[j] - a[j]) * u);
    return out;
  }

  function arm(limb, hand, sx, sy, hx, hy){
    var ang = Math.atan2(hy - sy, hx - sx);
    var ex = hx - Math.cos(ang) * 6, ey = hy - Math.sin(ang) * 6;
    var cx = (sx + ex) / 2, cy = (sy + ey) / 2 + 8;
    limb.setAttribute('d', 'M' + f(sx) + ' ' + f(sy) + ' Q' + f(cx) + ' ' + f(cy) + ' ' + f(ex) + ' ' + f(ey));
    hand.setAttribute('transform', 'translate(' + f(hx) + ' ' + f(hy) + ') rotate(' + f(ang * 180 / Math.PI) + ')');
  }

  function render(t){
    t = Number(t) || 0;
    t = ((t % 1200) + 1200) % 1200;
    var p = sample(t);
    var bx = p[0], by = p[1], hdx = p[2], hdy = p[3];
    var sy = 1 - hdy / 222;
    var sx = 1 + hdy / 400;
    body.setAttribute('transform',
      'translate(' + f(bx) + ' 0) translate(125 226) scale(' + f(sx) + ' ' + f(sy) + ') translate(-125 -226)');
    head.setAttribute('transform', 'translate(' + f(bx + hdx) + ' ' + f(hdy) + ')');
    var shoulderDy = hdy * 0.5 + by * 0.2;
    arm(limbL, handL, 92 + bx, 92 + shoulderDy, p[4], p[5]);
    arm(limbR, handR, 162 + bx, 88 + shoulderDy, p[6], p[7]);
  }

  window.nwb = { render: render };
  render(0);
})();
</script>
