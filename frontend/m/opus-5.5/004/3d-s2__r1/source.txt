<script type="module">
import * as THREE from 'three';

const W = 244, H = 240;
const renderer = new THREE.WebGLRenderer({ preserveDrawingBuffer: true, antialias: true });
renderer.setPixelRatio(1);
renderer.setSize(W, H);
document.body.appendChild(renderer.domElement);

const scene = new THREE.Scene();
scene.background = new THREE.Color(0xfbfbfb);

const camera = new THREE.PerspectiveCamera(30, W / H, 0.1, 100);
camera.position.set(0, 0, 10);
camera.lookAt(0, 0, 0);

scene.add(new THREE.HemisphereLight(0xffffff, 0xd8d0c0, 1.6));
const key = new THREE.DirectionalLight(0xffffff, 1.6);
key.position.set(1, 2, 6);
scene.add(key);
const back = new THREE.DirectionalLight(0xffffff, 0.8);
back.position.set(-2, 3, -5);
scene.add(back);

const floor = new THREE.Mesh(
  new THREE.PlaneGeometry(40, 40),
  new THREE.MeshStandardMaterial({ color: 0xf2f2f0, roughness: 1 })
);
floor.rotation.x = -Math.PI / 2;
floor.position.y = -3.0;
floor.name = 'floor';
scene.add(floor);

const yellow = new THREE.MeshStandardMaterial({ color: 0xf6c814, roughness: 0.55 });
const cream = new THREE.MeshStandardMaterial({ color: 0xf6dfc6, roughness: 0.6 });
const brown = new THREE.MeshStandardMaterial({ color: 0x5e3d1e, roughness: 0.6 });
const orange = new THREE.MeshStandardMaterial({ color: 0xe8960f, roughness: 0.5 });
const black = new THREE.MeshStandardMaterial({ color: 0x111111, roughness: 0.3 });

const BASEX = 0.0;
const character = new THREE.Group();
character.name = 'character-chick';
character.position.set(BASEX, 0, 0);
scene.add(character);

const prof = [
  [0, -3.0], [0.9, -2.95], [1.35, -2.75], [1.56, -2.2], [1.62, -1.5], [1.6, -1.0],
  [1.5, -0.3], [1.3, 0.25], [1.12, 0.9], [1.0, 1.4], [0.92, 1.8], [0.82, 2.1],
  [0.62, 2.38], [0.35, 2.55], [0, 2.6]
].map(p => new THREE.Vector2(p[0], p[1]));
const bodyGroup = new THREE.Group();
bodyGroup.name = 'body';
character.add(bodyGroup);
const body = new THREE.Mesh(new THREE.LatheGeometry(prof, 48), yellow);
bodyGroup.add(body);

const belly = new THREE.Mesh(new THREE.SphereGeometry(1, 40, 30), cream);
belly.name = 'belly';
belly.scale.set(1.1, 1.45, 0.55);
belly.position.set(0.05, -0.7, 1.05);
bodyGroup.add(belly);

const head = new THREE.Group();
head.name = 'head';
head.position.set(0, 1.8, 0);
character.add(head);

const eyes = new THREE.Group();
eyes.name = 'eyes';
head.add(eyes);
const eyeGeo = new THREE.SphereGeometry(0.07, 16, 12);
const eyeL = new THREE.Mesh(eyeGeo, black);
eyeL.name = 'eye-left';
eyeL.position.set(-0.32, 0.4, 0.69);
eyes.add(eyeL);
const eyeR = new THREE.Mesh(eyeGeo, black);
eyeR.name = 'eye-right';
eyeR.position.set(0.32, 0.4, 0.69);
eyes.add(eyeR);

const mouth = new THREE.Group();
mouth.name = 'mouth';
head.add(mouth);
const beak = new THREE.Mesh(new THREE.SphereGeometry(1, 24, 16), orange);
beak.name = 'beak';
beak.scale.set(0.34, 0.16, 0.22);
beak.position.set(0, -0.13, 0.9);
mouth.add(beak);

// Arms: rebuilt orientation each frame from shoulder to hand target
const L0 = 1.0;
const UP = new THREE.Vector3(0, 1, 0);
function makeArm(name, shoulder) {
  const g = new THREE.Group();
  g.name = name;
  g.position.copy(shoulder);
  character.add(g);
  const armMesh = new THREE.Mesh(new THREE.CapsuleGeometry(0.26, L0, 8, 16), yellow);
  armMesh.name = name + '-mesh';
  g.add(armMesh);
  const hg = new THREE.Group();
  hg.name = name.replace('arm', 'hand');
  g.add(hg);
  const h = new THREE.Mesh(new THREE.SphereGeometry(1, 20, 16), brown);
  h.scale.set(0.26, 0.32, 0.24);
  hg.add(h);
  return { g, armMesh, hg, shoulder: shoulder.clone() };
}
const armL = makeArm('arm-left', new THREE.Vector3(-1.1, -0.25, 0.3));
const armR = makeArm('arm-right', new THREE.Vector3(1.35, -0.15, 0.3));

const tmpDir = new THREE.Vector3();
function poseArm(arm, target) {
  tmpDir.subVectors(target, arm.shoulder);
  const len = Math.max(0.3, tmpDir.length());
  arm.g.quaternion.setFromUnitVectors(UP, tmpDir.clone().normalize());
  arm.armMesh.position.set(0, len / 2, 0);
  arm.armMesh.scale.set(1, len / L0 * 0.95, 1);
  arm.hg.position.set(0, len, 0);
}

// Keyframes: t, left hand px, right hand px, head yaw, body x px, roll
const PX = 44.8;
const K = [
  [0,    68, 90, 213, 78,  -0.35,  0,   0.0],
  [160,  15, 72, 158, 95,   0.0,  -4,   0.03],
  [340,  55,100, 210, 92,   0.0,   0,   0.0],
  [500,  18,105, 185,125,  -0.1,   2,   0.0],
  [680,  18,100, 175,108,   0.25,  2,  -0.03],
  [840,  60, 95, 210, 90,  -0.4, -12,   0.06],
  [1020, 15, 90, 170,108,   0.3,   0,  -0.02],
  [1180, 40,108, 205,105,   0.0,  -4,   0.02],
];
const PERIOD = 1200;

function sample(t) {
  t = ((t % PERIOD) + PERIOD) % PERIOD;
  let i = K.length - 1;
  for (let k = 0; k < K.length; k++) if (K[k][0] <= t) i = k;
  const a = K[i];
  const b = i + 1 < K.length ? K[i + 1] : K[0];
  const tb = i + 1 < K.length ? b[0] : PERIOD;
  let u = (t - a[0]) / (tb - a[0]);
  u = 0.5 - 0.5 * Math.cos(Math.PI * u);
  const out = [];
  for (let j = 1; j < a.length; j++) out.push(a[j] + (b[j] - a[j]) * u);
  return out;
}

const vL = new THREE.Vector3(), vR = new THREE.Vector3();
function render(t) {
  const s = sample(t);
  const [lx, ly, rx, ry, yaw, bx, roll] = s;
  const cx = BASEX + bx / PX;
  character.position.set(cx, 0, 0);
  character.rotation.set(0, 0, roll);
  head.rotation.set(0, yaw, 0);
  const cr = Math.cos(-roll), sr = Math.sin(-roll);
  function toLocal(px, py, v) {
    const wx = (px - 122) / PX - cx;
    const wy = (120 - py) / PX;
    v.set(wx * cr - wy * sr, wx * sr + wy * cr, 0.6);
    return v;
  }
  poseArm(armL, toLocal(lx, ly, vL));
  poseArm(armR, toLocal(rx, ry, vR));
  character.updateMatrixWorld(true);
  renderer.render(scene, camera);
}
render(0);

const characters = [character];
window.nwb = { scene, renderer, cameras: { ref: camera }, characters, render, ready: true };
</script>
