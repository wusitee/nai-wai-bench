function animatedLimb(name,parent,points,radii){
  const original=points.map(p=>new THREE.Vector3(...p));
  const segments=48,sides=32;
  const verts=new Float32Array((segments+1)*(sides+1)*3);
  const indices=[];
  for(let i=0;i<segments;i++){
    for(let k=0;k<sides;k++){
      const b=i*(sides+1)+k;
      indices.push(b,b+sides+1,b+1,b+1,b+sides+1,b+sides+2);
    }
  }
  const geo=new THREE.BufferGeometry();
  geo.setAttribute('position',new THREE.BufferAttribute(verts,3).setUsage(THREE.DynamicDrawUsage));
  geo.setIndex(indices);
  const mesh=new THREE.Mesh(geo,gold);
  mesh.name=name;
  mesh.castShadow=true;
  mesh.receiveShadow=true;
  mesh.frustumCulled=false;
  parent.add(mesh);

  const handWeights=[0,0.04,0.25,0.68,1];
  const elbowWeights=[0,0.68,1,0.52,0];
  const n=new THREE.Vector3();
  function pose(handOffset,elbowOffset){
    const pts=original.map((p,i)=>p.clone()
      .addScaledVector(handOffset,handWeights[i])
      .addScaledVector(elbowOffset,elbowWeights[i]));
    const path=new THREE.CatmullRomCurve3(pts);
    const frames=path.computeFrenetFrames(segments,false);
    for(let i=0;i<=segments;i++){
      const t=i/segments;
      const p=path.getPointAt(t);
      const q=t*(radii.length-1);
      const j=Math.min(radii.length-2,Math.floor(q));
      const r=THREE.MathUtils.lerp(radii[j],radii[j+1],q-j);
      for(let k=0;k<=sides;k++){
        const a=k/sides*Math.PI*2;
        n.copy(frames.normals[i]).multiplyScalar(Math.cos(a))
          .addScaledVector(frames.binormals[i],Math.sin(a));
        const index=(i*(sides+1)+k)*3;
        verts[index]=p.x+n.x*r;
        verts[index+1]=p.y+n.y*r;
        verts[index+2]=p.z+n.z*r;
      }
    }
    geo.attributes.position.needsUpdate=true;
    geo.computeVertexNormals();
  }
  pose(new THREE.Vector3(),new THREE.Vector3());
  return {mesh,pose};
}

function mitten(name,parent,pos,scale,rotation){
  const hand=group(name,parent);
  hand.position.set(...pos);
  hand.rotation.z=rotation;
  const geo=new THREE.SphereGeometry(1,48,36);
  const attr=geo.attributes.position;
  const cs=[];
  const bottom=new THREE.Color('#957027');
  const top=new THREE.Color('#4d431d');
  for(let i=0;i<attr.count;i++){
    const c=bottom.clone().lerp(top,THREE.MathUtils.smoothstep(attr.getY(i),-0.7,0.7));
    cs.push(c.r,c.g,c.b);
  }
  geo.setAttribute('color',new THREE.Float32BufferAttribute(cs,3));
  const mesh=new THREE.Mesh(geo,new THREE.MeshStandardMaterial({
    vertexColors:true,roughness:0.96
  }));
  mesh.name='soft-brown-mitten';
  mesh.scale.set(...scale);
  mesh.castShadow=true;
  mesh.receiveShadow=true;
  hand.add(mesh);
  return hand;
}

function render(t){
  const time=((t%1200)+1200)%1200;
  let i=0;
  while(i<poses.length-2 && time>=poses[i+1].t)i++;
  const a=poses[i],b=poses[i+1];
  let u=(time-a.t)/(b.t-a.t);
  u=u*u*(3-2*u);
  const value=k=>THREE.MathUtils.lerp(a[k],b[k],u);

  const bob=value('bob');
  const sway=value('sway');

  // Squash around the cropped lower body, keeping its weight grounded.
  const squash=1+bob/3.50;
  torso.position.y=-1.70*(1-squash);
  torso.scale.set(1+(1-squash)*0.11,squash,1);
  torso.rotation.z=sway*0.28;
  head.position.set(sway,bob,0);
  head.rotation.z=-sway*0.45;

  lh.set(value('lx'),value('ly'),value('lz'));
  rh.set(value('rx'),value('ry'),value('rz'));
  le.set(value('lex'),value('ley'),value('lz')*0.22);
  re.set(value('rex'),value('rey'),value('rz')*0.22);

  leftLimb.pose(lh,le);
  rightLimb.pose(rh,re);

  leftElbow.position.copy(leftElbowBase).add(le);
  rightElbow.position.copy(rightElbowBase).add(re);
  leftHand.position.copy(leftHandBase).add(lh);
  rightHand.position.copy(rightHandBase).add(rh);

  leftHand.rotation.set(value('lz')*0.7,-value('lx')*0.16,value('la'));
  rightHand.rotation.set(-value('rz')*0.35,-value('rx')*0.40,value('ra'));

  renderer.render(scene,ref);
}
