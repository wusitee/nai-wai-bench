function ellipsoid(name,parent,material,position,scale){

}

function taperedTube(name,parent,points,radii,material,segments=44,sides=16){
  const curve=new THREE.CatmullRomCurve3(points.map(p=>new THREE.Vector3(...p)));
  const frames=curve.computeFrenetFrames(segments,false);
  const vertices=[],normals=[],indices=[];
  for(let i=0;i<=segments;i++){
    const t=i/segments,p=curve.getPointAt(t);
    const q=t*(radii.length-1),j=Math.min(radii.length-2,Math.floor(q));
    const r=THREE.MathUtils.lerp(radii[j],radii[j+1],q-j);
    for(let k=0;k<=sides;k++){
      const a=k/sides*Math.PI*2;
      const n=frames.normals[i].clone().multiplyScalar(Math.cos(a))
        .addScaledVector(frames.binormals[i],Math.sin(a));
      const v=p.clone().addScaledVector(n,r);
      vertices.push(v.x,v.y,v.z);normals.push(n.x,n.y,n.z);
      if(i<segments&&k<sides){
        const a0=i*(sides+1)+k,b=a0+sides+1;
        indices.push(a0,b,a0+1,b,b+1,a0+1);
      }
    }
  }
  const geometry=new THREE.BufferGeometry();
  geometry.setAttribute('position',new THREE.Float32BufferAttribute(vertices,3));
  geometry.setAttribute('normal',new THREE.Float32BufferAttribute(normals,3));
  geometry.setIndex(indices);
  const mesh=new THREE.Mesh(geometry,material);
  mesh.name=name;mesh.castShadow=true;mesh.receiveShadow=true;parent.add(mesh);
  return mesh;
}

const profile=[
 [1.05,.30],[1.15,.49],[1.32,.63],[1.60,.77],
 [1.91,.89],[2.20,.94],[2.48,.87],[2.75,.76],

];
function radiusAt(y){
  for(let i=0;i<profile.length-1;i++){
    if(y<=profile[i+1][0]){
      const t=(y-profile[i][0])/(profile[i+1][0]-profile[i][0]);
      const s=t*t*(3-2*t);
      return THREE.MathUtils.lerp(profile[i][1],profile[i+1][1],s);
    }
  }
  return profile.at(-1)[1];
}
const pos=[],colors=[],idx=[];
const rings=100,sides=80;
const orange=new THREE.Color('#e8a523');
const yellow=new THREE.Color('#f3c32b');
const cream=new THREE.Color('#ffdf8c');
for(let j=0;j<=rings;j++){
  const y=1.05+(4-1.05)*j/rings;
  const radius=radiusAt(y);
  const depth=radius*(.76+.035*Math.sin(y*2));
  for(let k=0;k<=sides;k++){
    const a=k/sides*Math.PI*2;
    const x=Math.sin(a)*radius;
    const z=Math.cos(a)*depth;
    pos.push(x-.025*Math.sin((y-1)*1.3),y,z);
    const c=orange.clone().lerp(yellow,THREE.MathUtils.smoothstep(y,1.65,3.65));
    const front=Math.max(0,Math.cos(a));
    const belly=Math.exp(-Math.pow((y-2.60)/.68,2))
      *Math.exp(-Math.pow(x/.69,4))*Math.pow(front,4)*.77;
    c.lerp(cream,belly);
    const fleck=Math.sin(j*51.17+k*83.79)*.008;
    c.offsetHSL(0,0,fleck);
    colors.push(c.r,c.g,c.b);
    if(j<rings&&k<sides){
      const p=j*(sides+1)+k,q=p+sides+1;
      idx.push(p,p+1,q,p+1,q+1,q);
    }
  }
}

  ellipsoid('palm',hand,handMat,[0,-.055,0],[.105,.165,.063]);
  const starts=[-.071,-.025,.025,.071];
  const lengths=[.20,.265,.25,.19];
  for(let i=0;i<4;i++){
    const sx=starts[i],splay=(i-1.5)*.026;
    taperedTube('finger-'+(i+1),hand,
      [[sx,-.12,.013],
       [sx+splay,-.21,.035],
       [sx+splay*1.35,-.12-lengths[i],.065]],
      [.032,.027,.016],fingertipMat,16,10);
    ellipsoid('fingertip-'+(i+1),hand,fingertipMat,
      [sx+splay*1.35,-.12-lengths[i],.065],[.017,.028,.019]);
  }
  const ts=side>0?-1:1;
  taperedTube('thumb',hand,
    [[ts*.075,-.015,.025],[ts*.139,-.095,.075],[ts*.15,-.19,.10]],
    [.043,.034,.021],handMat,18,12);
  return hand;
}
