function ellipsoid(name,parent,mat,p,s){

function tube(name,parent,points,radii,mat,segments=22,sides=12){
  const curve=new THREE.CatmullRomCurve3(points.map(p=>new THREE.Vector3(...p)));
  const frames=curve.computeFrenetFrames(segments,false);
  const vertices=[],normals=[],indices=[];
  for(let i=0;i<=segments;i++){
    const t=i/segments,p=curve.getPointAt(t);
    const f=t*(radii.length-1),j=Math.min(radii.length-2,Math.floor(f));
    const r=THREE.MathUtils.lerp(radii[j],radii[j+1],f-j);
    for(let k=0;k<=sides;k++){
      const a=k/sides*Math.PI*2;
      const n=frames.normals[i].clone().multiplyScalar(Math.cos(a)).addScaledVector(frames.binormals[i],Math.sin(a));
      const v=p.clone().addScaledVector(n,r);
      vertices.push(v.x,v.y,v.z);normals.push(n.x,n.y,n.z);
      if(i<segments&&k<sides){
        const u=i*(sides+1)+k,vv=u+sides+1;
        indices.push(u,vv,u+1,vv,vv+1,u+1);
      }
    }
  }
  const geo=new THREE.BufferGeometry();
  geo.setAttribute('position',new THREE.Float32BufferAttribute(vertices,3));
  geo.setAttribute('normal',new THREE.Float32BufferAttribute(normals,3));
  geo.setIndex(indices);
  const mesh=new THREE.Mesh(geo,mat);
  mesh.name=name;mesh.castShadow=true;mesh.receiveShadow=true;parent.add(mesh);
  return mesh;
}

function radiusAt(y){
  let i=0;
  while(i<profile.length-2&&y>profile[i+1][0])i++;
  const a=profile[i],b=profile[i+1],t=THREE.MathUtils.clamp((y-a[0])/(b[0]-a[0]),0,1);
  const before=profile[Math.max(0,i-1)],after=profile[Math.min(profile.length-1,i+2)];
  const m0=(b[1]-before[1])/(b[0]-before[0]);
  const m1=(after[1]-a[1])/(after[0]-a[0]);
  const dt=b[0]-a[0];
  return (2*t*t*t-3*t*t+1)*a[1]+(t*t*t-2*t*t+t)*dt*m0+
    (-2*t*t*t+3*t*t)*b[1]+(t*t*t-t*t)*dt*m1;
}

function segment(name,parent,r0,r1,mat){
  const mesh=new THREE.Mesh(new THREE.CylinderGeometry(r1,r0,1,22,1),mat);
  mesh.name=name;mesh.castShadow=true;mesh.receiveShadow=true;parent.add(mesh);

function setSegment(mesh,a,b){
  va.set(...a);vb.set(...b);
  delta.subVectors(vb,va);
  mesh.position.copy(va).add(vb).multiplyScalar(.5);
  mesh.scale.set(1,delta.length(),1);
  mesh.quaternion.setFromUnitVectors(axisY,delta.normalize());
}

  return {g,side,upper,lower,shoulder,elbow,wrist,hand,fingers};
}

function poseArm(arm,e,w,rotation,scale,spread){
  const shoulder=[arm.side*.45,3.49,.015];
  setSegment(arm.upper,shoulder,e);
  setSegment(arm.lower,e,w);
  arm.elbow.position.set(...e);
  arm.wrist.position.set(...w);
  arm.hand.position.set(...w);
  arm.hand.rotation.set(rotation[0],rotation[1],rotation[2],'XYZ');
  arm.wrist.quaternion.copy(arm.hand.quaternion);
  arm.hand.scale.setScalar(scale);
  arm.fingers.scale.x=spread;
}

function interpolate(a,b,u){
  if(Array.isArray(a))return a.map((v,i)=>v+(b[i]-v)*u);
  return a+(b-a)*u;
}
function render(t){
  const time=((Number(t)||0)%5540+5540)%5540;
  let index=0;
  while(index<poses.length-2&&time>=poses[index+1].t)index++;
  const a=poses[index],b=poses[index+1];
  let u=(time-a.t)/(b.t-a.t);
  u=u*u*(3-2*u);
  const p={};
  for(const name of ['r','h','le','lw','re','rw','lh','rh','hs','sp','eye','smile','step']){
    p[name]=interpolate(a[name],b[name],u);
  }
  character.position.set(p.r[0],p.r[1],0);
  character.scale.setScalar(p.r[2]);
  character.rotation.set(0,p.r[3],p.r[4],'YXZ');

  head.position.set(p.h[5],4.07+p.h[3],p.h[4]);
  head.rotation.set(p.h[0],p.h[1],p.h[2],'YXZ');
  neck.position.z=(p.h[4]-.02)*.32;
  neck.rotation.x=p.h[0]*.25;

  for(const part of eyeParts){
    part.eye.scale.set(1,p.eye,1);
    part.lid.position.y=.075+.012*p.eye;
  }
  brows.position.y=(1-p.eye)*-.065;
  mouthLine.scale.y=p.smile;
  mouth.scale.y=.86+.14*(1-p.eye);

  poseArm(rightArm,p.re,p.rw,p.rh,p.hs[1],p.sp[1]);
  poseLeg(leftLeg,p.step[0],Math.max(0,p.step[0]));
  poseLeg(rightLeg,p.step[1],Math.max(0,-p.step[1]));

  renderer.render(scene,ref);
}
