const poses=[

  {rise:.17,turn:.12,open:0,wx:-7,wy:-2,pen:-5,hx:-8,hy:-2,hr:1,rx:-1,ry:.5},

const left=byId("character-left");
const right=byId("character-right");
const resting=byId("left-resting-hand");
const writing=byId("left-writing-hand");
const pencil=byId("writing-pencil");

/* Keep the foreground hands within the left character group while
   preserving the original painting order over the books and pencil. */
const foreground=document.createElementNS("http://www.w3.org/2000/svg","g");
foreground.setAttribute("id","left-foreground-parts");
left.appendChild(foreground);
foreground.appendChild(resting);
foreground.appendChild(pencil);
foreground.appendChild(writing);

/* The books go behind the character's foreground parts. */
const svg=document.querySelector("svg");
svg.insertBefore(byId("desk-books"),left);

/* Restore the original overlap: the right character passes in front of
   the left torso, and its hand touches the belly above the workbook. */
svg.insertBefore(right,byId("loose-stationery"));
svg.appendChild(byId("loose-stationery"));

function render(input){
  let t=Number(input);
  if(!Number.isFinite(t))t=0;
  t=((t%1500)+1500)%1500;
  let i=0;
  while(i<times.length-2&&t>=times[i+1])i++;
  const u=(t-times[i])/(times[i+1]-times[i]);
  const a=poses[i],b=poses[i+1],s={};
  for(const key of Object.keys(a))s[key]=a[key]+(b[key]-a[key])*u;
  const p=s.rise,q=s.turn;
  for(const m of morphs){
    const amount=m.element.id==="left-mouth-line"?s.open:p;
    let n=0;
    const d=m.start.replace(numberPattern,()=>{
      const j=n++;
      return (m.a[j]+(m.b[j]-m.a[j])*amount).toFixed(4);
    });
    m.element.setAttribute("d",d);
  }

  set("left-head","transform",
    "translate("+(-5*p)+" "+(-16*p)+") translate(60 85) scale("+(1-.08*q)+" 1) translate(-60 -85)");
  move("left-eye-outer",14*q,-8*q);
  move("left-eye-inner",-.3*q,-10*q);
  move("left-outer-pupil",.8*q,-.25*q);
  move("left-inner-pupil",.2*q,-.5*q);
  move("left-mouth",12*q,-14*q);
  set("left-mouth-opening","opacity",s.open);
  set("left-mouth-line","stroke",
    "rgb("+Math.round(169-112*s.open)+","+Math.round(121-76*s.open)+","+Math.round(25-4*s.open)+")");
  set("left-mouth-line","stroke-width",1.7+.5*s.open);
  set("left-lower-lip","opacity",.7*(1-.8*s.open));

  move("left-resting-hand",-5*p,-.5*p);
  set("left-writing-hand","transform",
    "translate("+s.wx+" "+s.wy+") rotate("+(-4*p)+" 32 201)");
  set("writing-pencil","transform",
    "translate("+s.wx+" "+s.wy+") rotate("+s.pen+" 32 198)");

  move("right-head",s.rx,s.ry);
  move("right-near-eye",-1.5*q,-1.2*q);
  move("right-far-eye",-.6*q,-1*q);
  move("right-mouth",-.4*q,-3*q);
  set("right-mouth-line","d",
    "M121.1 "+(88.4-.8*q)+"C126.9 "+(91.2-3.1*q)+" 133.6 "+(91.4-3.1*q)+" 139 "+(88.7-.7*q));
  set("right-mouth-line","stroke-width",2+.3*q);
  set("right-mouth-lip","opacity",1-.4*q);
  set("right-far-arm","transform",
    "translate("+(-3*p)+" "+(-3.5*p)+")");
  move("right-torso",-.6*p,0);

  const x=s.hx,y=s.hy;
  set("right-embracing-hand","transform",
    "translate("+x+" "+y+") rotate("+s.hr+" 109 174)");
  set("right-arm-shape","d",
    "M187 110C197 109 204 119 204 129"+
    "C205 145 "+(196+x*.12)+" "+(158+y*.15)+" "+(183+x*.18)+" "+(168+y*.25)+
    "C"+(167+x*.4)+" "+(180+y*.5)+" "+(148+x*.7)+" "+(187+y*.8)+" "+(130+x*.85)+" "+(188+y)+
    "C"+(122+x)+" "+(188+y)+" "+(114+x)+" "+(184+y)+" "+(106+x)+" "+(181+y)+
    "L"+(106+x)+" "+(165+y)+
    "C"+(123+x)+" "+(165+y)+" "+(140+x*.65)+" "+(162+y*.65)+" "+(152+x*.4)+" "+(155+y*.4)+
    "C"+(164+x*.2)+" "+(149+y*.2)+" 173 142 177 132C181 121 179 114 187 110Z");
  set("right-arm-shadow","d",
    "M187 121C193 137 "+(177+x*.25)+" "+(156+y*.3)+" "+(158+x*.5)+" "+(164+y*.5)+
    "L"+(112+x)+" "+(169+y)+"L"+(107+x)+" "+(185+y)+
    "C"+(127+x*.8)+" "+(194+y)+" "+(159+x*.3)+" "+(190+y*.4)+" 184 175C204 161 211 144 202 126Z");
  set("right-arm-light","d",
    "M191 116C200 124 197 141 186 153"+
    "C"+(172+x*.2)+" "+(167+y*.2)+" "+(156+x*.5)+" "+(173+y*.6)+" "+(139+x*.8)+" "+(176+y*.8)+
    "C"+(165+x*.3)+" "+(166+y*.3)+" 184 151 188 137C192 128 190 122 191 116Z");
}
window.nwb={render};
render(0);
})();
