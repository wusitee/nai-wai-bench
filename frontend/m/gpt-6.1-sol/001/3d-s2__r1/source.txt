function tube(parent,name,points,radii,mat=yellow){
 const pts=points.map(p=>new THREE.Vector3(...p));
 const curve=new THREE.CatmullRomCurve3(pts);
 const rings=36,sides=14;
 const positions=[],indices=[],weights=[];
 for(let i=0;i<=rings;i++){
  const t=i/rings;
  const c=curve.getPoint(t);
  const tangent=curve.getTangent(t).normalize();
  let side=new THREE.Vector3(0,0,1).cross(tangent);
  if(side.lengthSq()<.01)side.set(1,0,0).cross(tangent);
  side.normalize();
  const other=new THREE.Vector3().crossVectors(tangent,side).normalize();
  const f=t*(radii.length-1);
  const j=Math.min(Math.floor(f),radii.length-2);
  const r=THREE.MathUtils.lerp(radii[j],radii[j+1],f-j);
  for(let k=0;k<=sides;k++){
   const a=k/sides*Math.PI*2;
   const v=c.clone()
    .addScaledVector(side,Math.cos(a)*r)
    .addScaledVector(other,Math.sin(a)*r);
   positions.push(v.x,v.y,v.z);
   weights.push(t*t*(3-2*t));
   if(i<rings&&k<sides){
    const q=i*(sides+1)+k;
    indices.push(q,q+sides+1,q+1,q+1,q+sides+1,q+sides+2);
   }
  }
 }
 const geo=new THREE.BufferGeometry();
 geo.setAttribute('position',new THREE.Float32BufferAttribute(positions,3));
 geo.setIndex(indices);
 geo.computeVertexNormals();
 const g=group(parent,name);
 const mesh=new THREE.Mesh(geo,mat);
 mesh.castShadow=true;
 mesh.receiveShadow=true;
 g.add(mesh);
 const capStart=ball(g,'rounded-start',points[0],[radii[0],radii[0],radii[0]],mat);
 const r=radii[radii.length-1];
 const capEnd=ball(g,'rounded-end',points[points.length-1],[r,r,r],mat);
 g.userData.tube={
  geometry:geo,
  rest:new Float32Array(geo.attributes.position.array),
  weights:new Float32Array(weights),
  capStart,capEnd,
  start:pts[0].clone(),
  end:pts[pts.length-1].clone()
 };
 return g;
}
function deformTube(g,start,end){
 const data=g.userData.tube;
 const a=data.geometry.attributes.position.array;
 for(let i=0;i<data.weights.length;i++){
  const w=data.weights[i],v=i*3;
  a[v]=data.rest[v]+start[0]*(1-w)+end[0]*w;
  a[v+1]=data.rest[v+1]+start[1]*(1-w)+end[1]*w;
  a[v+2]=data.rest[v+2]+start[2]*(1-w)+end[2]*w;
 }
 data.geometry.attributes.position.needsUpdate=true;
 data.geometry.computeVertexNormals();
 data.geometry.computeBoundingSphere();
 data.capStart.position.set(
  data.start.x+start[0],data.start.y+start[1],data.start.z+start[2]
 );
 data.capEnd.position.set(
  data.end.x+end[0],data.end.y+end[1],data.end.z+end[2]
 );
}

function pivotPose(object,pivot,dx,dy,dz,angle){
 object.rotation.set(0,0,angle);
 rotatedPivot.copy(pivot).applyQuaternion(object.quaternion);
 object.position.set(
  pivot.x-rotatedPivot.x+dx,
  pivot.y-rotatedPivot.y+dy,
  pivot.z-rotatedPivot.z+dz
 );
}

function render(t){
 const input=Number.isFinite(t)?t:0;
 let time=Math.round((((input%1500)+1500)%1500)*100000)/100000;

 while(k<poses.length-2&&time>=poses[k+1][0])k++;
 const a=poses[k],b=poses[k+1];
 let f=(time-a[0])/(b[0]-a[0]);
 f=f*f*(3-2*f);
 for(let j=1;j<pose.length;j++)pose[j]=a[j]+(b[j]-a[j])*f;
