let seed=71983;
function rnd(){
  seed=(Math.imul(seed,1664525)+1013904223)>>>0;
  return seed/4294967296;
}
function group(name,parent){
  const g=new THREE.Group();
  g.name=name;
  if(parent)parent.add(g);
  return g;
}
function fur(parent,radii,count,color,mask=null,length=.028){
  const p=[],c=[];
  const base=new THREE.Color(color);
  for(let i=0;i<count;i++){
    const ny=2*rnd()-1,phi=rnd()*Math.PI*2;
    const rr=Math.sqrt(1-ny*ny);
    const n=new THREE.Vector3(rr*Math.cos(phi),ny,rr*Math.sin(phi));
    const v=new THREE.Vector3(n.x*radii[0],n.y*radii[1],n.z*radii[2]);
    if(mask && !mask(v,n))continue;
    const normal=new THREE.Vector3(n.x/radii[0],n.y/radii[1],n.z/radii[2]).normalize();
    let tangent=new THREE.Vector3(0,1,0).cross(normal);
    if(tangent.lengthSq()<.001)tangent.set(1,0,0);
    tangent.normalize();
    const w=.002+rnd()*.0028;
    const len=length*(.55+rnd()*.8);
    const root=v.clone().addScaledVector(normal,-.002);
    const tip=v.clone().addScaledVector(normal,len);
    tip.y-=len*.22;
    p.push(
      root.x+tangent.x*w,root.y+tangent.y*w,root.z+tangent.z*w,
      root.x-tangent.x*w,root.y-tangent.y*w,root.z-tangent.z*w,
      tip.x,tip.y,tip.z
    );
    const shade=.81+rnd()*.27;
    for(let j=0;j<3;j++)c.push(base.r*shade,base.g*shade,base.b*shade);
  }
  const geo=new THREE.BufferGeometry();
  geo.setAttribute('position',new THREE.Float32BufferAttribute(p,3));
  geo.setAttribute('color',new THREE.Float32BufferAttribute(c,3));
  geo.computeVertexNormals();
  const mesh=new THREE.Mesh(geo,new THREE.MeshStandardMaterial({
    vertexColors:true,roughness:1,side:THREE.DoubleSide
  }));
  mesh.name='short-golden-fur';
  parent.add(mesh);
}
function ellipsoid(name,parent,pos,scale,mat,furCount=0,furColor='#dcb022',mask=null){
  const g=group(name,parent);
  g.position.set(...pos);
  const mesh=new THREE.Mesh(new THREE.SphereGeometry(1,48,36),mat);
  mesh.scale.set(...scale);
  mesh.castShadow=true;
  mesh.receiveShadow=true;
  g.add(mesh);
  if(furCount)fur(g,scale,furCount,furColor,mask);
  return g;
}
function curveTube(name,parent,points,radius,material,segments=24){
  const curve=new THREE.CatmullRomCurve3(points.map(p=>new THREE.Vector3(...p)));
  const mesh=new THREE.Mesh(new THREE.TubeGeometry(curve,segments,radius,9,false),material);
  mesh.name=name;
  mesh.castShadow=true;
  mesh.receiveShadow=true;
  parent.add(mesh);
  return mesh;
}
function limb(name,parent,a,b,width,depth,material,count){
  const av=new THREE.Vector3(...a),bv=new THREE.Vector3(...b);
  const center=av.clone().add(bv).multiplyScalar(.5);
  const g=ellipsoid(name,parent,center.toArray(),
    [width,av.distanceTo(bv)*.5+width*.32,depth],material,count);
  g.quaternion.setFromUnitVectors(new THREE.Vector3(0,1,0),av.clone().sub(bv).normalize());
  return g;
}

const RX=.525,RY=.69,RZ=.455;
const mouthRX=.327,mouthRY=.402,mouthCY=-.084;
function mouthBoundary(a){
  const s=Math.sin(a),c=Math.cos(a);
  return new THREE.Vector2(mouthRX*c*(1+.075*s),mouthCY+mouthRY*s);
}
function faceZ(x,y){
  return RZ*Math.sqrt(Math.max(.005,1-x*x/(RX*RX)-y*y/(RY*RY)));
}
{
  const positions=[],indices=[],n=96,nr=25;
  for(let r=0;r<=nr;r++){
    const t=r/nr;
    for(let j=0;j<=n;j++){
      const a=j/n*Math.PI*2;
      const inner=mouthBoundary(a);
      const x=inner.x*(1-t)+RX*Math.cos(a)*t;
      const y=inner.y*(1-t)+RY*Math.sin(a)*t;
      positions.push(x,y,faceZ(x,y));
    }
  }
  for(let r=0;r<nr;r++)for(let j=0;j<n;j++){
    const a=r*(n+1)+j,c=a+n+1;
    indices.push(a,c,a+1,a+1,c,c+1);
  }
  const geo=new THREE.BufferGeometry();
  geo.setAttribute('position',new THREE.Float32BufferAttribute(positions,3));
  geo.setIndex(indices);geo.computeVertexNormals();
  const mesh=new THREE.Mesh(geo,headGold);
  mesh.castShadow=true;mesh.receiveShadow=true;
  mesh.name='face-around-open-mouth';
  head.add(mesh);

  const bp=[],bi=[];
  for(let r=0;r<=nr;r++){
    const t=r/nr;
    for(let j=0;j<=n;j++){
      const a=j/n*Math.PI*2;
      bp.push(RX*t*Math.cos(a),RY*t*Math.sin(a),-RZ*Math.sqrt(Math.max(0,1-t*t)));
    }
  }
  for(let r=0;r<nr;r++)for(let j=0;j<n;j++){
    const a=r*(n+1)+j,c=a+n+1;
    bi.push(a,a+1,c,a+1,c+1,c);
  }
  const bg=new THREE.BufferGeometry();
  bg.setAttribute('position',new THREE.Float32BufferAttribute(bp,3));
  bg.setIndex(bi);bg.computeVertexNormals();

  head.add(back);
}
