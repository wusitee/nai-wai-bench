function rnd(){
  seed=(Math.imul(seed,1664525)+1013904223)>>>0;
  return seed/4294967296;
}

function fur(parent,radii,count,color,mask=null,length=.028){
  const p=[],c=[],base=new THREE.Color(color);
  for(let i=0;i<count;i++){
    const ny=2*rnd()-1,phi=rnd()*Math.PI*2,rr=Math.sqrt(1-ny*ny);
    const n=new THREE.Vector3(rr*Math.cos(phi),ny,rr*Math.sin(phi));
    const v=new THREE.Vector3(n.x*radii[0],n.y*radii[1],n.z*radii[2]);
    if(mask&&!mask(v,n))continue;
    const normal=new THREE.Vector3(n.x/radii[0],n.y/radii[1],n.z/radii[2]).normalize();
    const tangent=new THREE.Vector3(0,1,0).cross(normal);
    if(tangent.lengthSq()<.001)tangent.set(1,0,0);
    tangent.normalize();
    const w=.002+rnd()*.0028,len=length*(.55+rnd()*.8);
    const root=v.clone().addScaledVector(normal,-.002);
    const tip=v.clone().addScaledVector(normal,len);
    tip.y-=len*.22;
    p.push(root.x+tangent.x*w,root.y+tangent.y*w,root.z+tangent.z*w,
      root.x-tangent.x*w,root.y-tangent.y*w,root.z-tangent.z*w,
      tip.x,tip.y,tip.z);
    const shade=.81+rnd()*.27;
    for(let j=0;j<3;j++)c.push(base.r*shade,base.g*shade,base.b*shade);
  }
  const geo=new THREE.BufferGeometry();
  geo.setAttribute('position',new THREE.Float32BufferAttribute(p,3));
  geo.setAttribute('color',new THREE.Float32BufferAttribute(c,3));
  geo.computeVertexNormals();
  const mesh=new THREE.Mesh(geo,new THREE.MeshStandardMaterial({
    vertexColors:true,roughness:1,side:THREE.DoubleSide
  }));
  mesh.name='short-golden-fur';
  parent.add(mesh);
}

function ellipsoid(name,parent,pos,scale,mat,furCount=0,furColor='#dcb022',mask=null){

}
function curveTube(name,parent,points,radius,material,segments=24){
  const curve=new THREE.CatmullRomCurve3(points.map(p=>new THREE.Vector3(...p)));
  const mesh=new THREE.Mesh(new THREE.TubeGeometry(curve,segments,radius,9,false),material);
  mesh.name=name;
  mesh.castShadow=true;
  mesh.receiveShadow=true;
  parent.add(mesh);
  return mesh;
}

function limb(name,parent,a,b,width,depth,material,count){
  const av=new THREE.Vector3(...a),bv=new THREE.Vector3(...b);
  const radiusY=av.distanceTo(bv)*.5+width*.32;
  const g=ellipsoid(name,parent,av.clone().add(bv).multiplyScalar(.5).toArray(),
    [width,radiusY,depth],material,count);
  g.quaternion.setFromUnitVectors(up,av.clone().sub(bv).normalize());
  g.userData.width=width;
  g.userData.radiusY=radiusY;
  return g;
}
function poseLimb(g,a,b){
  va.set(...a);
  vb.set(...b);
  g.position.copy(va).add(vb).multiplyScalar(.5);
  direction.copy(va).sub(vb);
  const length=direction.length();
  g.quaternion.setFromUnitVectors(up,direction.normalize());
  g.scale.set(1,(length*.5+g.userData.width*.32)/g.userData.radiusY,1);
}

function mouthBoundary(a){
  const s=Math.sin(a),c=Math.cos(a);
  return new THREE.Vector2(mouthRX*c*(1+.075*s),mouthCY+mouthRY*s);
}
function faceZ(x,y){
  return RZ*Math.sqrt(Math.max(.005,1-x*x/(RX*RX)-y*y/(RY*RY)));
}
{
  const positions=[],indices=[],n=96,nr=25;

      const x=inner.x*(1-t)+RX*Math.cos(a)*t;
      const y=inner.y*(1-t)+RY*Math.sin(a)*t;
      positions.push(x,y,faceZ(x,y));
    }
  }
  for(let r=0;r<nr;r++)for(let j=0;j<n;j++){

function render(t){

  let index=0;
  while(index<poses.length-2&&time>=poses[index+1][0])index++;
  const a=poses[index],b=poses[index+1];
  let u=(time-a[0])/(b[0]-a[0]);
  u=u*u*(3-2*u);
  for(let k=1;k<9;k++)pose[k]=a[k]+(b[k]-a[k])*u;

  const c=pose[1],sway=pose[8];

  body.position.set(sway,2.12-.325*c,-.035+.125*c);
  body.scale.set(1+.105*c,1-.205*c,1+.075*c);
  body.rotation.set(.105*c,0,.025*c);

  neck.position.set(-.04+sway+(pose[2]+.035)*.23,3.03-.735*c,.015*c-.015);
  neck.scale.set(1+.055*c,1-.13*c,1+.04*c);

  head.position.set(pose[2],pose[3],pose[4]);
  head.rotation.set(pose[5],-.035*c,pose[6]);

    p.haunch.rotation.z=-side*.14*c;
    const hip=[x+side*.08*c,1.04-.105*c,.06+.065*c];
    const knee=[x+side*(.055+.185*c),.59-.075*c,.27+.23*c];
    poseLimb(p.thigh,hip,[knee[0],knee[1]-.03,knee[2]-.01]);
    place(p.knee,knee);
    poseLimb(p.shin,[knee[0],knee[1]-.02,knee[2]-.04],
      [x+side*(.035+.10*c),.04,.035+.035*c]);
    place(p.foot,[x+side*(.045+.10*c),.035,.20+.035*c]);
    p.foot.rotation.y=-side*.10*c;
  }

  const ls=onBody(-.66,2.79,-.015);
  const rs=onBody(.65,2.79,-.035);
  place(leftShoulder,ls);
  place(rightShoulder,rs);
  leftShoulder.rotation.z=-.21*c;
  rightShoulder.rotation.z=.23*c;

  const le=[-.91-.345*c+sway,1.99-.205*c,.12+.22*c];
