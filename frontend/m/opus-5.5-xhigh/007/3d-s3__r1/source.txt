function profileLathe(pts, segs, n) {
  const curve = new THREE.SplineCurve(pts.map(p => new THREE.Vector2(p[0], p[1])));
  const sp = curve.getPoints(n).map(v => new THREE.Vector2(Math.max(0, v.x), v.y));
  sp[0].x = 0; sp[sp.length - 1].x = 0;
  return new THREE.LatheGeometry(sp, segs, Math.PI, Math.PI * 2);
}

function limbGeo(len, r1, r2, segs = 24) {
  const pts = [], n = 8;
  for (let i = 0; i <= n; i++) {
    const a = -Math.PI / 2 + i / n * Math.PI / 2;
    pts.push(new THREE.Vector2(Math.max(0, Math.cos(a) * r2), -len + Math.sin(a) * r2));
  }
  for (let i = 0; i <= n; i++) {
    const a = i / n * Math.PI / 2;
    pts.push(new THREE.Vector2(Math.max(0, Math.cos(a) * r1), Math.sin(a) * r1));
  }
  pts[0].x = 0; pts[pts.length - 1].x = 0;
  return new THREE.LatheGeometry(pts, segs);
}

function mix(a, b, e) {
  const o = {};
  for (const k in a) {
    const x = a[k], y = b[k];
    if (Array.isArray(x)) {
      const r = new Array(x.length);
      for (let i = 0; i < x.length; i++) r[i] = x[i] + (y[i] - x[i]) * e;
      o[k] = r;
    } else o[k] = x + (y - x) * e;
  }
  return o;
}

// ---------- pose solving ----------
const _X = new THREE.Vector3(), _Y = new THREE.Vector3(), _Z = new THREE.Vector3(), _M = new THREE.Matrix4();
function orient(obj, yDir, zRef) {
  _Y.copy(yDir);
  if (_Y.lengthSq() < 1e-10) _Y.set(0, 1, 0);
  _Y.normalize();
  _Z.copy(zRef).addScaledVector(_Y, -zRef.dot(_Y));
  if (_Z.lengthSq() < 1e-6) {
    _Z.set(0, 0, 1).addScaledVector(_Y, -_Y.z);
    if (_Z.lengthSq() < 1e-6) _Z.set(1, 0, 0).addScaledVector(_Y, -_Y.x);
  }
  _Z.normalize();
  _X.crossVectors(_Y, _Z);
  _M.makeBasis(_X, _Y, _Z);
  obj.quaternion.setFromRotationMatrix(_M);
}

const aD = new THREE.Vector3(), aPerp = new THREE.Vector3(), aE = new THREE.Vector3(),
  aT = new THREE.Vector3(), aTmp = new THREE.Vector3();
function solveArm(a, S, T, pole, palm) {
  aD.subVectors(T, S);
  let d = aD.length();
  if (d < 1e-5) { aD.set(0, -1, 0); d = 1e-5; } else aD.divideScalar(d);
  d = Math.min(Math.max(d, Math.abs(L1 - L2) + 0.02), (L1 + L2) * 0.9995);
  const cosA = Math.max(-1, Math.min(1, (L1 * L1 + d * d - L2 * L2) / (2 * L1 * d)));
  const sinA = Math.sqrt(Math.max(0, 1 - cosA * cosA));
  aPerp.copy(pole).addScaledVector(aD, -pole.dot(aD));
  if (aPerp.lengthSq() < 1e-8) {
    aPerp.set(0, 0, -1).addScaledVector(aD, aD.z);
    if (aPerp.lengthSq() < 1e-8) aPerp.set(1, 0, 0).addScaledVector(aD, -aD.x);
  }
  aPerp.normalize();
  aE.copy(S).addScaledVector(aD, L1 * cosA).addScaledVector(aPerp, L1 * sinA);
  aT.copy(S).addScaledVector(aD, d);
  a.upper.position.copy(S);
  orient(a.upper, aTmp.subVectors(S, aE), palm);
  a.fore.position.copy(aE);
  orient(a.fore, aTmp.subVectors(aE, aT), palm);
}

const sp = new THREE.Vector3(), sn = new THREE.Vector3();
function headSurf(x, dy, off, A, B, C) {
  const k = 1 - (x / A) ** 2 - (dy / B) ** 2;
  const z = C * Math.sqrt(Math.max(k, 0));
  sn.set(x / (A * A), dy / (B * B), z / (C * C)).normalize();
  sp.set(x, dy, z).addScaledVector(sn, off);
}

function setAmt(o, a, bx = 1, by = 1, bz = 1) {
  a = clamp01(a);
  o.visible = a > 0.002;
  o.scale.set(bx * a, by * a, bz * a);
}

const vS = new THREE.Vector3(), vT = new THREE.Vector3(), vP = new THREE.Vector3(), vN = new THREE.Vector3();
function applyPose(k) {
  camera.position.set(k.cam[0], k.cam[1], k.cam[2]);
  camera.lookAt(k.cam[0], k.cam[1], 0);

  penguin.position.set(0, 0, 0);
  penguin.rotation.set(0, k.rotY, 0);

  const sx = k.tor[0], sy = k.tor[1], sz = k.tor[2];
  torso.scale.set(sx, sy, 0.88 * sz);
  bodyG.rotation.set(k.lean[0], 0, k.lean[1]);

  headG.position.set(k.head[0], 1.24 + 3.76 * sy + k.head[1] - PIV, k.head[2]);
  headG.rotation.set(k.hrot[0], k.hrot[1], k.hrot[2]);
  const A = k.hs[0], B = k.hs[1], C = k.hs[2], CY = k.hs[3];
  headMesh.scale.set(A, B, C);
  headMesh.position.set(0, CY, 0);

  const open = clamp01(k.eye[2]);
  for (const ep of eyeParts) {
    headSurf(ep.s * k.eye[0], k.eye[1], 0.006, A, B, C);
    ep.eG.position.set(sp.x, sp.y + CY, sp.z);
    ep.eG.quaternion.setFromUnitVectors(Zaxis, sn);
    setAmt(ep.lid, 1 - open);
    setAmt(ep.open, open);
  }

  headSurf(0, k.mouth[0], 0, A, B, C);
  mouthG.position.set(sp.x, sp.y + CY, sp.z);
  mouthG.quaternion.setFromUnitVectors(Zaxis, sn);
  setAmt(beak, k.mouth[1], 1, 0.8, 1);
  setAmt(smile, k.mouth[2]);
  setAmt(lips, k.mouth[3], 0.1, 0.055, 0.05);

  setAmt(propG, k.prop);

  const shY = 1.24 + 3.01 * sy - PIV;
  for (const a of arms) {
    const t = a.s < 0 ? k.lt : k.rt;
    const p = a.s < 0 ? k.lp : k.rp;
    const n = a.s < 0 ? k.ln : k.rn;
    vS.set(a.s * 0.88 * sx, shY, k.shZ);
    vT.set(t[0], t[1] - PIV, t[2]);
    vP.set(p[0], p[1], p[2]);
    vN.set(n[0], n[1], n[2]);
    solveArm(a, vS, vT, vP, vN);
  }
}

function render(t) {
  let tt = Number(t);
  if (!isFinite(tt)) tt = 0;
  tt = tt % PERIOD;
  if (tt < 0) tt += PERIOD;
  let i = 0;
  while (i < 7 && tt >= KT[i + 1]) i++;
  const u = clamp01((tt - KT[i]) / (KT[i + 1] - KT[i]));
  const e = u * u * (3 - 2 * u);
  const k = mix(KEYS[i], KEYS[(i + 1) % 8], e);
  applyPose(k);
  renderer.render(scene, camera);
}
render(0);

window.nwb = {
  scene,
  renderer,
  cameras: { ref: camera },
  characters: [penguin],
  render,
  ready: true
};
