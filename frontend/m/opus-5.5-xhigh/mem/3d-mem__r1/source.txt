<script type="module">
import * as THREE from 'three';
import { mergeVertices } from 'three/addons/utils/BufferGeometryUtils.js';

const renderer = new THREE.WebGLRenderer({ preserveDrawingBuffer: true, antialias: true });
renderer.setPixelRatio(1);
renderer.setSize(240, 240);
document.body.appendChild(renderer.domElement);

const scene = new THREE.Scene();
scene.background = new THREE.Color(0xe6f2ff);

const camera = new THREE.PerspectiveCamera(35, 1, 0.1, 100);
camera.position.set(0, 1.6, 5.2);
camera.lookAt(0, 1.28, 0);

// Lights
scene.add(new THREE.HemisphereLight(0xffffff, 0xcbb894, 1.05));
const key = new THREE.DirectionalLight(0xffffff, 1.8); key.position.set(2.5, 4, 5); scene.add(key);
const back = new THREE.DirectionalLight(0xffffff, 1.2); back.position.set(-3, 3, -4); scene.add(back);
const side = new THREE.DirectionalLight(0xfff4e0, 0.6); side.position.set(-4, 1, 1.5); scene.add(side);
const under = new THREE.DirectionalLight(0xffffff, 0.3); under.position.set(0, -3, 2); scene.add(under);

// Materials
const bodyMat = new THREE.MeshStandardMaterial({ color: 0xffc93a, roughness: 0.5, metalness: 0 });
const bellyMat = new THREE.MeshStandardMaterial({ color: 0xfff2c2, roughness: 0.6, metalness: 0 });
const eyeMat = new THREE.MeshStandardMaterial({ color: 0x0e0e10, roughness: 0.18, metalness: 0 });
const whiteMat = new THREE.MeshBasicMaterial({ color: 0xffffff });
const mouthEdgeMat = new THREE.MeshBasicMaterial({ color: 0x4a1208 });
const mouthMat = new THREE.MeshBasicMaterial({ color: 0x7c1420 });
const throatMat = new THREE.MeshBasicMaterial({ color: 0x3a060c });
const tongueMat = new THREE.MeshStandardMaterial({ color: 0xff6b7c, roughness: 0.5, metalness: 0 });
const cheekMat = new THREE.MeshStandardMaterial({ color: 0xffa0a0, roughness: 0.7, metalness: 0 });

function smooth(g) {
  g.deleteAttribute('normal');
  if (g.attributes.uv) g.deleteAttribute('uv');
  const m = mergeVertices(g);
  m.computeVertexNormals();
  return m;
}

// Body shape: egg, wider at bottom
const H = 1.25, R = 1.0, D = 0.95, K = 0.15;
const bodyGeoRaw = new THREE.SphereGeometry(1, 72, 54);
{
  const p = bodyGeoRaw.attributes.position;
  for (let i = 0; i < p.count; i++) {
    const ux = p.getX(i), uy = p.getY(i), uz = p.getZ(i);
    const w = 1 - K * uy;
    p.setXYZ(i, R * ux * w, H * uy, D * uz * w);
  }
}
const bodyGeo = smooth(bodyGeoRaw);

function surfZ(x, y) {
  const uy = Math.max(-0.999, Math.min(0.999, y / H));
  const w = 1 - K * uy;
  const ux = x / (R * w);
  const s = Math.max(0, 1 - uy * uy - ux * ux);
  return D * w * Math.sqrt(s);
}
function surfPoint(x, y, off) {
  const e = 0.002;
  const z = surfZ(x, y);
  const dx = (surfZ(x + e, y) - surfZ(x - e, y)) / (2 * e);
  const dy = (surfZ(x, y + e) - surfZ(x, y - e)) / (2 * e);
  const n = new THREE.Vector3(-dx, -dy, 1).normalize();
  return [x + n.x * off, y + n.y * off, z + n.z * off];
}

const ell = (rx, ry, n = 2) => th => {
  const c = Math.abs(Math.cos(th)) / rx, s = Math.abs(Math.sin(th)) / ry;
  return Math.pow(Math.pow(c, n) + Math.pow(s, n), -1 / n);
};
// laughing mouth: smile-curved top edge, deep round bottom
const mouthFn = (rx, ryBot, a, b, ryTop) => th => {
  const s = Math.sin(th), c = Math.cos(th);
  const e = ell(rx, s > 0 ? ryTop : ryBot)(th);
  if (s <= 0) return e;
  const A = b * c * c / (rx * rx);
  let rt;
  if (A < 1e-9) rt = a / s;
  else {
    const disc = s * s - 4 * A * a;
    rt = disc >= 0 ? (s - Math.sqrt(disc)) / (2 * A) : Infinity;
  }
  return Math.min(e, rt);
};

function decal(cx, cy, rFn, offFn, mat, rings = 10, segs = 48) {
  const pos = [], idx = [];
  pos.push(...surfPoint(cx, cy, offFn(cx, cy, 0)));
  for (let i = 1; i <= rings; i++) {
    const f = i / rings;
    for (let j = 0; j < segs; j++) {
      const th = j / segs * Math.PI * 2;
      const r = rFn(th) * f;
      const x = cx + r * Math.cos(th), y = cy + r * Math.sin(th);
      pos.push(...surfPoint(x, y, offFn(x, y, f)));
    }
  }
  for (let j = 0; j < segs; j++) idx.push(0, 1 + j, 1 + (j + 1) % segs);
  for (let i = 1; i < rings; i++) {
    const a = 1 + (i - 1) * segs, b = 1 + i * segs;
    for (let j = 0; j < segs; j++) {
      const j1 = (j + 1) % segs;
      idx.push(a + j, b + j, b + j1);
      idx.push(a + j, b + j1, a + j1);
    }
  }
  const g = new THREE.BufferGeometry();
  g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
  g.setIndex(idx);
  g.computeVertexNormals();
  return new THREE.Mesh(g, mat);
}

// group placed at a surface point, contents in body-local coords
function anchoredGroup(name, parent, x, y) {
  const g = new THREE.Group(); g.name = name;
  const c = surfPoint(x, y, 0);
  g.position.set(c[0], c[1], c[2]);
  parent.add(g);
  const inner = new THREE.Group();
  inner.position.set(-c[0], -c[1], -c[2]);
  g.add(inner);
  return inner;
}

// ---------- Character ----------
const character = new THREE.Group();
character.name = 'character-nailong';
scene.add(character);

const body = new THREE.Group();
body.name = 'body';
body.position.set(0, 1.3, 0);
character.add(body);

const bodyMesh = new THREE.Mesh(bodyGeo, bodyMat);
bodyMesh.name = 'body-mesh';
body.add(bodyMesh);

// Belly patch
const bellyInner = anchoredGroup('belly', body, 0, -0.72);
bellyInner.add(decal(0, -0.72, ell(0.62, 0.42), (x, y, f) => 0.004 + 0.012 * (1 - f * f), bellyMat, 12, 64));

// Head (face features on the upper blob)
const head = new THREE.Group();
head.name = 'head';
body.add(head);

const eyes = new THREE.Group(); eyes.name = 'eyes'; head.add(eyes);
for (const sd of [1, -1]) {
  const ex = sd * 0.38, ey = 0.62, rx = 0.165, ry = 0.21;
  const eyeOff = (x, y, f) => -0.006 + 0.055 * (1 - f * f);
  const eyeH = (x, y) => {
    const f2 = ((x - ex) / rx) ** 2 + ((y - ey) / ry) ** 2;
    return -0.006 + 0.055 * Math.max(0, 1 - f2);
  };
  const inner = anchoredGroup(sd > 0 ? 'eye-left' : 'eye-right', eyes, ex, ey);
  inner.add(decal(ex, ey, ell(rx, ry), eyeOff, eyeMat, 10, 48));
  inner.add(decal(ex - 0.05, ey + 0.075, ell(0.06, 0.065), (x, y) => eyeH(x, y) + 0.006, whiteMat, 4, 24));
  inner.add(decal(ex + 0.055, ey - 0.085, ell(0.028, 0.028), (x, y) => eyeH(x, y) + 0.006, whiteMat, 3, 16));
}

// Mouth (big laughing mouth)
const mcy = 0.15;
const mouthInner = anchoredGroup('mouth', head, 0, mcy);
mouthInner.add(decal(0, mcy, mouthFn(0.5, 0.39, 0.105, 0.07, 0.32), () => 0.006, mouthEdgeMat, 12, 96));
mouthInner.add(decal(0, mcy, mouthFn(0.47, 0.36, 0.08, 0.07, 0.29), () => 0.012, mouthMat, 12, 96));
mouthInner.add(decal(0, mcy + 0.03, ell(0.26, 0.075), () => 0.017, throatMat, 6, 40));
const tongueParent = mouthInner.parent;
const tongueInner = anchoredGroup('tongue', tongueParent, 0, mcy - 0.21);
// tongueInner is offset relative to mouth group; compensate so coords stay body-local
{
  const mc = surfPoint(0, mcy, 0);
  tongueInner.position.x -= mc[0]; tongueInner.position.y -= mc[1]; tongueInner.position.z -= mc[2];
}
tongueInner.add(decal(0, mcy - 0.21, ell(0.24, 0.12), (x, y, f) => 0.02 + 0.02 * (1 - f * f), tongueMat, 8, 40));

// Cheeks
for (const sd of [1, -1]) {
  const inner = anchoredGroup(sd > 0 ? 'cheek-left' : 'cheek-right', head, sd * 0.68, 0.32);
  inner.add(decal(sd * 0.68, 0.32, ell(0.13, 0.08), () => 0.006, cheekMat, 5, 32));
}

// Arms
function makeArm(sd) {
  const g = new THREE.Group();
  g.name = sd > 0 ? 'arm-left' : 'arm-right';
  g.position.set(sd * 0.86, -0.12, 0.12);
  const upper = new THREE.Mesh(new THREE.CapsuleGeometry(0.15, 0.3, 8, 20), bodyMat);
  upper.position.y = -0.2;
  g.add(upper);
  const hand = new THREE.Group();
  hand.name = sd > 0 ? 'hand-left' : 'hand-right';
  hand.position.y = -0.42;
  g.add(hand);
  const hm = new THREE.Mesh(new THREE.SphereGeometry(0.17, 24, 16), bodyMat);
  hm.scale.set(1, 0.9, 1);
  hand.add(hm);
  g.rotation.set(-0.45, 0, sd * 0.65);
  return g;
}
body.add(makeArm(1));
body.add(makeArm(-1));

// Tail
const tail = new THREE.Group();
tail.name = 'tail';
tail.position.set(0, -0.95, -0.6);
tail.rotation.x = -Math.PI / 2 + 0.35;
{
  const pts = [new THREE.Vector2(0, -0.1)];
  for (let i = 0; i <= 18; i++) {
    const s = i / 18;
    const r = 0.32 * Math.sqrt(Math.max(0, 1 - s * s)) * (1 - 0.45 * s);
    pts.push(new THREE.Vector2(Math.max(r, 0), -0.1 + s * 0.95));
  }
  const tg = smooth(new THREE.LatheGeometry(pts, 28));
  tail.add(new THREE.Mesh(tg, bodyMat));
}
body.add(tail);

// Legs
function makeLeg(sd) {
  const g = new THREE.Group();
  g.name = sd > 0 ? 'leg-left' : 'leg-right';
  g.position.set(sd * 0.42, 0.35, 0.12);
  const thigh = new THREE.Mesh(new THREE.SphereGeometry(0.3, 28, 20), bodyMat);
  thigh.scale.set(1, 0.9, 1);
  g.add(thigh);
  const foot = new THREE.Group();
  foot.name = sd > 0 ? 'foot-left' : 'foot-right';
  foot.position.set(0, -0.2, 0.12);
  g.add(foot);
  const fm = new THREE.Mesh(new THREE.SphereGeometry(0.3, 28, 20), bodyMat);
  fm.scale.set(0.95, 0.5, 1.25);
  foot.add(fm);
  return g;
}
character.add(makeLeg(1));
character.add(makeLeg(-1));

// Soft ground shadow
{
  const sc = document.createElement('canvas');
  sc.width = sc.height = 64;
  const cx = sc.getContext('2d');
  const gr = cx.createRadialGradient(32, 32, 0, 32, 32, 32);
  gr.addColorStop(0, 'rgba(0,0,0,0.32)');
  gr.addColorStop(0.6, 'rgba(0,0,0,0.12)');
  gr.addColorStop(1, 'rgba(0,0,0,0)');
  cx.fillStyle = gr;
  cx.fillRect(0, 0, 64, 64);
  const shTex = new THREE.CanvasTexture(sc);
  shTex.colorSpace = THREE.SRGBColorSpace;
  const shadow = new THREE.Mesh(
    new THREE.PlaneGeometry(2.8, 2.2),
    new THREE.MeshBasicMaterial({ map: shTex, transparent: true, depthWrite: false })
  );
  shadow.rotation.x = -Math.PI / 2;
  shadow.position.y = 0.002;
  shadow.name = 'ground-shadow';
  scene.add(shadow);
}

function render(t) {
  renderer.render(scene, camera);
}

render(0);

window.nwb = {
  scene,
  renderer,
  cameras: { ref: camera },
  characters: [character],
  render,
  ready: true
};
