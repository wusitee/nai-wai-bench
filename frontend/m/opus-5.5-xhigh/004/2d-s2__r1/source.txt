<script>
(function () {
  var PERIOD = 1200;
  // [time, scaleX, scaleY, headRot, leftHandX, leftHandY, rightHandX, rightHandY]
  var K = [
    [0,    1.00, 1.00,  0.0, 69, 87,  210, 80],
    [160,  1.05, 1.05,  5.0, 18, 70,  158, 95],
    [340,  1.08, 0.96,  0.0, 55, 100, 212, 92],
    [500,  1.20, 0.95,  1.5, 20, 103, 183, 125],
    [680,  1.05, 1.00,  3.0, 20, 93,  178, 108],
    [840,  1.02, 1.02,  0.0, 68, 95,  210, 92],
    [1020, 1.20, 0.98,  2.5, 18, 88,  173, 110],
    [1180, 1.12, 0.955, 2.5, 48, 102, 208, 103],
    [1200, 1.00, 1.00,  0.0, 69, 87,  210, 80]
  ];

  var el = function (id) { return document.getElementById(id); };
  var rig = el('body-rig');
  var head = el('head');
  var shadow = el('shadow');
  var parts = {
    left:  { fill: el('arm-left-fill'),  outline: el('arm-left-outline'),  shine: el('arm-left-shine'),  hand: el('hand-left'),  bx: 80,  by: 140, side: -1 },
    right: { fill: el('arm-right-fill'), outline: el('arm-right-outline'), shine: el('arm-right-shine'), hand: el('hand-right'), bx: 178, by: 142, side: 1 }
  };

  function f(n) { return (Math.round(n * 100) / 100).toString(); }

  function sample(t) {
    if (!isFinite(t)) t = 0;
    t = ((t % PERIOD) + PERIOD) % PERIOD;
    var i = 0;
    while (i < K.length - 2 && t >= K[i + 1][0]) i++;
    var a = K[i], b = K[i + 1];
    var u = (t - a[0]) / (b[0] - a[0]);
    if (u < 0) u = 0; if (u > 1) u = 1;
    var e = 0.5 - 0.5 * Math.cos(Math.PI * u);
    var out = [];
    for (var j = 1; j < a.length; j++) out.push(a[j] + (b[j] - a[j]) * e);
    return out;
  }

  function quadPt(S, C, W, s) {
    var m = 1 - s;
    return [m * m * S[0] + 2 * m * s * C[0] + s * s * W[0],
            m * m * S[1] + 2 * m * s * C[1] + s * s * W[1]];
  }
  function quadSeg(S, C, W, a, b) {
    var p0 = quadPt(S, C, W, a), p2 = quadPt(S, C, W, b);
    var w0 = (1 - a) * (1 - b), w1 = (1 - a) * b + a * (1 - b), w2 = a * b;
    var q = [w0 * S[0] + w1 * C[0] + w2 * W[0], w0 * S[1] + w1 * C[1] + w2 * W[1]];
    return 'M' + f(p0[0]) + ' ' + f(p0[1]) + ' Q' + f(q[0]) + ' ' + f(q[1]) + ' ' + f(p2[0]) + ' ' + f(p2[1]);
  }

  function setArm(p, hx, hy, sx, sy) {
    var Sx = 121 + (p.bx - 121) * sx;
    var Sy = 230 - (230 - p.by) * sy;
    var dx = hx - Sx, dy = hy - Sy;
    var L = Math.sqrt(dx * dx + dy * dy) || 1;
    var ux = dx / L, uy = dy / L;
    var Wx = hx - ux * 9, Wy = hy - uy * 9;
    var nx = -uy, ny = ux;
    var Cx = (Sx + Wx) / 2 + nx * 4 * p.side;
    var Cy = (Sy + Wy) / 2 + ny * 4 * p.side;
    var S = [Sx, Sy], C = [Cx, Cy], W = [Wx, Wy];
    p.fill.setAttribute('d', 'M' + f(Sx) + ' ' + f(Sy) + ' Q' + f(Cx) + ' ' + f(Cy) + ' ' + f(Wx) + ' ' + f(Wy));
    p.outline.setAttribute('d', quadSeg(S, C, W, 0.4, 1));
    p.shine.setAttribute('d', quadSeg(S, C, W, 0.5, 0.85));
    var ang = Math.atan2(dx, -dy) * 180 / Math.PI;
    var rot = 0.5 * ang + 16 * p.side;
    p.hand.setAttribute('transform', 'translate(' + f(hx) + ' ' + f(hy) + ') rotate(' + f(rot) + ')');
  }

  function render(t) {
    var v = sample(Number(t));
    var sx = v[0], sy = v[1], hr = v[2];
    rig.setAttribute('transform', 'translate(121 230) scale(' + f(sx * 1000) / 1000 + ' ' + f(sy * 1000) / 1000 + ') translate(-121 -230)');
    head.setAttribute('transform', 'rotate(' + f(hr) + ' 121 40)');
    shadow.setAttribute('rx', f(64 * sx));
    shadow.setAttribute('cx', f(121 + 4 * sx));
    setArm(parts.left, v[3], v[4], sx, sy);
    setArm(parts.right, v[5], v[6], sx, sy);
  }

  window.nwb = { render: render };
  render(0);
})();
</script>
