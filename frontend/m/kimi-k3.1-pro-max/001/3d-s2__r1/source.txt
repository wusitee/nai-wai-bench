function standard(color, roughness = 0.65) {

function addMesh(parent, name, geometry, material, cast = true, receive = true) {
  const mesh = new THREE.Mesh(geometry, material);
  mesh.name = name;
  mesh.castShadow = cast;
  mesh.receiveShadow = receive;
  parent.add(mesh);
  return mesh;
}

let seed = 73129;
function random() {
  seed = (seed * 1664525 + 1013904223) >>> 0;
  return seed / 4294967296;

function makeWoodTexture() {
  const canvas = document.createElement('canvas');
  canvas.width = canvas.height = 64;
  const g = canvas.getContext('2d');
  g.fillStyle = '#aa713b';
  g.fillRect(0, 0, 64, 64);
  for (let i = 0; i < 46; i++) {
    const y = random() * 64;
    g.strokeStyle = `rgba(${75 + random() * 35},${42 + random() * 25},${20 + random() * 15},${0.05 + random() * 0.12})`;
    g.lineWidth = 0.35 + random() * 1.1;
    g.beginPath();
    g.moveTo(0, y);
    for (let x = 0; x <= 64; x += 8) {
      g.lineTo(x, y + Math.sin(x * 0.12 + i) * (0.5 + random()));
    }
    g.stroke();
  }
  for (let i = 0; i < 5; i++) {
    g.strokeStyle = 'rgba(74,42,20,0.10)';
    g.beginPath();
    g.ellipse(random() * 64, random() * 64, 5 + random() * 8, 1 + random() * 2, 0, 0, Math.PI * 2);
    g.stroke();
  }
  const texture = new THREE.CanvasTexture(canvas);
  texture.colorSpace = THREE.SRGBColorSpace;
  texture.wrapS = texture.wrapT = THREE.RepeatWrapping;
  texture.repeat.set(3, 2);
  return texture;
}

function radiusAt(samples, y) {
  if (y <= samples[0].y) return Math.max(0, samples[0].x);
  for (let i = 1; i < samples.length; i++) {
    if (samples[i].y >= y) {
      const a = samples[i - 1], b = samples[i];
      const f = (y - a.y) / Math.max(0.00001, b.y - a.y);
      return Math.max(0, THREE.MathUtils.lerp(a.x, b.x, f));
    }
  }
  return 0;
}

function makeOrganicBody(parent, name, controls, zScale, material) {
  const points = controls.map(([y, r]) => new THREE.Vector3(r, y, 0));
  const curve = new THREE.CatmullRomCurve3(points, false, 'centripetal');
  const samples = curve.getPoints(128);
  const profile = samples.map(p => new THREE.Vector2(Math.max(0, p.x), p.y));
  const geometry = new THREE.LatheGeometry(profile, 64);
  const body = addMesh(parent, name, geometry, material);
  body.scale.z = zScale;
  return { body, samples, zScale };
}

function makeBelly(parent, name, bodyData, y0, y1, maxAngle, material) {
  const rows = 32, cols = 26;
  const positions = [], indices = [];
  const { samples, zScale } = bodyData;
  for (let v = 0; v <= rows; v++) {
    const fv = v / rows;
    const y = THREE.MathUtils.lerp(y0, y1, fv);
    const half = maxAngle * Math.pow(Math.max(0, Math.sin(Math.PI * fv)), 0.56);
    const radius = radiusAt(samples, y);
    for (let u = 0; u <= cols; u++) {
      const fu = u / cols;
      const angle = (fu * 2 - 1) * half;
      positions.push(
        radius * Math.sin(angle),
        y,
        (radius * Math.cos(angle) + 0.012) * zScale
      );
    }
  }
  for (let v = 0; v < rows; v++) {
    for (let u = 0; u < cols; u++) {
      const a = v * (cols + 1) + u;
      const b = (v + 1) * (cols + 1) + u;
      const c = b + 1;
      const d = a + 1;
      indices.push(a, d, b, b, d, c);
    }
  }
  const geometry = new THREE.BufferGeometry();
  geometry.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
  geometry.setIndex(indices);
  geometry.computeVertexNormals();
  return addMesh(parent, name, geometry, material, false, true);
}

function makeLimb(parent, name, points, r0, r1, material, steps = 32, radial = 14) {
  const group = new THREE.Group();
  group.name = name;
  parent.add(group);
  const curve = new THREE.CatmullRomCurve3(points.map(p => new THREE.Vector3(...p)), false, 'centripetal');
  const frames = curve.computeFrenetFrames(steps, false);
  const positions = [], indices = [];
  for (let i = 0; i <= steps; i++) {
    const t = i / steps;
    const center = curve.getPointAt(t);
    const smooth = t * t * (3 - 2 * t);
    const radius = THREE.MathUtils.lerp(r0, r1, smooth) * (1 - 0.035 * Math.sin(Math.PI * t));
    for (let j = 0; j <= radial; j++) {
      const angle = j / radial * Math.PI * 2;
      const normal = frames.normals[i];
      const binormal = frames.binormals[i];
      positions.push(
        center.x + radius * (Math.cos(angle) * normal.x + Math.sin(angle) * binormal.x),
        center.y + radius * (Math.cos(angle) * normal.y + Math.sin(angle) * binormal.y),
        center.z + radius * (Math.cos(angle) * normal.z + Math.sin(angle) * binormal.z)
      );
    }
  }
  for (let i = 0; i < steps; i++) {
    for (let j = 0; j < radial; j++) {
      const a = i * (radial + 1) + j;
      const b = (i + 1) * (radial + 1) + j;
      const c = b + 1;
      const d = a + 1;
      indices.push(a, d, b, b, d, c);
    }
  }
  const startCenter = positions.length / 3;
  const p0 = curve.getPointAt(0);
  positions.push(p0.x, p0.y, p0.z);
  const endCenter = positions.length / 3;
  const p1 = curve.getPointAt(1);
  positions.push(p1.x, p1.y, p1.z);
  for (let j = 0; j < radial; j++) {
    indices.push(startCenter, j + 1, j);
    const a = steps * (radial + 1) + j;
    indices.push(endCenter, a, a + 1);
  }
  const geometry = new THREE.BufferGeometry();
  geometry.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
  geometry.setIndex(indices);
  geometry.computeVertexNormals();
  addMesh(group, name + '-surface', geometry, material);
  return group;
}

function pageHeight(u, v) {
  return 0.022 + 0.038 * Math.sin(Math.PI * u) + 0.018 * (1 - u) + 0.008 * Math.sin(v * Math.PI);
}

function makeOpenNotebook(name, x, z, pageWidth, depth, rotation) {
  const book = new THREE.Group();
  book.name = name;
  book.position.set(x, 0.735, z);
  book.rotation.y = rotation;
  scene.add(book);

  const lineMaterial = new THREE.LineBasicMaterial({
    color: 0x9da9a5,
    transparent: true,
    opacity: 0.32,
    depthWrite: false
  });

  for (const side of [-1, 1]) {
    const pageGroup = new THREE.Group();
    pageGroup.name = side < 0 ? name + '-left-page' : name + '-right-page';
    book.add(pageGroup);
    addBox(pageGroup, 'page-block', pageWidth, 0.025, depth, side * pageWidth / 2, 0.005, 0, M.paperEdge);

    const cols = 20, rows = 14;
    const positions = [], indices = [];
    for (let v = 0; v <= rows; v++) {
      for (let u = 0; u <= cols; u++) {
        const fu = u / cols, fv = v / rows;
        positions.push(side * fu * pageWidth, pageHeight(fu, fv), (fv - 0.5) * depth);
      }
    }
    for (let v = 0; v < rows; v++) {
      for (let u = 0; u < cols; u++) {
        const a = v * (cols + 1) + u;
        const b = (v + 1) * (cols + 1) + u;
        const c = b + 1;
        const d = a + 1;
        if (side > 0) indices.push(a, b, d, b, c, d);
        else indices.push(a, d, b, b, d, c);
      }
    }
    const geometry = new THREE.BufferGeometry();
    geometry.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
    geometry.setIndex(indices);
    geometry.computeVertexNormals();
    addMesh(pageGroup, 'curved-paper', geometry, M.paper, false, true);

    const linePositions = [];
    for (let line = 1; line <= 7; line++) {
      const v = line / 9;
      for (let j = 0; j < 12; j++) {
        for (const u of [0.13 + j * 0.061, 0.13 + (j + 1) * 0.061]) {
          linePositions.push(side * u * pageWidth, pageHeight(u, v) + 0.004, (v - 0.5) * depth);
        }
      }
    }
    const lineGeometry = new THREE.BufferGeometry();
    lineGeometry.setAttribute('position', new THREE.Float32BufferAttribute(linePositions, 3));
    pageGroup.add(new THREE.LineSegments(lineGeometry, lineMaterial));
  }

  const spineCurve = new THREE.CatmullRomCurve3([
    new THREE.Vector3(0, 0.035, -depth / 2),
    new THREE.Vector3(0, 0.042, 0),
    new THREE.Vector3(0, 0.035, depth / 2)
  ]);
  addMesh(book, name + '-spine', new THREE.TubeGeometry(spineCurve, 16, 0.011, 6, false), M.paperEdge, false, true);
  return book;
}

function makePencil(name, start, end, colorMaterial = M.pencil, radius = 0.027) {
  const a = new THREE.Vector3(...start), b = new THREE.Vector3(...end);
  const direction = b.clone().sub(a);
  const length = direction.length();
  direction.normalize();
  const group = new THREE.Group();
  group.name = name;
  group.position.copy(a).add(b).multiplyScalar(0.5);
  group.quaternion.setFromUnitVectors(yAxis, direction);
  scene.add(group);

  const bodyLength = Math.max(0.05, length - 0.12);
  const body = addMesh(group, name + '-body', new THREE.CylinderGeometry(radius, radius, bodyLength, 10), colorMaterial);
  body.position.y = -0.055;
  const wood = addMesh(group, name + '-wood-tip', new THREE.ConeGeometry(radius * 1.18, 0.12, 10), M.pencilWood);
  wood.position.y = length / 2 - 0.06;
  const graphite = addMesh(group, name + '-graphite', new THREE.ConeGeometry(radius * 0.43, 0.045, 8), M.graphite);
  graphite.position.y = length / 2 - 0.006;
  return group;
}

const LOOP = 1500;

const mouthKeys = [[0, 0], [200, 0], [400, 0.05], [600, 0.80], [800, 1], [1000, 1], [1200, 1], [1400, 1], [1500, 0]];

function sample(keys, t) {
  if (t <= keys[0][0]) return keys[0][1];
  for (let i = 1; i < keys.length; i++) {
    if (t <= keys[i][0]) {
      const a = keys[i - 1], b = keys[i];
      let u = (t - a[0]) / (b[0] - a[0]);
      u = u * u * (3 - 2 * u);
      return THREE.MathUtils.lerp(a[1], b[1], u);
    }
  }
  return keys[keys.length - 1][1];
}

function render(t) {
  const phase = ((t % LOOP) + LOOP) % LOOP;
  const lean = sample(leanKeys, phase);
  const nuzzle = sample(nuzzleKeys, phase);
  const retreat = sample(retreatKeys, phase);
  const patY = sample(patYKeys, phase);
  const patX = sample(patXKeys, phase);
  const mouthOpen = sample(mouthKeys, phase);
  const writingStroke = Math.sin(phase * Math.PI * 2 / 300) * (1 - retreat);

  leftBody.position.set(0.05 * lean, 0.42 * lean, 0);
  leftBody.rotation.set(0, 0.06 * lean, -0.16 * lean);

  leftHead.position.set(0.15 + 0.45 * lean, 0.42 * lean, 0);
  leftHead.rotation.set(0, 0.06 * lean, -0.035 * lean);

  leftEyeRight.scale.setScalar(0.91 * (1 + 0.06 * lean));
  leftEyeLeftBall.position.set(0.018 + 0.018 * lean, -0.025, 0.083);
  leftEyeRightBall.position.set(0.018 + 0.018 * lean, -0.035, 0.083);

  leftMouth.visible = mouthOpen < 0.45;
  leftOpenMouth.visible = mouthOpen > 0.02;
  leftOpenMouth.scale.set(0.068 * mouthOpen, 0.095 * mouthOpen, 0.028);

  writingArm.position.set(-0.82 - 0.18 * retreat, 2.05 - 1.10 * retreat, 0.05);
  writingArm.rotation.set(0, 0, -1.05 * retreat + 0.018 * writingStroke);
  writingHand.rotation.set(
    0.05 * writingStroke,
    0,
    -0.28 - 0.12 * retreat + 0.04 * writingStroke
  );

  rightCharacter.position.set(1.45 - 0.12 * nuzzle, 0.85 + 0.02 * nuzzle, -0.45 + 0.03 * nuzzle);
  rightCharacter.rotation.set(0, -0.25 - 0.12 * nuzzle, 0.08 + 0.05 * nuzzle);

  rightComfortHand.rotation.set(0.08 * patY, 0, 0.18 - 0.12 * patY);

  renderer.render(scene, camera);
}
