function makeGroup(name, parent, position = [0, 0, 0]) {
  const g = new THREE.Group();
  g.name = name;
  g.position.set(...position);
  parent.add(g);
  return g;
}

function ellipsoid(parent, name, position, scale, material, rotation = [0, 0, 0]) {
  const m = new THREE.Mesh(unitSphere, material);
  m.name = name;
  m.position.set(...position);
  m.scale.set(...scale);
  m.rotation.set(...rotation);
  parent.add(m);
  return m;
}

function capsuleBetween(parent, name, a, b, radius, material) {
  const start = new THREE.Vector3(...a);
  const end = new THREE.Vector3(...b);
  const direction = end.clone().sub(start);
  const length = direction.length();
  const geometry = new THREE.CapsuleGeometry(
    radius,
    Math.max(0.001, length - 2 * radius),
    8,
    20
  );
  const m = new THREE.Mesh(geometry, material);
  m.name = name;
  m.position.copy(start).add(end).multiplyScalar(0.5);
  m.quaternion.setFromUnitVectors(
    new THREE.Vector3(0, 1, 0),
    direction.normalize()
  );
  parent.add(m);
  return m;
}

function tube(parent, name, curve, radius, material, segments = 40) {
  const m = new THREE.Mesh(
    new THREE.TubeGeometry(curve, segments, radius, 10, false),
    material
  );
  m.name = name;
  parent.add(m);
  return m;
}

function taperedTube(points, radii, segments = 56, radialSegments = 18) {
  const curve = new THREE.CatmullRomCurve3(points);
  const frames = curve.computeFrenetFrames(segments, false);
  const positions = [];
  const indices = [];

  for (let i = 0; i <= segments; i++) {
    const t = i / segments;
    const center = curve.getPointAt(t);
    const k = t * (radii.length - 1);
    const ri = Math.min(Math.floor(k), radii.length - 2);
    const radius = THREE.MathUtils.lerp(radii[ri], radii[ri + 1], k - ri);

    for (let j = 0; j <= radialSegments; j++) {
      const angle = j / radialSegments * Math.PI * 2;
      const c = -Math.cos(angle) * radius;
      const s = Math.sin(angle) * radius;
      positions.push(
        center.x + c * frames.normals[i].x + s * frames.binormals[i].x,
        center.y + c * frames.normals[i].y + s * frames.binormals[i].y,
        center.z + c * frames.normals[i].z + s * frames.binormals[i].z
      );
    }
  }

  for (let i = 1; i <= segments; i++) {
    for (let j = 1; j <= radialSegments; j++) {
      const a = (radialSegments + 1) * (i - 1) + j - 1;
      const b = (radialSegments + 1) * i + j - 1;
      const c = (radialSegments + 1) * i + j;
      const d = (radialSegments + 1) * (i - 1) + j;
      indices.push(a, b, d, b, c, d);
    }
  }

  const startCenter = positions.length / 3;
  const start = curve.getPointAt(0);
  positions.push(start.x, start.y, start.z);
  for (let j = 0; j < radialSegments; j++) {
    indices.push(startCenter, j, j + 1);
  }

  const endCenter = positions.length / 3;
  const end = curve.getPointAt(1);
  positions.push(end.x, end.y, end.z);
  const endRing = segments * (radialSegments + 1);
  for (let j = 0; j < radialSegments; j++) {
    indices.push(endCenter, endRing + j + 1, endRing + j);
  }

  const geometry = new THREE.BufferGeometry();
  geometry.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
  geometry.setIndex(indices);
  geometry.computeVertexNormals();
  return geometry;
}
