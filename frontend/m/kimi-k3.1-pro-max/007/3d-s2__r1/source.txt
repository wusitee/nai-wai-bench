function ellipsoid(parent, name, position, scale, material, rotationZ = 0) {
  const mesh = new THREE.Mesh(unitSphere, material);
  mesh.name = name;
  mesh.position.copy(position);
  mesh.scale.copy(scale);
  mesh.rotation.z = rotationZ;
  parent.add(mesh);
  return mesh;
}

function smoothLathe(profile, material, zScale = 1, segments = 48) {
  const curve = new THREE.CatmullRomCurve3(
    profile.map(([radius, y]) => new THREE.Vector3(radius, y, 0)),
    false,
    'centripetal'
  );
  const points = curve.getPoints(72).map(
    p => new THREE.Vector2(Math.max(0, p.x), p.y)
  );
  const geometry = new THREE.LatheGeometry(points, segments);
  geometry.scale(1, 1, zScale);
  geometry.computeVertexNormals();
  return new THREE.Mesh(geometry, material);
}

function cylinderBetween(parent, name, a, b, radiusA, radiusB, material) {
  const direction = new THREE.Vector3().subVectors(b, a);
  const length = direction.length();
  const geometry = new THREE.CylinderGeometry(
    radiusB, radiusA, length, 20, 2, false
  );
  const mesh = new THREE.Mesh(geometry, material);
  mesh.name = name;
  mesh.position.copy(a).add(b).multiplyScalar(0.5);
  mesh.quaternion.setFromUnitVectors(upAxis, direction.normalize());
  mesh.userData.baseLength = length;
  parent.add(mesh);
  return mesh;
}

function setSegment(mesh, a, b) {
  const direction = new THREE.Vector3().subVectors(b, a);
  const length = Math.max(0.00001, direction.length());
  mesh.position.copy(a).add(b).multiplyScalar(0.5);
  mesh.quaternion.setFromUnitVectors(upAxis, direction.normalize());
  mesh.scale.set(1, length / mesh.userData.baseLength, 1);
}

function capsuleBetween(parent, name, a, b, radius, material) {
  const direction = new THREE.Vector3().subVectors(b, a);
  const length = direction.length();
  const geometry = new THREE.CapsuleGeometry(
    radius,
    Math.max(0.001, length - radius * 2),
    6,
    12
  );
  const mesh = new THREE.Mesh(geometry, material);
  mesh.name = name;
  mesh.position.copy(a).add(b).multiplyScalar(0.5);
  mesh.quaternion.setFromUnitVectors(upAxis, direction.normalize());
  parent.add(mesh);
  return mesh;
}

function faceTube(parent, name, points, radius) {

  const mesh = new THREE.Mesh(
    new THREE.TubeGeometry(curve, 18, radius, 7, false),
    faceLineMaterial
  );
  mesh.name = name;
  parent.add(mesh);
  return mesh;
}

function updateMouth(curve) {
  const points = [];
  for (let i = 0; i <= 8; i++) {
    const u = -1 + i / 4;
    points.push(new THREE.Vector3(
      -0.14 + (i / 8) * 0.35,
      -0.15 - 0.02 * (1 - u * u) - curve * 0.075 * (1 - u * u),
      0.512 + 0.028 * (1 - u * u)
    ));
  }
  for (let i = 0; i < mouthSegments.length; i++) {
    setSegment(mouthSegments[i], points[i], points[i + 1]);
  }
}

function cloneValue(value) {
  if (Array.isArray(value)) return value.map(cloneValue);
  if (typeof value === 'object' && value !== null) {
    const result = {};
    for (const key of Object.keys(value)) result[key] = cloneValue(value[key]);
    return result;
  }
  return value;
}

function mixValue(a, b, u) {
  if (typeof a === 'number') return a + (b - a) * u;
  if (Array.isArray(a)) return a.map((value, i) => mixValue(value, b[i], u));
  const result = {};
  for (const key of Object.keys(a)) result[key] = mixValue(a[key], b[key], u);
  return result;
}

function samplePose(time) {
  let index = 0;
  while (index < keyTimes.length - 2 && time > keyTimes[index + 1]) index++;
  const start = keyTimes[index];
  const end = keyTimes[index + 1];
  let u = THREE.MathUtils.clamp((time - start) / (end - start), 0, 1);
  u = u * u * (3 - 2 * u);
  return mixValue(poses[index], poses[index + 1], u);
}

const shoulderVector = new THREE.Vector3();
const elbowVector = new THREE.Vector3();
const wristVector = new THREE.Vector3();
const localElbowVector = new THREE.Vector3();
const localWristVector = new THREE.Vector3();

function applyArm(rig, data) {
  shoulderVector.fromArray(data.shoulder);
  elbowVector.fromArray(data.elbow);
  wristVector.fromArray(data.wrist);
  localElbowVector.subVectors(elbowVector, shoulderVector);
  localWristVector.subVectors(wristVector, elbowVector);

  rig.group.position.copy(shoulderVector);
  setSegment(rig.upper, origin, localElbowVector);
  rig.elbowCap.position.copy(localElbowVector);
  rig.forearm.position.copy(localElbowVector);
  setSegment(rig.lower, origin, localWristVector);
  rig.wristCap.position.copy(localWristVector);
  rig.hand.position.copy(localWristVector);
  rig.hand.rotation.set(...data.rotation);
  rig.hand.scale.fromArray(data.scale);
}

function render(t) {
  const numericTime = Number.isFinite(t) ? t : 0;
  const time = ((numericTime % LOOP) + LOOP) % LOOP;
  const p = samplePose(time);

  character.position.fromArray(p.position);
  character.rotation.set(...p.rotation);
  torsoMesh.scale.fromArray(p.torsoScale);
  bellyMesh.position.fromArray(p.bellyPosition);
  bellyMesh.scale.fromArray(p.bellyScale);

  head.position.fromArray(p.headPosition);
  head.rotation.set(...p.headRotation);
  head.scale.fromArray(p.headScale);

  leftLeg.rotation.set(...p.leftLeg);
  rightLeg.rotation.set(...p.rightLeg);

  applyArm(leftArmRig, p.leftArm);
  applyArm(rightArmRig, p.rightArm);

  closedEyes.visible = p.eye < 0.5;
  openEyes.visible = p.eye > 0.01;
  for (const mesh of openEyeMeshes) {
    const base = mesh.userData.baseScale;
    mesh.scale.set(
      base.x * (0.70 + 0.30 * p.eye),
      base.y * Math.max(0.001, p.eye),
      base.z
    );
  }
  leftLid.position.y = leftLid.userData.baseY + p.eye * 0.055;
  rightLid.position.y = rightLid.userData.baseY + p.eye * 0.055;

  updateMouth(p.mouth);

  heldObject.visible = p.object > 0.001;
  heldObject.scale.setScalar(Math.max(0.0001, p.object));

  camera.position.set(p.camera[0], p.camera[1], 10);
  camera.lookAt(p.camera[0], p.camera[1], 0);
  camera.zoom = p.camera[2];
  camera.updateProjectionMatrix();

  renderer.render(scene, camera);
}
