import * as THREE from 'three';

const renderer = new THREE.WebGLRenderer({

  antialias: true
});

const scene = new THREE.Scene();
scene.background = new THREE.Color('#ededeb');

const camera = new THREE.OrthographicCamera(
  -2.2857142857, 2.2857142857, 2.8, -2.8, 0.1, 50
);
camera.position.set(0, 2.62, 10);
camera.lookAt(0, 2.62, 0);
camera.updateProjectionMatrix();

scene.add(new THREE.HemisphereLight(0xffffff, 0x817d5b, 1.8));

const key = new THREE.DirectionalLight(0xfffdf2, 2.2);
key.position.set(-3.5, 7, 6);
scene.add(key);

const fill = new THREE.DirectionalLight(0xf4f6ff, 0.65);
fill.position.set(4, 3, -5);
scene.add(fill);

const yellow = new THREE.MeshStandardMaterial({
  color: '#bdb83a',
  roughness: 0.96,
  metalness: 0
});
const headYellow = new THREE.MeshStandardMaterial({
  color: '#b8b63e',
  roughness: 0.98,
  metalness: 0
});
const muzzleYellow = new THREE.MeshStandardMaterial({
  color: '#c0bb49',
  roughness: 1,
  metalness: 0
});
const handMaterial = new THREE.MeshStandardMaterial({
  color: '#484e45',
  roughness: 0.94,
  metalness: 0
});
const faceLineMaterial = new THREE.MeshStandardMaterial({
  color: '#62643a',
  roughness: 1,
  metalness: 0
});

const unitSphere = new THREE.SphereGeometry(1, 28, 20);

function ellipsoid(parent, name, position, scale, material, rotationZ = 0) {
  const mesh = new THREE.Mesh(unitSphere, material);
  mesh.name = name;
  mesh.position.copy(position);
  mesh.scale.copy(scale);
  mesh.rotation.z = rotationZ;
  parent.add(mesh);
  return mesh;
}

function smoothLathe(profile, material, zScale = 1, segments = 48) {
  const curve = new THREE.CatmullRomCurve3(
    profile.map(([radius, y]) => new THREE.Vector3(radius, y, 0)),
    false,
    'centripetal'
  );
  const points = curve.getPoints(72).map(
    p => new THREE.Vector2(Math.max(0, p.x), p.y)
  );
  const geometry = new THREE.LatheGeometry(points, segments);
  geometry.scale(1, 1, zScale);
  geometry.computeVertexNormals();
  return new THREE.Mesh(geometry, material);
}

function cylinderBetween(parent, name, a, b, radiusA, radiusB, material) {
  const direction = new THREE.Vector3().subVectors(b, a);
  const length = direction.length();
  const geometry = new THREE.CylinderGeometry(
    radiusB, radiusA, length, 20, 2, false
  );
  const mesh = new THREE.Mesh(geometry, material);
  mesh.name = name;
  mesh.position.copy(a).add(b).multiplyScalar(0.5);
  mesh.quaternion.setFromUnitVectors(
    new THREE.Vector3(0, 1, 0),
    direction.normalize()
  );
  parent.add(mesh);
  return mesh;
}

function capsuleBetween(parent, name, a, b, radius, material) {
  const direction = new THREE.Vector3().subVectors(b, a);
  const length = direction.length();
  const geometry = new THREE.CapsuleGeometry(
    radius,
    Math.max(0.001, length - radius * 2),
    6,
    12
  );
  const mesh = new THREE.Mesh(geometry, material);
  mesh.name = name;
  mesh.position.copy(a).add(b).multiplyScalar(0.5);
  mesh.quaternion.setFromUnitVectors(
    new THREE.Vector3(0, 1, 0),
    direction.normalize()
  );
  parent.add(mesh);
  return mesh;
}

function faceTube(parent, name, points, radius) {
  const curve = new THREE.CatmullRomCurve3(points, false, 'centripetal');
  const mesh = new THREE.Mesh(
    new THREE.TubeGeometry(curve, 18, radius, 7, false),
    faceLineMaterial
  );
  mesh.name = name;
  parent.add(mesh);
  return mesh;
}

const character = new THREE.Group();
character.name = 'character-yellow-bellied-figure';
character.position.x = -0.44;

const legs = new THREE.Group();

function makeLeg(name, x) {
  const leg = new THREE.Group();
  leg.name = name;
  leg.position.set(x, 1.45, 0);
  legs.add(leg);

  const mesh = smoothLathe([
    [0, -2.10],
    [0.17, -2.10],
    [0.22, -1.91],
    [0.235, -1.51],
    [0.27, -1.00],
    [0.33, -0.55],
    [0.39, -0.10],
    [0.36, 0.17],
    [0, 0.23]
  ], yellow, 0.84);
  mesh.name = `${name}-volume`;
  leg.add(mesh);

  const foot = new THREE.Group();
  foot.name = `${name}-foot`;
  foot.position.set(0, -2.04, 0.12);
  leg.add(foot);
  ellipsoid(
    foot,
    `${name}-foot-volume`,
    new THREE.Vector3(0, -0.04, 0.10),
    new THREE.Vector3(0.25, 0.18, 0.40),
    yellow
  );
  return leg;
}

const torsoMesh = smoothLathe([

const belly = new THREE.Group();
belly.name = 'belly';
torso.add(belly);

const bellyGeometry = new THREE.SphereGeometry(1, 48, 32);
const bellyPositions = bellyGeometry.attributes.position;
const bellyColors = new Float32Array(bellyPositions.count * 3);
const cream = new THREE.Color('#d9d5a8');
const edgeYellow = new THREE.Color('#bdb83a');
const lowerCream = new THREE.Color('#d2cc94');

for (let i = 0; i < bellyPositions.count; i++) {
  const x = bellyPositions.getX(i);
  const y = bellyPositions.getY(i);
  const radial = Math.min(1, Math.hypot(x, y));
  const edge = THREE.MathUtils.smoothstep(radial, 0.58, 0.985);
  const vertical = THREE.MathUtils.clamp((y + 1) * 0.5, 0, 1);
  const centerColor = lowerCream.clone().lerp(cream, 0.35 + vertical * 0.65);
  centerColor.lerp(edgeYellow, edge);
  bellyColors[i * 3] = centerColor.r;
  bellyColors[i * 3 + 1] = centerColor.g;
  bellyColors[i * 3 + 2] = centerColor.b;
}
bellyGeometry.setAttribute('color', new THREE.BufferAttribute(bellyColors, 3));

const bellyMesh = new THREE.Mesh(
  bellyGeometry,
  new THREE.MeshStandardMaterial({
    color: 0xffffff,
    vertexColors: true,
    roughness: 1,
    metalness: 0

);
bellyMesh.name = 'large-cream-belly';
bellyMesh.position.set(0.06, 2.78, 0.39);
bellyMesh.scale.set(0.91, 0.92, 0.40);
belly.add(bellyMesh);

function makeHand(hand, side) {
  const outward = side === 'left' ? -1 : 1;
  const inner = -outward;
  const tilt = side === 'left' ? -0.12 : 0.10;

  ellipsoid(
    hand,
    `${side}-palm`,
    new THREE.Vector3(outward * 0.025, -0.20, 0.02),
    new THREE.Vector3(0.13, 0.22, 0.095),
    handMaterial,
    tilt
  );

  const bases = [-0.088, -0.030, 0.030, 0.086];
  const lengths = [0.27, 0.33, 0.35, 0.29];
  for (let i = 0; i < 4; i++) {
    const bx = bases[i] + outward * 0.025;
    const fan = (i - 1.5) * 0.018 + outward * 0.025;
    capsuleBetween(
      hand,
      `${side}-finger-${i + 1}`,
      new THREE.Vector3(bx, -0.34, 0.025),
      new THREE.Vector3(bx + fan, -0.34 - lengths[i], 0.035),
      0.028,
      handMaterial
    );
  }

  capsuleBetween(
    hand,
    `${side}-thumb`,
    new THREE.Vector3(inner * 0.09, -0.13, 0.025),
    new THREE.Vector3(inner * 0.17, -0.32, 0.055),
    0.037,
    handMaterial
  );
}

function makeArm(side, shoulder, elbow, wrist) {
  const arm = new THREE.Group();
  arm.name = `${side}-arm`;
  arm.position.copy(shoulder);
  arms.add(arm);

  const localElbow = new THREE.Vector3().subVectors(elbow, shoulder);
  const localWrist = new THREE.Vector3().subVectors(wrist, elbow);

  ellipsoid(
    arm,
    `${side}-shoulder-cap`,
    new THREE.Vector3(),
    new THREE.Vector3(0.17, 0.17, 0.15),
    yellow
  );
  cylinderBetween(
    arm,
    `${side}-upper-arm`,
    new THREE.Vector3(),
    localElbow,
    0.17,
    0.125,
    yellow
  );
  ellipsoid(
    arm,
    `${side}-elbow-cap`,
    localElbow,
    new THREE.Vector3(0.126, 0.126, 0.112),
    yellow
  );

  const forearm = new THREE.Group();
  forearm.name = `${side}-forearm`;
  forearm.position.copy(localElbow);
  arm.add(forearm);

  cylinderBetween(
    forearm,
    `${side}-lower-arm`,
    new THREE.Vector3(),
    localWrist,
    0.122,
    0.083,
    yellow
  );
  ellipsoid(
    forearm,
    `${side}-wrist-cap`,
    localWrist,
    new THREE.Vector3(0.086, 0.09, 0.077),
    yellow
  );

  const hand = new THREE.Group();
  hand.name = `${side}-hand`;
  hand.position.copy(localWrist);
  forearm.add(hand);
  makeHand(hand, side);
}
