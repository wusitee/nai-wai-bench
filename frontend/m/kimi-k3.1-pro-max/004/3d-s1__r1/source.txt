<script type="module">
import * as THREE from 'three';

const scene = new THREE.Scene();
scene.background = new THREE.Color(0xf4f3ee);

const renderer = new THREE.WebGLRenderer({
  antialias: true,
  preserveDrawingBuffer: true
});
renderer.setPixelRatio(1);
renderer.setSize(244, 240);
renderer.outputColorSpace = THREE.SRGBColorSpace;
renderer.toneMapping = THREE.ACESFilmicToneMapping;
renderer.toneMappingExposure = 1.08;
document.body.appendChild(renderer.domElement);

const camera = new THREE.OrthographicCamera(-2.094, 2.094, 2.06, -2.06, 0.1, 100);
camera.position.set(0, 2.1, 10);
camera.lookAt(0, 2.1, 0);

scene.add(new THREE.HemisphereLight(0xfffdf4, 0xcfc3a5, 1.75));

const keyLight = new THREE.DirectionalLight(0xfff9e9, 2.15);
keyLight.position.set(-3.5, 6, 8);
scene.add(keyLight);

const fillLight = new THREE.DirectionalLight(0xffffff, 0.65);
fillLight.position.set(4, 3, 5);
scene.add(fillLight);

const backLight = new THREE.DirectionalLight(0xfff7df, 1.05);
backLight.position.set(-2, 4, -5);
scene.add(backLight);

const character = new THREE.Group();
character.name = 'character-yellow';
character.position.x = 0.16;
scene.add(character);

const smoothstep = (a, b, x) => {
  const v = THREE.MathUtils.clamp((x - a) / (b - a), 0, 1);
  return v * v * (3 - 2 * v);
};

const profile = [
  [0.00, -0.52],
  [0.85, -0.46],
  [1.25, -0.25],
  [1.43,  0.00],
  [1.51,  0.38],
  [1.50,  0.78],
  [1.43,  1.15],
  [1.31,  1.52],
  [1.16,  1.86],
  [1.01,  2.17],
  [0.92,  2.46],
  [0.88,  2.75],
  [0.83,  3.03],
  [0.77,  3.28],
  [0.69,  3.50],
  [0.60,  3.70],
  [0.45,  3.88],
  [0.28,  4.01],
  [0.00,  4.09]
];

const profileCurve = new THREE.CatmullRomCurve3(
  profile.map(([r, y]) => new THREE.Vector3(r, y, 0)),
  false,
  'centripetal'
);
const profileSamples = profileCurve.getPoints(180);
const lathePoints = profileSamples.map(p => new THREE.Vector2(Math.max(0, p.x), p.y));
const bodyGeometry = new THREE.LatheGeometry(lathePoints, 112);

function radiusAtY(y) {
  if (y <= profileSamples[0].y) return Math.max(0.001, profileSamples[0].x);
  for (let i = 1; i < profileSamples.length; i++) {
    const b = profileSamples[i];
    if (b.y >= y) {
      const a = profileSamples[i - 1];
      const f = (y - a.y) / Math.max(0.00001, b.y - a.y);
      return Math.max(0.001, THREE.MathUtils.lerp(a.x, b.x, f));
    }
  }
  return 0.001;
}

const depthScale = 0.72;
function frontZ(x, y) {
  const r = radiusAtY(y);
  return depthScale * Math.sqrt(Math.max(0, r * r - x * x));
}

const yellowLow = new THREE.Color(0xf0cd3e);
const yellowHigh = new THREE.Color(0xf8dc4d);
const cream = new THREE.Color(0xf3ebc9);
const bodyPositions = bodyGeometry.attributes.position;
const bodyColors = [];
const vertexColor = new THREE.Color();

for (let i = 0; i < bodyPositions.count; i++) {
  const x = bodyPositions.getX(i);
  const y = bodyPositions.getY(i);
  const z = bodyPositions.getZ(i);
  const r = radiusAtY(y);

  const headMix = smoothstep(1.5, 3.75, y) * 0.65;
  const ellipseDistance = Math.sqrt(
    Math.pow((x - 0.025) / 1.19, 2) +
    Math.pow((y - 0.68) / 1.80, 2)
  );
  const bellyShape = 1 - smoothstep(0.78, 1.08, ellipseDistance);
  const frontFacing = smoothstep(-0.12, 0.38, z / r);
  const bellyMix = bellyShape * frontFacing;

  vertexColor.copy(yellowLow).lerp(yellowHigh, headMix).lerp(cream, bellyMix);
  bodyColors.push(vertexColor.r, vertexColor.g, vertexColor.b);
}

bodyGeometry.setAttribute('color', new THREE.Float32BufferAttribute(bodyColors, 3));

const bodyMaterial = new THREE.MeshStandardMaterial({
  vertexColors: true,
  roughness: 0.9,
  metalness: 0
});
const body = new THREE.Mesh(bodyGeometry, bodyMaterial);
body.name = 'body';
body.scale.z = depthScale;
character.add(body);

const head = new THREE.Group();
head.name = 'head';
character.add(head);

const eyeMaterial = new THREE.MeshStandardMaterial({
  color: 0x191b13,
  roughness: 0.72
});
const eyeGeometry = new THREE.SphereGeometry(1, 24, 16);

function makeEye(name, x, y, scaleX, scaleY) {
  const eyeGroup = new THREE.Group();
  eyeGroup.name = name;
  eyeGroup.position.set(x, y, frontZ(x, y) + 0.012);

  const eye = new THREE.Mesh(eyeGeometry, eyeMaterial);
  eye.name = name + '-surface';
  eye.scale.set(scaleX, scaleY, 0.047);
  eyeGroup.add(eye);
  head.add(eyeGroup);
  return eyeGroup;
}

makeEye('eye-left', -0.43, 3.67, 0.058, 0.074);
makeEye('eye-right', 0.43, 3.67, 0.061, 0.076);

const mouth = new THREE.Group();
mouth.name = 'mouth';
head.add(mouth);

const smilePoints = [
  [-0.155, 3.485],
  [-0.085, 3.458],
  [0.000, 3.447],
  [0.080, 3.460],
  [0.135, 3.490]
].map(([x, y]) => new THREE.Vector3(x, y, frontZ(x, y) + 0.014));

const smileCurve = new THREE.CatmullRomCurve3(smilePoints, false, 'centripetal');
const smile = new THREE.Mesh(
  new THREE.TubeGeometry(smileCurve, 32, 0.013, 7, false),
  new THREE.MeshStandardMaterial({
    color: 0x806c30,
    roughness: 0.9
  })
);
smile.name = 'smile-curve';
mouth.add(smile);

const armMaterial = new THREE.MeshStandardMaterial({
  color: 0xf2d143,
  roughness: 0.9
});
const handMaterial = new THREE.MeshStandardMaterial({
  color: 0x594029,
  roughness: 0.95
});
const armCapGeometry = new THREE.SphereGeometry(1, 24, 16);
const handGeometry = new THREE.SphereGeometry(1, 28, 20);

function makeArm(side, points, handPosition, handScale, handRotation) {
  const arm = new THREE.Group();
  arm.name = 'arm-' + side;
  character.add(arm);

  const curve = new THREE.CatmullRomCurve3(
    points.map(p => new THREE.Vector3(...p)),
    false,
    'centripetal'
  );
  const tube = new THREE.Mesh(
    new THREE.TubeGeometry(curve, 48, 0.265, 16, false),
    armMaterial
  );
  tube.name = 'arm-' + side + '-volume';
  arm.add(tube);

  for (const [label, point] of [['shoulder', points[0]], ['wrist', points[points.length - 1]]]) {
    const cap = new THREE.Mesh(armCapGeometry, armMaterial);
    cap.name = 'arm-' + side + '-' + label;
    cap.position.set(...point);
    cap.scale.setScalar(0.265);
    arm.add(cap);
  }

  const hand = new THREE.Group();
  hand.name = 'hand-' + side;
  hand.position.set(...handPosition);
  hand.rotation.z = handRotation;
  arm.add(hand);

  const palm = new THREE.Mesh(handGeometry, handMaterial);
  palm.name = 'hand-' + side + '-palm';
  palm.scale.set(...handScale);
  hand.add(palm);

  return arm;
}

makeArm(
  'left',
  [
    [-0.81, 2.28, -0.07],
    [-1.12, 1.93, -0.04],
    [-1.39, 2.09, 0.00],
    [-1.39, 2.57, 0.04]
  ],
  [-1.36, 2.66, 0.075],
  [0.205, 0.295, 0.175],
  0.36
);

makeArm(
  'right',
  [
    [0.82, 2.28, -0.07],
    [1.20, 1.94, -0.04],
    [1.50, 2.16, 0.00],
    [1.43, 2.60, 0.035],
    [1.23, 2.87, 0.05]
  ],
  [1.20, 3.00, 0.08],
  [0.195, 0.300, 0.170],
  0.20
);

function render(t) {
  renderer.render(scene, camera);
}

window.nwb = {
  scene,
  renderer,
  cameras: { ref: camera },
  characters: [character],
  render,
  ready: true
};

render(0);
</script>
