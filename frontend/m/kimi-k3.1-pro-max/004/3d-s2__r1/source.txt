const smoothstep = (a, b, x) => {
  const v = THREE.MathUtils.clamp((x - a) / (b - a), 0, 1);
  return v * v * (3 - 2 * v);
};

const profileCurve = new THREE.CatmullRomCurve3(
  profile.map(([r, y]) => new THREE.Vector3(r, y, 0)),
  false,
  'centripetal'
);
const profileSamples = profileCurve.getPoints(180);
const lathePoints = profileSamples.map(
  p => new THREE.Vector2(Math.max(0, p.x), p.y)
);
const bodyGeometry = new THREE.LatheGeometry(lathePoints, 112);

function radiusAtY(y) {
  if (y <= profileSamples[0].y) {
    return Math.max(0.001, profileSamples[0].x);
  }
  for (let i = 1; i < profileSamples.length; i++) {
    const b = profileSamples[i];
    if (b.y >= y) {
      const a = profileSamples[i - 1];
      const f = (y - a.y) / Math.max(0.00001, b.y - a.y);
      return Math.max(0.001, THREE.MathUtils.lerp(a.x, b.x, f));
    }
  }
  return 0.001;
}

const depthScale = 0.72;

function frontZ(x, y) {
  const r = radiusAtY(y);
  return depthScale * Math.sqrt(Math.max(0, r * r - x * x));
}

const yellowLow = new THREE.Color(0xf0cd3e);
const yellowHigh = new THREE.Color(0xf8dc4d);
const cream = new THREE.Color(0xf3ebc9);
const bodyPositions = bodyGeometry.attributes.position;
const bodyColors = [];
const vertexColor = new THREE.Color();

for (let i = 0; i < bodyPositions.count; i++) {
  const x = bodyPositions.getX(i);
  const y = bodyPositions.getY(i);
  const z = bodyPositions.getZ(i);
  const r = radiusAtY(y);

  const headMix = smoothstep(1.5, 3.75, y) * 0.65;
  const ellipseDistance = Math.sqrt(
    Math.pow((x - 0.015) / 1.19, 2) +
    Math.pow((y - 1.42) / 1.22, 2)
  );
  const bellyShape = 1 - smoothstep(0.78, 1.08, ellipseDistance);
  const frontFacing = smoothstep(-0.12, 0.38, z / r);
  const bellyMix = bellyShape * frontFacing;

  vertexColor.copy(yellowLow).lerp(yellowHigh, headMix).lerp(cream, bellyMix);
  bodyColors.push(vertexColor.r, vertexColor.g, vertexColor.b);
}

function makeArm(side) {
  const sign = side === 'left' ? -1 : 1;
  const group = new THREE.Group();
  group.name = 'arm-' + side;
  character.add(group);

  const shoulder = new THREE.Vector3(sign * 0.82, 2.24, -0.05);
  const shoulderCap = new THREE.Mesh(armSphereGeometry, armMaterial);
  shoulderCap.name = 'arm-' + side + '-shoulder';
  shoulderCap.position.copy(shoulder);
  shoulderCap.scale.set(0.28, 0.29, 0.25);
  group.add(shoulderCap);

  const segmentCount = 10;
  const segments = [];
  const points = [];

  for (let i = 0; i <= segmentCount; i++) {
    points.push(new THREE.Vector3());
  }

  for (let i = 0; i < segmentCount; i++) {
    const segment = new THREE.Mesh(armSphereGeometry, armMaterial);
    segment.name = 'arm-' + side + '-volume-' + i;
    group.add(segment);
    segments.push(segment);
  }

  const hand = new THREE.Group();
  hand.name = 'hand-' + side;
  group.add(hand);

  const palm = new THREE.Mesh(handGeometry, handMaterial);
  palm.name = 'hand-' + side + '-palm';
  hand.add(palm);

  return {
    side,
    sign,
    group,
    shoulder,
    segments,
    points,
    hand,
    palm,
    direction: new THREE.Vector3()
  };
}

const leftArm = makeArm('left');

function setCurvePoint(rig, u, control, handPosition, target) {
  const v = 1 - u;
  const shoulder = rig.shoulder;

  target.set(
    v * v * shoulder.x + 2 * v * u * control.x + u * u * handPosition.x,
    v * v * shoulder.y + 2 * v * u * control.y + u * u * handPosition.y,
    v * v * shoulder.z + 2 * v * u * control.z + u * u * handPosition.z
  );

  const surface = frontZ(target.x, target.y);
  if (surface > 0) {
    target.z = Math.max(target.z, surface + 0.14);
  }
}

const controlPoint = new THREE.Vector3();
const handPosition = new THREE.Vector3();

function updateArm(rig, x, y, angle, handStretch) {
  const shoulder = rig.shoulder;
  const z = frontZ(x, y) + 0.17;
  handPosition.set(x, y, z);

  controlPoint.set(
    rig.sign < 0
      ? Math.min(shoulder.x, x) - 0.18
      : Math.max(shoulder.x, x) + 0.18,
    Math.min(shoulder.y, y) - (rig.sign < 0 ? 0.60 : 0.55),
    Math.max(0.20, (shoulder.z + z) * 0.5 + 0.08)
  );

  for (let i = 0; i < rig.points.length; i++) {
    setCurvePoint(
      rig,
      i / (rig.points.length - 1),
      controlPoint,
      handPosition,
      rig.points[i]
    );
  }

  for (let i = 0; i < rig.segments.length; i++) {
    const a = rig.points[i];
    const b = rig.points[i + 1];
    const segment = rig.segments[i];
    const direction = rig.direction.subVectors(b, a);
    const length = Math.max(0.0001, direction.length());
    const radius = THREE.MathUtils.lerp(
      0.265,
      0.22,
      (i + 0.5) / rig.segments.length
    );

    segment.position.copy(a).add(b).multiplyScalar(0.5);
    segment.quaternion.setFromUnitVectors(
      upVector,
      direction.multiplyScalar(1 / length)
    );
    segment.scale.set(radius, length * 0.5 + radius * 0.42, radius * 0.92);
  }

  rig.hand.position.copy(handPosition);
  rig.hand.rotation.set(0, 0, angle);

  if (rig.sign < 0) {
    rig.palm.scale.set(0.205, 0.295 * handStretch, 0.175);
  } else {
    rig.palm.scale.set(0.215, 0.305 * handStretch, 0.18);
  }
}

const track = [
  { ...keys[7], t: -20 },

const poseFields = [
  'lx', 'ly', 'la', 'ls',
  'rx', 'ry', 'ra', 'rs',
  'lean', 'turn', 'bounce', 'squash'
];
const pose = {};

function samplePose(milliseconds) {
  const t = ((milliseconds % 1200) + 1200) % 1200;
  let i = 1;

  while (i < track.length - 2 && t > track[i + 1].t) {
    i++;
  }

  const a = track[i - 1];
  const b = track[i];
  const c = track[i + 1];
  const d = track[i + 2];
  const dt = c.t - b.t;
  const u = THREE.MathUtils.clamp((t - b.t) / dt, 0, 1);
  const u2 = u * u;
  const u3 = u2 * u;
  const h00 = 2 * u3 - 3 * u2 + 1;
  const h10 = u3 - 2 * u2 + u;
  const h01 = -2 * u3 + 3 * u2;
  const h11 = u3 - u2;

  for (const field of poseFields) {
    const p0 = a[field];
    const p1 = b[field];
    const p2 = c[field];
    const p3 = d[field];
    const m1 = (p2 - p0) / (c.t - a.t);
    const m2 = (p3 - p1) / (d.t - b.t);

    pose[field] =
      h00 * p1 +
      h10 * dt * m1 +
      h01 * p2 +
      h11 * dt * m2;
  }

  return pose;
}

function render(t) {
  const p = samplePose(t);

  character.position.set(0, p.bounce, 0);
  character.rotation.set(0, p.turn, p.lean);
  character.scale.set(
    1 + p.squash * 0.45,
    1 - p.squash,
    1
  );

  updateArm(leftArm, p.lx, p.ly, p.la, p.ls);
  updateArm(rightArm, p.rx, p.ry, p.ra, p.rs);

  renderer.render(scene, camera);
}
