function frontZ(x, y) {
  for (let i = 0; i < bodySamples.length - 1; i++) {
    const a = bodySamples[i];
    const b = bodySamples[i + 1];
    if (y <= a.y && y >= b.y) {
      const t = (a.y - y) / Math.max(0.00001, a.y - b.y);
      const rx = THREE.MathUtils.lerp(Math.max(0.001, a.x), Math.max(0.001, b.x), t);
      const rz = THREE.MathUtils.lerp(Math.max(0.001, a.z), Math.max(0.001, b.z), t);
      const q = Math.min(0.995, Math.abs(x) / rx);
      return rz * Math.sqrt(Math.max(0.005, 1 - q * q));
    }
  }
  return 0.08;
}

function createBody() {
  const rings = 170;
  const sides = 96;
  const positions = [];
  const indices = [];

  for (let i = 0; i <= rings; i++) {
    const p = bodyProfile.getPoint(i / rings);
    const rx = Math.max(0.001, p.x);
    const rz = Math.max(0.001, p.z);
    for (let j = 0; j < sides; j++) {
      const a = j / sides * Math.PI * 2;
      positions.push(rx * Math.cos(a), p.y, rz * Math.sin(a));
    }
  }

  for (let i = 0; i < rings; i++) {
    for (let j = 0; j < sides; j++) {
      const n = (j + 1) % sides;
      const a = i * sides + j;
      const b = (i + 1) * sides + j;
      const c = (i + 1) * sides + n;
      const d = i * sides + n;
      indices.push(a, d, b, b, d, c);
    }
  }

  const top = positions.length / 3;
  positions.push(0, profilePoints[0].y, 0);
  const bottom = positions.length / 3;
  positions.push(0, profilePoints[profilePoints.length - 1].y, 0);

  for (let j = 0; j < sides; j++) {
    const n = (j + 1) % sides;
    indices.push(top, n, j);
    const a = rings * sides + j;
    const b = rings * sides + n;
    indices.push(bottom, a, b);
  }

  const geometry = new THREE.BufferGeometry();
  geometry.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
  geometry.setIndex(indices);
  geometry.computeVertexNormals();

  const mesh = new THREE.Mesh(geometry, skin);
  mesh.name = 'continuous-head-and-torso';
  mesh.castShadow = true;
  mesh.receiveShadow = true;
  return mesh;
}

function shapeFromCommands(commands) {
  const shape = new THREE.Shape();
  for (const command of commands) {
    if (command[0] === 'M') {
      const p = P(command[1], command[2]);
      shape.moveTo(p.x, p.y);
    } else if (command[0] === 'L') {
      const p = P(command[1], command[2]);
      shape.lineTo(p.x, p.y);
    } else if (command[0] === 'C') {
      const a = P(command[1], command[2]);
      const b = P(command[3], command[4]);
      const c = P(command[5], command[6]);
      shape.bezierCurveTo(a.x, a.y, b.x, b.y, c.x, c.y);
    } else if (command[0] === 'Q') {
      const a = P(command[1], command[2]);
      const b = P(command[3], command[4]);
      shape.quadraticCurveTo(a.x, a.y, b.x, b.y);
    }
  }
  shape.closePath();
  return shape;
}

function makePatch(shape, material, offset, name) {
  let boundary = shape.getPoints(18);
  if (boundary.length > 1 && boundary[0].distanceTo(boundary[boundary.length - 1]) < 0.00001) {
    boundary.pop();
  }
  boundary = boundary.filter((p, i) => i === 0 || p.distanceTo(boundary[i - 1]) > 0.00001);

  let area = 0;
  for (let i = 0; i < boundary.length; i++) {
    const a = boundary[i];
    const b = boundary[(i + 1) % boundary.length];
    area += a.x * b.y - b.x * a.y;
  }
  if (area < 0) boundary.reverse();

  const center = new THREE.Vector2();
  for (const p of boundary) center.add(p);
  center.multiplyScalar(1 / boundary.length);

  const count = boundary.length;
  const rings = 16;
  const positions = [center.x, center.y, frontZ(center.x, center.y) + offset];
  const indices = [];

  for (let r = 1; r <= rings; r++) {
    const f = r / rings;
    for (const p of boundary) {
      const x = THREE.MathUtils.lerp(center.x, p.x, f);
      const y = THREE.MathUtils.lerp(center.y, p.y, f);
      positions.push(x, y, frontZ(x, y) + offset);
    }
  }

  for (let j = 0; j < count; j++) {
    indices.push(0, 1 + j, 1 + (j + 1) % count);
  }

  for (let r = 1; r < rings; r++) {
    const inner = 1 + (r - 1) * count;
    const outer = 1 + r * count;
    for (let j = 0; j < count; j++) {
      const n = (j + 1) % count;
      indices.push(inner + j, outer + j, outer + n);
      indices.push(inner + j, outer + n, inner + n);
    }
  }

  const geometry = new THREE.BufferGeometry();
  geometry.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
  geometry.setIndex(indices);
  geometry.computeVertexNormals();

  const mesh = new THREE.Mesh(geometry, material);
  mesh.name = name;
  mesh.receiveShadow = true;
  return mesh;
}

function makeLimb(data, material, name, segments = 64, sides = 24) {
  const centers = data.map(d => P(d[0], d[1], d[2]));
  const radii = data.map(d => new THREE.Vector3(d[3] / 40, d[4] / 40, 0));
  const centerCurve = new THREE.CatmullRomCurve3(centers, false, 'centripetal');
  const radiusCurve = new THREE.CatmullRomCurve3(radii, false, 'centripetal');
  const positions = [];
  const indices = [];

  for (let i = 0; i <= segments; i++) {
    const t = i / segments;
    const c = centerCurve.getPoint(t);
    const tangent = centerCurve.getTangent(t).normalize();
    const normal = new THREE.Vector3(tangent.y, -tangent.x, 0);
    if (normal.lengthSq() < 0.00001) normal.set(1, 0, 0);
    normal.normalize();
    const depthAxis = new THREE.Vector3().crossVectors(normal, tangent).normalize();
    if (depthAxis.z < 0) depthAxis.multiplyScalar(-1);
    const r = radiusCurve.getPoint(t);

    for (let j = 0; j < sides; j++) {
      const a = j / sides * Math.PI * 2;
      const p = c.clone()
        .addScaledVector(normal, Math.cos(a) * Math.max(0.006, r.x))
        .addScaledVector(depthAxis, Math.sin(a) * Math.max(0.006, r.y));
      positions.push(p.x, p.y, p.z);
    }
  }

  for (let i = 0; i < segments; i++) {
    for (let j = 0; j < sides; j++) {
      const n = (j + 1) % sides;
      const a = i * sides + j;
      const b = (i + 1) * sides + j;
      const c = (i + 1) * sides + n;
      const d = i * sides + n;
      indices.push(a, b, d, b, c, d);
    }
  }

  const startCenter = positions.length / 3;
  const start = centerCurve.getPoint(0);
  positions.push(start.x, start.y, start.z);
  const endCenter = positions.length / 3;
  const end = centerCurve.getPoint(1);
  positions.push(end.x, end.y, end.z);

  for (let j = 0; j < sides; j++) {
    const n = (j + 1) % sides;
    indices.push(startCenter, j, n);
    const a = segments * sides + j;
    const b = segments * sides + n;
    indices.push(endCenter, b, a);
  }

  const geometry = new THREE.BufferGeometry();
  geometry.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
  geometry.setIndex(indices);
  geometry.computeVertexNormals();

  const mesh = new THREE.Mesh(geometry, material);
  mesh.name = name;
  mesh.castShadow = true;
  mesh.receiveShadow = true;
  return mesh;
}

  mesh.position.copy(position);

  mesh.rotation.z = rotationZ;
  mesh.castShadow = true;
  mesh.receiveShadow = true;
  parent.add(mesh);
  return mesh;
}

function makeEye(name, start, control, end) {
  const curve = new THREE.QuadraticBezierCurve3(
    P(start[0], start[1]),
    P(control[0], control[1]),
    P(end[0], end[1])
  );
  const points = curve.getPoints(28).map(p => {
    p.z = frontZ(p.x, p.y) + 0.052;
    return p;
  });
  const path = new THREE.CatmullRomCurve3(points);
  const geometry = new THREE.TubeGeometry(path, 48, 0.041, 10, false);
  const mesh = new THREE.Mesh(geometry, eyeMaterial);
  mesh.name = name;
  mesh.castShadow = true;
  eyes.add(mesh);
  ellipsoid(eyes, name + '-start', points[0], [0.041, 0.041, 0.041], eyeMaterial);
  ellipsoid(eyes, name + '-end', points[points.length - 1], [0.041, 0.041, 0.041], eyeMaterial);
}
