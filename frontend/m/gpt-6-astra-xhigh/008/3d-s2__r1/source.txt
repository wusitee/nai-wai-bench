let randomState=731927;
function rnd(){
  randomState^=randomState<<13;
  randomState^=randomState>>>17;
  randomState^=randomState<<5;
  return (randomState>>>0)/4294967296;
}
const noiseData=new Uint8Array(64*64*4);

  const n=95+Math.floor(rnd()*140);
  noiseData[i*4]=noiseData[i*4+1]=noiseData[i*4+2]=n;
  noiseData[i*4+3]=255;
}

// Fur samples remain attached to the same triangles as the limbs bend.
function pelt(surface,count,color,length=.019,opacity=.28){
  const geometry=surface.geometry;
  const index=geometry.index;
  const triCount=(index?index.count:geometry.attributes.position.count)/3;
  const ids=new Uint32Array(count*3);
  const weights=new Float32Array(count*3);
  const lengths=new Float32Array(count);
  const jitter=new Float32Array(count*3);
  const colors=new Float32Array(count*6);
  const tint=new THREE.Color(color);
  for(let k=0;k<count;k++){
    const tri=Math.floor(rnd()*triCount);
    for(let j=0;j<3;j++)ids[k*3+j]=index?index.getX(tri*3+j):tri*3+j;
    const r=Math.sqrt(rnd()),s=rnd();
    weights[k*3]=1-r;
    weights[k*3+1]=r*(1-s);
    weights[k*3+2]=r*s;
    lengths[k]=length*(.35+.9*rnd());
    jitter[k*3]=(rnd()-.5)*.009;
    jitter[k*3+1]=(rnd()-.5)*.009;
    jitter[k*3+2]=0;
    const shade=.83+rnd()*.3;
    for(let j=0;j<2;j++){
      colors[k*6+j*3]=tint.r*shade;
      colors[k*6+j*3+1]=tint.g*shade;
      colors[k*6+j*3+2]=tint.b*shade;
    }
  }
  const g=new THREE.BufferGeometry();
  g.setAttribute('position',new THREE.BufferAttribute(new Float32Array(count*6),3));
  g.setAttribute('color',new THREE.BufferAttribute(colors,3));
  const strands=new THREE.LineSegments(g,new THREE.LineBasicMaterial({
    vertexColors:true,transparent:true,opacity,depthWrite:false
  }));
  strands.name='fine-fur';
  strands.frustumCulled=false;
  surface.add(strands);
  surface.userData.fur={geometry:g,ids,weights,lengths,jitter,count};
  updateFur(surface);
}
function updateFur(surface){
  const f=surface.userData.fur;
  if(!f)return;
  const p=surface.geometry.attributes.position.array;
  const n=surface.geometry.attributes.normal.array;
  const out=f.geometry.attributes.position.array;
  for(let k=0;k<f.count;k++){
    const b=k*3;
    const ia=f.ids[b]*3,ib=f.ids[b+1]*3,ic=f.ids[b+2]*3;
    const wa=f.weights[b],wb=f.weights[b+1],wc=f.weights[b+2];
    const x=p[ia]*wa+p[ib]*wb+p[ic]*wc;
    const y=p[ia+1]*wa+p[ib+1]*wb+p[ic+1]*wc;
    const z=p[ia+2]*wa+p[ib+2]*wb+p[ic+2]*wc;
    let nx=n[ia]*wa+n[ib]*wb+n[ic]*wc;
    let ny=n[ia+1]*wa+n[ib+1]*wb+n[ic+1]*wc;
    let nz=n[ia+2]*wa+n[ib+2]*wb+n[ic+2]*wc;
    const inv=1/(Math.hypot(nx,ny,nz)||1);
    nx*=inv;ny*=inv;nz*=inv;
    const j=k*6,l=f.lengths[k];
    out[j]=x;out[j+1]=y;out[j+2]=z;
    out[j+3]=x+nx*l+f.jitter[b];
    out[j+4]=y+ny*l+f.jitter[b+1];
    out[j+5]=z+nz*l+f.jitter[b+2];
  }
  f.geometry.attributes.position.needsUpdate=true;
}

function organicGeometry(points,radii,segments=60,sides=24){
  const g=new THREE.BufferGeometry();
  const vertices=(segments+1)*(sides+1)+2;
  g.setAttribute('position',new THREE.BufferAttribute(new Float32Array(vertices*3),3));
  const uv=new Float32Array(vertices*2),indices=[];
  for(let i=0;i<=segments;i++){
    for(let j=0;j<=sides;j++){
      const a=i*(sides+1)+j;
      uv[a*2]=j/sides;
      uv[a*2+1]=i/segments;
      if(i<segments&&j<sides){
        const b=a+sides+1;
        indices.push(a,b,a+1,b,b+1,a+1);
      }
    }
  }
  const cap=(segments+1)*(sides+1);
  for(let e=0;e<2;e++){
    const end=e===0?0:segments;
    uv[(cap+e)*2]=uv[(cap+e)*2+1]=.5;
    for(let j=0;j<sides;j++){
      const a=end*(sides+1)+j;
      if(e===0)indices.push(cap+e,a,a+1);
      else indices.push(cap+e,a+1,a);
    }
  }
  g.setAttribute('uv',new THREE.BufferAttribute(uv,2));
  g.setIndex(indices);
  g.userData.organic={segments,sides,radii:[...radii]};
  updateOrganic(g,points);
  return g;
}
function updateOrganic(g,points){
  const {segments,sides,radii}=g.userData.organic;
  const curve=new THREE.CatmullRomCurve3(points.map(p=>new THREE.Vector3(...p)));
  const frames=curve.computeFrenetFrames(segments,false);
  const positions=g.attributes.position.array;
  let at=0;
  for(let i=0;i<=segments;i++){
    const t=i/segments,center=curve.getPointAt(t);
    const f=t*(radii.length-1);
    const n=Math.min(radii.length-2,Math.floor(f));
    const q=f-n,s=q*q*(3-2*q);
    const radius=THREE.MathUtils.lerp(radii[n],radii[n+1],s);
    const normal=frames.normals[i],binormal=frames.binormals[i];
    for(let j=0;j<=sides;j++){
      const theta=j/sides*Math.PI*2;
      const a=Math.cos(theta)*radius,b=Math.sin(theta)*radius;
      positions[at++]=center.x+normal.x*a+binormal.x*b;
      positions[at++]=center.y+normal.y*a+binormal.y*b;
      positions[at++]=center.z+normal.z*a+binormal.z*b;
    }
  }
  for(const t of [0,1]){
    const c=curve.getPointAt(t);
    positions[at++]=c.x;positions[at++]=c.y;positions[at++]=c.z;
  }
  g.attributes.position.needsUpdate=true;
  g.computeVertexNormals();
}
function tube(parent,name,points,radius,mat,closed=false){

    points.map(p=>p.isVector3?p:new THREE.Vector3(...p)),closed
  );
  return mesh(new THREE.TubeGeometry(path,Math.max(32,points.length*3),radius,10,closed),
    mat,parent,name);
}

function profile(y){
  let i=0;
  while(i<bodyProfile.length-2&&y>bodyProfile[i+1][0])i++;
  const a=bodyProfile[i],b=bodyProfile[i+1];
  const t=THREE.MathUtils.clamp((y-a[0])/(b[0]-a[0]),0,1);
  const s=t*t*(3-2*t);
  return {
    rx:THREE.MathUtils.lerp(a[1],b[1],s),
    rz:THREE.MathUtils.lerp(a[2],b[2],s),
    x:THREE.MathUtils.lerp(a[3],b[3],s)
  };
}
function bodyFront(x,y){
  const p=profile(y);
  return p.rz*Math.sqrt(Math.max(.001,1-((x-p.x)/p.rx)**2));
}

function headRadius(y){
  const n=THREE.MathUtils.clamp(((y-.04)/.66+1)*.5,0,1);
  return .49*(.87+.13*n);
}
function headZ(x,y){
  return .435*Math.sqrt(Math.max(0,1-(x/headRadius(y))**2-((y-.04)/.66)**2));
}
function mouthBoundary(angle,scale=1){
  const s=Math.sin(angle);
  return new THREE.Vector3(
    -.018+scale*.319*Math.cos(angle)*(1+.19*s),
    -.095+scale*.405*s,0
  );
}
function headSurface(front){
  const positions=[],normals=[],uvs=[],indices=[];
  const rings=44,sides=144;
  for(let i=0;i<=rings;i++){
    const r=i/rings;
    for(let j=0;j<=sides;j++){
      const a=j/sides*Math.PI*2;
      const outerY=.04+.66*Math.sin(a);
      const outerX=headRadius(outerY)*Math.cos(a);
      let x,y,z;
      if(front){
        const inner=mouthBoundary(a);
        x=THREE.MathUtils.lerp(inner.x,outerX,r);
        y=THREE.MathUtils.lerp(inner.y,outerY,r);
        z=headZ(x,y);
      }else{
        x=outerX*r;
        y=.04+(outerY-.04)*r;
        z=-headZ(x,y);
      }
      positions.push(x,y,z);
      const n=new THREE.Vector3(
        x/(.475*.475),(y-.04)/(.66*.66),z/(.435*.435)
      ).normalize();
      normals.push(n.x,n.y,n.z);
      uvs.push(j/sides,front?r*.5:.5+r*.5);
      if(i<rings&&j<sides){
        const a0=i*(sides+1)+j,b=a0+sides+1;
        if(front)indices.push(a0,b,a0+1,a0+1,b,b+1);
        else indices.push(a0,a0+1,b,a0+1,b+1,b);
      }
    }
  }
  const g=new THREE.BufferGeometry();
  g.setAttribute('position',new THREE.Float32BufferAttribute(positions,3));
  g.setAttribute('normal',new THREE.Float32BufferAttribute(normals,3));
  g.setAttribute('uv',new THREE.Float32BufferAttribute(uvs,2));
  g.setIndex(indices);
  g.userData.restPosition=g.attributes.position.array.slice();
  g.userData.restNormal=g.attributes.normal.array.slice();
  return g;
}

const scalarFields=['dx','dy','bend','lean','twist','stretch','width','squat','jaw'];
const vectorFields=['h','r','le','lw','re','rw'];

function samplePose(t){
  const ms=Number.isFinite(t)?((t%1220)+1220)%1220:0;
  let i=0;
  while(i<poses.length-2&&ms>poses[i+1].t)i++;
  const a=poses[i],b=poses[i+1];
  let u=(ms-a.t)/(b.t-a.t);
  u=u*u*(3-2*u);
  const p={};
  for(const k of scalarFields)p[k]=a[k]+(b[k]-a[k])*u;
  for(const k of vectorFields)p[k]=a[k].map((v,j)=>v+(b[k][j]-v)*u);
  return p;
}
function torsoPoint(p){
  return new THREE.Vector3(p[0]-.4,p[1]-.85,p[2])
    .multiply(hips.scale).applyQuaternion(hips.quaternion).add(hips.position);
}
function poseArm(arm,elbow,wrist,squat){
  const shoulder=torsoPoint(arm.base[0]);
  const e=new THREE.Vector3(...elbow),w=new THREE.Vector3(...wrist);
  const rest=arm.base;
  const bias1=new THREE.Vector3(
    rest[1][0]-(rest[0][0]+rest[2][0])*.5,
    rest[1][1]-(rest[0][1]+rest[2][1])*.5,
    rest[1][2]-(rest[0][2]+rest[2][2])*.5
  );
  const bias3=new THREE.Vector3(
    rest[3][0]-(rest[2][0]+rest[4][0])*.5,
    rest[3][1]-(rest[2][1]+rest[4][1])*.5,
    rest[3][2]-(rest[2][2]+rest[4][2])*.5
  );
  const upper=shoulder.clone().lerp(e,.5).addScaledVector(bias1,1-.72*squat);
  const lower=e.clone().lerp(w,.5).addScaledVector(bias3,1-.45*squat);
  const points=[shoulder,upper,e,lower,w].map(v=>v.clone().sub(shoulder).toArray());
  arm.group.position.copy(shoulder);
  updateOrganic(arm.volume.geometry,points);
  updateFur(arm.volume);
  const last=points[4];
  arm.hand.position.set(last[0]-.025,last[1]-.035,last[2]+.035);
  arm.hand.rotation.set(-.22*squat,0,
    arm.side==='left'?-.09-.12*squat:.10-.20*squat);
}
function poseLeg(leg,side,p){
  const root=torsoPoint(leg.base[0]);
  const isLeft=side==='left';
  const s=p.squat,base=leg.base;
  const knee=[
    THREE.MathUtils.lerp(base[1][0],isLeft?-.42:1.04,s)+p.dx*.35,
    THREE.MathUtils.lerp(base[1][1],isLeft?.32:.30,s),
    THREE.MathUtils.lerp(base[1][2],isLeft?.55:.57,s)
  ];
  const shin=[
    THREE.MathUtils.lerp(base[2][0],isLeft?-.41:1.02,s),
    base[2][1],
    THREE.MathUtils.lerp(base[2][2],.29,s)
  ];
  const end=[...base[3]];
  const world=[root.toArray(),knee,shin,end];
  const local=world.map(v=>[v[0]-root.x,v[1]-root.y,v[2]-root.z]);
  leg.group.position.copy(root);
  updateOrganic(leg.volume.geometry,local);
  updateFur(leg.volume);
  leg.foot.position.set(...local[3]);
}
function deformJaw(surface,opening){
  const g=surface.geometry;
  const rest=g.userData.restPosition,norm=g.userData.restNormal;
  const positions=g.attributes.position.array,normals=g.attributes.normal.array;
  for(let i=0;i<rest.length;i+=3){
    const lower=rest[i+1]<.31;
    positions[i]=rest[i];
    positions[i+1]=lower?.31+(rest[i+1]-.31)*opening:rest[i+1];
    positions[i+2]=rest[i+2];
    let nx=norm[i],ny=norm[i+1],nz=norm[i+2];
    if(lower)ny/=opening;
    const inv=1/(Math.hypot(nx,ny,nz)||1);
    normals[i]=nx*inv;normals[i+1]=ny*inv;normals[i+2]=nz*inv;
  }
  g.attributes.position.needsUpdate=true;
  g.attributes.normal.needsUpdate=true;
  updateFur(surface);
}

  softShadow.scale.set(1+.06*p.squat,1+.12*p.squat,1);
  renderer.render(scene,ref);
}
