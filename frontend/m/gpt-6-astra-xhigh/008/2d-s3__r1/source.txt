    const input = Number(t);

    let frame = 0;
    while (frame < times.length - 1 && time >= times[frame+1]) frame++;
    const next = (frame+1) % times.length;
    const endTime = next === 0 ? period : times[next];
    const duration = endTime-times[frame];
    const u = (time-times[frame])/duration;
    const u2 = u*u;
    const u3 = u2*u;

    function sample(rows,component) {
      const a = rows[frame][component];
      const b = rows[next][component];
      if (frame === 7) return a+(b-a)*(3*u2-2*u3);
      const ma = frame === 0 ? 0 :

      const mb = next === 7 ? 0 :
