let randomState=731927;
function rnd(){
  randomState^=randomState<<13;
  randomState^=randomState>>>17;
  randomState^=randomState<<5;
  return (randomState>>>0)/4294967296;
}
const noiseData=new Uint8Array(64*64*4);

  const n=95+Math.floor(rnd()*140);
  noiseData[i*4]=n;
  noiseData[i*4+1]=n;
  noiseData[i*4+2]=n;
  noiseData[i*4+3]=255;
}

function pelt(surface,count,color,length=.019,opacity=.28){
  const geometry=surface.geometry;
  const pos=geometry.attributes.position;
  const norms=geometry.attributes.normal;
  const indices=geometry.index;
  const triCount=(indices?indices.count:pos.count)/3;
  const vertices=[],colors=[];
  const tint=new THREE.Color(color);
  const a=new THREE.Vector3(),b=new THREE.Vector3(),c=new THREE.Vector3();
  const na=new THREE.Vector3(),nb=new THREE.Vector3(),nc=new THREE.Vector3();
  const pt=new THREE.Vector3(),normal=new THREE.Vector3();
  for(let k=0;k<count;k++){
    const triangle=Math.floor(rnd()*triCount);
    const ia=indices?indices.getX(triangle*3):triangle*3;
    const ib=indices?indices.getX(triangle*3+1):triangle*3+1;
    const ic=indices?indices.getX(triangle*3+2):triangle*3+2;
    a.fromBufferAttribute(pos,ia);b.fromBufferAttribute(pos,ib);c.fromBufferAttribute(pos,ic);
    na.fromBufferAttribute(norms,ia);nb.fromBufferAttribute(norms,ib);nc.fromBufferAttribute(norms,ic);
    const r=Math.sqrt(rnd()),s=rnd();
    pt.copy(a).multiplyScalar(1-r).addScaledVector(b,r*(1-s)).addScaledVector(c,r*s);
    normal.copy(na).multiplyScalar(1-r).addScaledVector(nb,r*(1-s)).addScaledVector(nc,r*s).normalize();
    const len=length*(.35+.9*rnd());
    vertices.push(pt.x,pt.y,pt.z);
    vertices.push(
      pt.x+normal.x*len+(rnd()-.5)*.009,
      pt.y+normal.y*len+(rnd()-.5)*.009,
      pt.z+normal.z*len+(rnd()-.5)*.009
    );
    const shade=.83+rnd()*.3;
    for(let j=0;j<2;j++)colors.push(tint.r*shade,tint.g*shade,tint.b*shade);
  }
  const g=new THREE.BufferGeometry();
  g.setAttribute('position',new THREE.Float32BufferAttribute(vertices,3));
  g.setAttribute('color',new THREE.Float32BufferAttribute(colors,3));
  const strands=new THREE.LineSegments(g,new THREE.LineBasicMaterial({
    vertexColors:true,transparent:true,opacity,depthWrite:false
  }));
  strands.name='fine-fur';
  surface.add(strands);
}
function organicGeometry(points,radii,segments=60,sides=24){
  const curve=new THREE.CatmullRomCurve3(points.map(p=>new THREE.Vector3(...p)));
  const frames=curve.computeFrenetFrames(segments,false);
  const positions=[],uvs=[],indices=[];
  for(let i=0;i<=segments;i++){
    const t=i/segments;
    const center=curve.getPointAt(t);
    const f=t*(radii.length-1);
    const n=Math.min(radii.length-2,Math.floor(f));
    const q=f-n;
    const blend=q*q*(3-2*q);
    const radius=THREE.MathUtils.lerp(radii[n],radii[n+1],blend);
    for(let j=0;j<=sides;j++){
      const theta=j/sides*Math.PI*2;
      const v=center.clone()
        .addScaledVector(frames.normals[i],Math.cos(theta)*radius)
        .addScaledVector(frames.binormals[i],Math.sin(theta)*radius);
      positions.push(v.x,v.y,v.z);
      uvs.push(j/sides,t);
      if(i<segments&&j<sides){
        const a=i*(sides+1)+j,b=a+sides+1;
        indices.push(a,b,a+1,b,b+1,a+1);
      }
    }
  }
  for(const end of [0,segments]){
    const center=curve.getPointAt(end/segments);
    const c=positions.length/3;
    positions.push(center.x,center.y,center.z);
    uvs.push(.5,.5);
    for(let j=0;j<sides;j++){
      const a=end*(sides+1)+j;
      if(end===0)indices.push(c,a,a+1);
      else indices.push(c,a+1,a);
    }
  }
  const geometry=new THREE.BufferGeometry();
  geometry.setAttribute('position',new THREE.Float32BufferAttribute(positions,3));
  geometry.setAttribute('uv',new THREE.Float32BufferAttribute(uvs,2));
  geometry.setIndex(indices);
  geometry.computeVertexNormals();
  return geometry;
}
function tube(parent,name,points,radius,mat,closed=false){
  const path=new THREE.CatmullRomCurve3(points.map(p=>p.isVector3?p:new THREE.Vector3(...p)),closed);
  return mesh(new THREE.TubeGeometry(path,Math.max(32,points.length*3),radius,10,closed),mat,parent,name);
}

function profile(y){
  let i=0;
  while(i<bodyProfile.length-2&&y>bodyProfile[i+1][0])i++;
  const a=bodyProfile[i],b=bodyProfile[i+1];
  const t=THREE.MathUtils.clamp((y-a[0])/(b[0]-a[0]),0,1);
  const s=t*t*(3-2*t);
  return {
    rx:THREE.MathUtils.lerp(a[1],b[1],s),
    rz:THREE.MathUtils.lerp(a[2],b[2],s),
    x:THREE.MathUtils.lerp(a[3],b[3],s)
  };
}
function bodyFront(x,y){
  const p=profile(y);
  return p.rz*Math.sqrt(Math.max(.001,1-((x-p.x)/p.rx)**2));
}
{
  const positions=[],uvs=[],indices=[];
  const rings=90,sides=96;

      const angle=j/sides*Math.PI*2;
      positions.push(p.x+p.rx*Math.cos(angle),y,p.rz*Math.sin(angle));
      uvs.push(j/sides,i/rings);

function headRadius(y){
  const n=THREE.MathUtils.clamp(((y-.04)/.66+1)*.5,0,1);
  return .49*(.87+.13*n);
}
function headZ(x,y){
  return .435*Math.sqrt(Math.max(0,1-(x/headRadius(y))**2-((y-.04)/.66)**2));
}
function mouthBoundary(angle,scale=1){
  const s=Math.sin(angle);
  return new THREE.Vector3(
    -.018+scale*.319*Math.cos(angle)*(1+.19*s),
    -.095+scale*.405*s,
    0
  );
}
function headSurface(front){
  const positions=[],normals=[],uvs=[],indices=[];
  const rings=44,sides=144;
  for(let i=0;i<=rings;i++){
    const r=i/rings;
    for(let j=0;j<=sides;j++){
      const a=j/sides*Math.PI*2;
      const outerY=.04+.66*Math.sin(a);
      const outerX=headRadius(outerY)*Math.cos(a);
      let x,y,z;
      if(front){
        const inner=mouthBoundary(a);
        x=THREE.MathUtils.lerp(inner.x,outerX,r);
        y=THREE.MathUtils.lerp(inner.y,outerY,r);
        z=headZ(x,y);
      }else{
        x=outerX*r;
        y=.04+(outerY-.04)*r;
        z=-headZ(x,y);
      }
      positions.push(x,y,z);
      const n=new THREE.Vector3(x/(.475*.475),(y-.04)/(.66*.66),z/(.435*.435)).normalize();
      normals.push(n.x,n.y,n.z);
      uvs.push(j/sides,front?r*.5:.5+r*.5);
      if(i<rings&&j<sides){
        const a0=i*(sides+1)+j,b=a0+sides+1;
        if(front)indices.push(a0,b,a0+1,a0+1,b,b+1);
        else indices.push(a0,a0+1,b,a0+1,b+1,b);
      }
    }
  }
  const g=new THREE.BufferGeometry();
  g.setAttribute('position',new THREE.Float32BufferAttribute(positions,3));
  g.setAttribute('normal',new THREE.Float32BufferAttribute(normals,3));
  g.setAttribute('uv',new THREE.Float32BufferAttribute(uvs,2));
  g.setIndex(indices);
  return g;
}

const shadowTexture=new THREE.DataTexture(shadowPixels,64,64,THREE.RGBAFormat);
shadowTexture.needsUpdate=true;
shadowTexture.magFilter=THREE.LinearFilter;
const softShadow=new THREE.Mesh(
  new THREE.PlaneGeometry(3.0,1.65),
  new THREE.MeshBasicMaterial({
    map:shadowTexture,transparent:true,depthWrite:false,toneMapped:false
  })
);
softShadow.name='soft-contact-shadow';
softShadow.rotation.x=-Math.PI/2;
softShadow.position.set(.40,-.309,.08);
scene.add(softShadow);

function render(t){
  renderer.render(scene,ref);
}
window.nwb={scene,renderer,cameras:{ref},characters:[bear],render,ready:true};
render(0);
