function fuzz(surface,count,color,length=.021){
  const geo=surface.geometry;
  const p=geo.getAttribute('position');
  const n=geo.getAttribute('normal');
  const ix=geo.index;
  const triangles=ix?ix.count/3:p.count/3;
  const cumulative=new Float64Array(triangles);
  const a=new THREE.Vector3(),b=new THREE.Vector3(),c=new THREE.Vector3();
  const ab=new THREE.Vector3(),ac=new THREE.Vector3();
  let total=0;
  for(let j=0;j<triangles;j++){
    a.fromBufferAttribute(p,ix?ix.getX(j*3):j*3);
    b.fromBufferAttribute(p,ix?ix.getX(j*3+1):j*3+1);
    c.fromBufferAttribute(p,ix?ix.getX(j*3+2):j*3+2);
    total+=ab.subVectors(b,a).cross(ac.subVectors(c,a)).length()*.5;
    cumulative[j]=total;
  }
  const positions=[],normals=[],colors=[];
  const baseColor=new THREE.Color(color);
  const normal=new THREE.Vector3(),tmp=new THREE.Vector3(),tangent=new THREE.Vector3();
  for(let j=0;j<count;j++){
    const pick=rand()*total;
    let lo=0,hi=triangles-1;
    while(lo<hi){
      const mid=(lo+hi)>>1;
      if(cumulative[mid]<pick)lo=mid+1;else hi=mid;
    }
    const ids=[0,1,2].map(k=>ix?ix.getX(lo*3+k):lo*3+k);
    const u=Math.sqrt(rand()),v=rand();
    const weights=[1-u,u*(1-v),u*v];
    const pos=new THREE.Vector3();
    normal.set(0,0,0);
    for(let k=0;k<3;k++){
      pos.addScaledVector(tmp.fromBufferAttribute(p,ids[k]),weights[k]);
      normal.addScaledVector(tmp.fromBufferAttribute(n,ids[k]),weights[k]);
    }
    normal.normalize();
    tangent.set(rand()-.5,rand()-.5,rand()-.5).cross(normal).normalize();
    const width=.0015+rand()*.0018;
    const len=length*(.42+rand()*.85);
    const tip=pos.clone().addScaledVector(normal,len).addScaledVector(tangent,len*(rand()-.5)*.6);
    pos.addScaledVector(normal,.002);
    const vertices=[pos.clone().addScaledVector(tangent,-width),pos.clone().addScaledVector(tangent,width),tip];
    const shade=.87+rand()*.22;
    for(const q of vertices){
      positions.push(q.x,q.y,q.z);
      normals.push(normal.x,normal.y,normal.z);
      colors.push(baseColor.r*shade,baseColor.g*shade,baseColor.b*shade);
    }
  }
  const fg=new THREE.BufferGeometry();
  fg.setAttribute('position',new THREE.Float32BufferAttribute(positions,3));
  fg.setAttribute('normal',new THREE.Float32BufferAttribute(normals,3));
  fg.setAttribute('color',new THREE.Float32BufferAttribute(colors,3));
  const fm=new THREE.Mesh(fg,fuzzMat);
  fm.name='short-soft-fur';
  fm.receiveShadow=true;
  surface.add(fm);
}
function furryEllipsoid(parent,name,pos,scale,mat,count=0,color='#ffcf34'){

  return m;
}

function profileValue(rows,u,col){
  const q=u*(rows.length-1);
  const i=Math.min(rows.length-2,Math.floor(q)),t=q-i;
  const p0=rows[Math.max(0,i-1)][col],p1=rows[i][col];
  const p2=rows[i+1][col],p3=rows[Math.min(rows.length-1,i+2)][col];
  return .5*((2*p1)+(-p0+p2)*t+(2*p0-5*p1+4*p2-p3)*t*t+(-p0+3*p1-3*p2+p3)*t*t*t);
}
function torso(parent,rows){
  const pos=[],uv=[],indices=[];
  const rings=72,sides=64;
  for(let j=0;j<=rings;j++){
    const u=j/rings;
    const y=profileValue(rows,u,0);
    const rx=Math.max(.001,profileValue(rows,u,1));
    const rz=Math.max(.001,profileValue(rows,u,2));
    const cx=profileValue(rows,u,3),cz=profileValue(rows,u,4);
    for(let k=0;k<=sides;k++){
      const a=k/sides*Math.PI*2;
      pos.push(cx+Math.sin(a)*rx,y,cz+Math.cos(a)*rz);
      uv.push(k/sides,u);
    }
  }
  for(let j=0;j<rings;j++)for(let k=0;k<sides;k++){
    const a=j*(sides+1)+k,b=a+sides+1;
    indices.push(a,a+1,b,a+1,b+1,b);
  }
  const g=new THREE.BufferGeometry();
  g.setAttribute('position',new THREE.Float32BufferAttribute(pos,3));
  g.setAttribute('uv',new THREE.Float32BufferAttribute(uv,2));
  g.setIndex(indices);
  g.computeVertexNormals();
  const body=mesh(group(parent,'torso'),'continuous-pear-shaped-body',g,yellow);
  fuzz(body,7000,'#ffcc29',.020);
  return body;
}
function local(character,point){
  return V(point).sub(character.position).toArray();
}
function arm(character,name,worldPoints,radii){
  const armGroup=group(character,name);
  const curve=new THREE.CatmullRomCurve3(worldPoints.map(p=>V(local(character,p))));
  const segs=56,sides=20;
  const frames=curve.computeFrenetFrames(segs,false);
  const pos=[],uv=[],indices=[];
  for(let i=0;i<=segs;i++){
    const u=i/segs,at=curve.getPointAt(u);
    const ri=u*(radii.length-1),r0=Math.min(radii.length-2,Math.floor(ri)),f=ri-r0;
    const radius=THREE.MathUtils.lerp(radii[r0],radii[r0+1],f);
    for(let k=0;k<=sides;k++){
      const theta=k/sides*Math.PI*2;
      const q=at.clone().addScaledVector(frames.normals[i],Math.cos(theta)*radius)
        .addScaledVector(frames.binormals[i],Math.sin(theta)*radius);
      pos.push(q.x,q.y,q.z);uv.push(k/sides,u);
    }
  }
  for(let i=0;i<segs;i++)for(let k=0;k<sides;k++){
    const a=i*(sides+1)+k,b=a+sides+1;
    indices.push(a,a+1,b,a+1,b+1,b);
  }
  const geometry=new THREE.BufferGeometry();
  geometry.setAttribute('position',new THREE.Float32BufferAttribute(pos,3));
  geometry.setAttribute('uv',new THREE.Float32BufferAttribute(uv,2));
  geometry.setIndex(indices);
  geometry.computeVertexNormals();
  const m=mesh(armGroup,'soft-yellow-arm',geometry,yellow);
  fuzz(m,1600,'#ffcc2e',.018);
  ellipsoid(armGroup,'rounded-shoulder',local(character,worldPoints[0]),[radii[0],radii[0],radii[0]],yellow);
  return armGroup;
}

  return .023+.092*Math.sin(Math.PI*t)+.013*t+.012*Math.cos(z/depth*Math.PI*2);
}
function makePage(parent,name,side,width,depth,height,mat){
  const pos=[],uv=[],idx=[],nx=30,nz=18;
  for(let j=0;j<=nz;j++){
    const z=(j/nz-.5)*depth;
    for(let i=0;i<=nx;i++){
      const t=i/nx;
      pos.push(side*t*width,height+pageHeight(t,z,depth),z);
      uv.push(t,j/nz);
    }
  }
  for(let j=0;j<nz;j++)for(let i=0;i<nx;i++){
    const a=j*(nx+1)+i,b=a+nx+1;
    if(side>0)idx.push(a,b,a+1,a+1,b,b+1);
    else idx.push(a,a+1,b,a+1,b+1,b);
  }
  const g=new THREE.BufferGeometry();
  g.setAttribute('position',new THREE.Float32BufferAttribute(pos,3));
  g.setAttribute('uv',new THREE.Float32BufferAttribute(uv,2));
  g.setIndex(idx);g.computeVertexNormals();
  return mesh(parent,name,g,mat);
}
function book(name,position,angle,width,depth,coverColor,writing=false){
  const g=group(scene,name);
  g.position.set(...position);
  g.rotation.y=angle;
  box(g,'thin-book-cover',[0,-.005,0],[width*2+.08,.028,depth+.055],material(coverColor,.94));
  for(let layer=0;layer<6;layer++){
    const h=.008+layer*.008;
    const mat=layer===5?paperMat:(layer%2?paperEdgeMat:paperMat);
    makePage(g,'left-page-layer-'+layer,-1,width,depth,h,mat);
    makePage(g,'right-page-layer-'+layer,1,width,depth,h,mat);
  }
  line(g,'central-binding-crease',[
    [0,.070,-depth/2],[0,.083,0],[0,.070,depth/2]
  ],.009,material('#baa984',1));
  if(writing){
    const pencilWriting=material('#aaa18a',1);
    for(let row=0;row<7;row++){
      const z=-.36+row*.115;
      const pts=[];
      for(let j=0;j<12;j++){
        const x=-width*.83+j*.047;
        const zz=z+Math.sin(j*2.0+row)*.009;
        pts.push([x,.049+pageHeight(-x/width,zz,depth)+.002,zz]);
      }
      line(g,'faint-pencil-writing-'+row,pts,.0028,pencilWriting);
    }
  }
  return g;
}

function cylinderBetween(parent,name,a,b,radius,mat,segments=12){
  const av=V(a),bv=V(b),direction=bv.clone().sub(av);
  const m=mesh(parent,name,new THREE.CylinderGeometry(radius,radius,direction.length(),segments),mat);
  m.position.copy(av).add(bv).multiplyScalar(.5);
  m.quaternion.setFromUnitVectors(new THREE.Vector3(0,1,0),direction.normalize());
  return m;
}
function coneBetween(parent,name,a,b,radius,mat,segments=10){

  m.position.copy(av).add(bv).multiplyScalar(.5);
  m.quaternion.setFromUnitVectors(new THREE.Vector3(0,1,0),direction.normalize());
  return m;
}
