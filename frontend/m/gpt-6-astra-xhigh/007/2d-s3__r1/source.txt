const numberRE=/-?\d*\.?\d+(?:e[-+]?\d+)?/gi;
function compile(paths){
  return {
    pattern:paths[0],
    values:paths.map(s=>(s.match(numberRE)||[]).map(Number))
  };
}
const bodyData=compile(bodies),bellyData=compile(bellies);
const leftData=compile(leftLegs),rightData=compile(rightLegs),headData=compile(headPaths);
function pathAt(data,a,b,u){
  let n=0;
  return data.pattern.replace(numberRE,()=>{
    const v=data.values[a][n]+(data.values[b][n]-data.values[a][n])*u;
    n++;
    return Math.round(v*1000)/1000;
  });
}
function angleAt(a,b,u){
  return a+(((b-a+540)%360)-180)*u;
}
function point(a,t){
  const s=1-t;
  return [
    s*s*s*a[0]+3*s*s*t*a[2]+3*s*t*t*a[4]+t*t*t*a[6],
    s*s*s*a[1]+3*s*s*t*a[3]+3*s*t*t*a[5]+t*t*t*a[7]
  ];
}
function tube(a,bulge=1.8){
  const left=[],right=[];
  for(let i=0;i<=28;i++){
    const t=i/28,s=1-t,p=point(a,t);
    const dx=3*s*s*(a[2]-a[0])+6*s*t*(a[4]-a[2])+3*t*t*(a[6]-a[4]);
    const dy=3*s*s*(a[3]-a[1])+6*s*t*(a[5]-a[3])+3*t*t*(a[7]-a[5]);
    const length=Math.hypot(dx,dy)||1;
    const r=a[8]*(1-t)+a[9]*t+Math.sin(Math.PI*t)*bulge;
    const nx=-dy/length*r,ny=dx/length*r;
    left.push([p[0]+nx,p[1]+ny]);
    right.push([p[0]-nx,p[1]-ny]);
  }
  const coord=p=>p[0].toFixed(3)+" "+p[1].toFixed(3);
  let d="M"+coord(left[0]);
  for(let i=1;i<left.length;i++)d+="L"+coord(left[i]);
  d+=`A${a[9]} ${a[9]} 0 0 0 `+coord(right[28]);
  for(let i=27;i>=0;i--)d+="L"+coord(right[i]);
  d+=`A${a[8]} ${a[8]} 0 0 0 `+coord(left[0])+"Z";
  return d;
}
function tail(a,t){
  const mix=(p,q)=>[p[0]+(q[0]-p[0])*t,p[1]+(q[1]-p[1])*t];
  const p0=[a[0],a[1]],p1=[a[2],a[3]],p2=[a[4],a[5]],p3=[a[6],a[7]];
  const q0=mix(p0,p1),q1=mix(p1,p2),q2=mix(p2,p3);
  const r0=mix(q0,q1),r1=mix(q1,q2),s=mix(r0,r1);
  const radius=a[8]*(1-t)+a[9]*t+Math.sin(Math.PI*t)*1.8;
  return [...s,...r1,...q2,...p3,radius,a[9]];
}
function centerPath(a){
  return `M${a[0]} ${a[1]}C${a[2]} ${a[3]} ${a[4]} ${a[5]} ${a[6]} ${a[7]}`;
}

}

const clamp=x=>Math.max(0,Math.min(1,x));

function renderHand(which,arm,p){
  const h=hands[which];
  const [angle,sx,sy,spread,curl,mirror]=p;
  set(h.g,{transform:`translate(${arm[6]} ${arm[7]}) rotate(${angle}) scale(${sx*mirror} ${sy})`});
  const closed=clamp((curl-.42)/.35);
  const base=[-5.8,-2,2,5.7],ends=[-8,-2.5,3.5,8],fan=[-8,-2,4,8],lengths=[24,29,27,22];
  for(let k=0;k<4;k++){
    const x=base[k],ex=ends[k]+fan[k]*spread;
    const endY=lengths[k]*(1-curl)+(15+k*.45)*curl;
    const bendY=lengths[k]+curl*3;
    const cx=x+(ex-x)*.6;
    const d=`M${x} 11Q${cx} ${bendY} ${ex} ${endY}`;
    set(h.fingers[k],{d,opacity:1-closed});
    set(h.highlights[k],{
      d:`M${x-.7} 12Q${cx-.7} ${bendY-1} ${ex-.6} ${endY-1.1}`,
      opacity:.38*(1-.5*curl)*(1-closed)
    });
  }
  set(h.fist,{opacity:closed});
  set(h.folds,{opacity:closed*.35});
  const tx=(11+13*spread)*(1-closed)+9.4*closed;
  const ty=(17-20*spread)*(1-closed)+13*closed;
  const controlX=(13+6*spread)*(1-closed)+12*closed;
  set(h.thumb,{d:`M6 6Q${controlX} 8 ${tx} ${ty}`});
}
function renderArm(which,a,h){
  set(which+"-arm-outline",{d:tube(a)});
  set(which+"-arm-highlight",{d:centerPath(a)});
  const wrist=tail(a,.73);
  set(which+"-arm-wrist-shade",{d:tube(wrist,0)});
  set(which+"-wrist-color",{x1:wrist[0],y1:wrist[1],x2:a[6],y2:a[7]});
  renderHand(which,a,h);
}
function ellipse(id,cx,cy,rx,ry,opacity){

function mixColor(a,b,u){
  return "rgb("+a.map((v,i)=>Math.round(v+(b[i]-v)*u)).join(",")+")";
}
