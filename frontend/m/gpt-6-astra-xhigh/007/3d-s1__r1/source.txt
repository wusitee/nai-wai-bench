function ellipsoid(parent,name,position,scale,mat){
  const m=new THREE.Mesh(new THREE.SphereGeometry(1,32,24),mat);
  m.name=name;

}
function cubic(a,b,c,d,t){
  const t2=t*t,t3=t2*t;
  return .5*((2*b)+(-a+c)*t+(2*a-5*b+4*c-d)*t2+(-a+3*b-3*c+d)*t3);
}
function sampleProfile(profile,t){
  const s=clamp(t,0,1)*(profile.length-1);
  const i=Math.min(profile.length-2,Math.floor(s));
  const f=s-i;
  const a=profile[Math.max(0,i-1)],b=profile[i];
  const c=profile[i+1],d=profile[Math.min(profile.length-1,i+2)];
  return b.map((_,j)=>cubic(a[j],b[j],c[j],d[j],f));
}
function profileSurface(profile){
  return (t,theta)=>{
    const [y,rx,rz,cx,cz]=sampleProfile(profile,t);
    return v((cx||0)+Math.max(0,rx)*Math.sin(theta),y,
      (cz||0)+Math.max(0,rz)*Math.cos(theta));
  };
}
function surfaceMesh(parent,name,surface,colorFn,rows=80,columns=80){
  const positions=[],uvs=[],colors=[],indices=[];
  for(let i=0;i<=rows;i++){
    const t=i/rows;
    for(let j=0;j<=columns;j++){
      const theta=j/columns*Math.PI*2;
      const p=surface(t,theta);
      positions.push(p.x,p.y,p.z);
      uvs.push(j/columns,t);
      const c=colorFn(p,theta,t);
      colors.push(c.r,c.g,c.b);
    }
  }
  for(let i=0;i<rows;i++){
    for(let j=0;j<columns;j++){
      const a=i*(columns+1)+j,b=a+columns+1;
      indices.push(a,a+1,b,a+1,b+1,b);
    }
  }
  const geo=new THREE.BufferGeometry();
  geo.setAttribute('position',new THREE.Float32BufferAttribute(positions,3));
  geo.setAttribute('normal',new THREE.Float32BufferAttribute(new Float32Array(positions.length),3));
  geo.setAttribute('uv',new THREE.Float32BufferAttribute(uvs,2));
  geo.setAttribute('color',new THREE.Float32BufferAttribute(colors,3));
  geo.setIndex(indices);
  geo.computeVertexNormals();
  const mesh=new THREE.Mesh(geo,skinMaterial);
  mesh.name=name;
  mesh.castShadow=true;
  mesh.receiveShadow=true;
  parent.add(mesh);
  return mesh;
}

function fineFur(parent,name,surface,colorFn,count,length){
  const positions=[],colors=[];
  for(let i=0;i<count;i++){
    const t=.035+random()*.93;
    const theta=random()*Math.PI*2;
    const p=surface(t,theta);
    const vertical=surface(Math.min(.999,t+.001),theta).sub(surface(Math.max(.001,t-.001),theta)).normalize();
    const across=surface(t,theta+.002).sub(surface(t,theta-.002)).normalize();
    const normal=vertical.clone().cross(across).normalize();
    if(normal.dot(v(Math.sin(theta),0,Math.cos(theta)))<0) normal.negate();
    const base=p.clone().addScaledVector(normal,.0015);
    const width=.003+random()*.003;
    const end=base.clone().addScaledVector(normal,length*(.45+random()*.55));
    end.y-=length*.18;
    const a=base.clone().addScaledVector(across,width);
    const b=base.clone().addScaledVector(across,-width);
    const shade=colorFn(p,theta,t).multiplyScalar(.80+random()*.28);
    for(const q of [a,b,end]){
      positions.push(q.x,q.y,q.z);
      colors.push(shade.r,shade.g,shade.b);
    }
  }
  const geo=new THREE.BufferGeometry();
  geo.setAttribute('position',new THREE.Float32BufferAttribute(positions,3));
  geo.setAttribute('color',new THREE.Float32BufferAttribute(colors,3));
  geo.computeVertexNormals();
  const mesh=new THREE.Mesh(geo,new THREE.MeshStandardMaterial({
    vertexColors:true,roughness:1,side:THREE.DoubleSide,
    transparent:true,opacity:.47,depthWrite:false
  }));
  mesh.name=name;
  parent.add(mesh);
}

function curvedTube(parent,name,points,radii,colorFn,segments=48,sides=14){
  const curve=new THREE.CatmullRomCurve3(points);
  const frames=curve.computeFrenetFrames(segments,false);
  const positions=[],colors=[],uvs=[],indices=[];
  for(let i=0;i<=segments;i++){
    const t=i/segments,p=curve.getPointAt(t);
    const f=t*(radii.length-1),k=Math.min(radii.length-2,Math.floor(f));
    const r=THREE.MathUtils.lerp(radii[k],radii[k+1],f-k);
    const c=colorFn(t);
    for(let j=0;j<=sides;j++){
      const a=j/sides*Math.PI*2;
      const q=p.clone().addScaledVector(frames.normals[i],Math.cos(a)*r)
        .addScaledVector(frames.binormals[i],Math.sin(a)*r);
      positions.push(q.x,q.y,q.z);
      colors.push(c.r,c.g,c.b);
      uvs.push(j/sides,t);
    }
  }
  for(let i=0;i<segments;i++)for(let j=0;j<sides;j++){
    const a=i*(sides+1)+j,b=a+sides+1;
    indices.push(a,b,a+1,a+1,b,b+1);
  }
  const geo=new THREE.BufferGeometry();
  geo.setAttribute('position',new THREE.Float32BufferAttribute(positions,3));
  geo.setAttribute('color',new THREE.Float32BufferAttribute(colors,3));
  geo.setAttribute('uv',new THREE.Float32BufferAttribute(uvs,2));
  geo.setIndex(indices);
  geo.computeVertexNormals();
  const mesh=new THREE.Mesh(geo,skinMaterial);
  mesh.name=name;
  mesh.castShadow=true;
  mesh.receiveShadow=true;
  parent.add(mesh);
  return mesh;
}

function makeArm(side){
  const left=side<0;
  const shoulder=v(side*.60,4.34,-.045);
  const elbow=v(side*(left?1.075:1.055),3.03,.005);
  const wrist=v(side*(left?1.285:1.225),left?2.16:2.055,.075);
  const arm=group(left?'left-arm':'right-arm',character,shoulder);
  const upperPoints=[
    shoulder,
    v(side*.79,4.02,-.035),
    v(side*.96,3.49,-.015),
    elbow
  ].map(p=>p.clone().sub(shoulder));
  curvedTube(arm,'upper-arm',upperPoints,[.215,.205,.165,.13],
    t=>color(0xf6bc18).lerp(color(0xe8ad11),t*.55));
  const lower=group('forearm',arm,elbow.clone().sub(shoulder));
  const lowerPoints=[
    elbow,
    v(side*(left?1.16:1.12),2.64,.025),
    v(side*(left?1.235:1.18),2.30,.055),
    wrist
  ].map(p=>p.clone().sub(elbow));
  curvedTube(lower,'tapered-forearm',lowerPoints,[.132,.117,.095,.079],t=>{
    const c=color(0xe6ae1e);
    c.lerp(color(0x9b882d),clamp((t-.50)/.40,0,1)*.65);
    return c.lerp(color(0x55503a),clamp((t-.80)/.20,0,1)*.85);
  });
  ellipsoid(lower,'elbow',v(0,0,0),[.132,.16,.13],material(0xe8b11d));
  const palm=v(side*(left?1.335:1.285),left?1.985:1.88,.105);
  makeHand(lower,side,palm.sub(elbow));
}
