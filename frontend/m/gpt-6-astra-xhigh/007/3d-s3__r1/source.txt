function cubic(a,b,c,d,t){
  const t2=t*t,t3=t2*t;
  return .5*(2*b+(-a+c)*t+(2*a-5*b+4*c-d)*t2+(-a+3*b-3*c+d)*t3);
}
function sampleProfile(profile,t){
  const s=clamp(t,0,1)*(profile.length-1);
  const i=Math.min(profile.length-2,Math.floor(s)),f=s-i;
  const a=profile[Math.max(0,i-1)],b=profile[i];
  const c=profile[i+1],d=profile[Math.min(profile.length-1,i+2)];
  return b.map((_,j)=>cubic(a[j],b[j],c[j],d[j],f));
}
function profileSurface(profile){
  return (t,theta)=>{
    const [y,rx,rz,cx,cz]=sampleProfile(profile,t);
    return v((cx||0)+Math.max(0,rx)*Math.sin(theta),y,
      (cz||0)+Math.max(0,rz)*Math.cos(theta));
  };
}
function surfaceMesh(parent,name,surface,colorFn,rows=80,columns=72){
  const positions=[],uvs=[],colors=[],indices=[];
  for(let i=0;i<=rows;i++){
    const t=i/rows;
    for(let j=0;j<=columns;j++){
      const theta=j/columns*Math.PI*2,p=surface(t,theta);
      positions.push(p.x,p.y,p.z);
      uvs.push(j/columns,t);
      const c=colorFn(p,theta,t);
      colors.push(c.r,c.g,c.b);
    }
  }
  for(let i=0;i<rows;i++)for(let j=0;j<columns;j++){
    const a=i*(columns+1)+j,b=a+columns+1;
    indices.push(a,a+1,b,a+1,b+1,b);
  }
  const geo=new THREE.BufferGeometry();
  geo.setAttribute('position',new THREE.Float32BufferAttribute(positions,3));
  geo.setAttribute('uv',new THREE.Float32BufferAttribute(uvs,2));
  geo.setAttribute('color',new THREE.Float32BufferAttribute(colors,3));
  geo.setIndex(indices);
  geo.computeVertexNormals();
  const mesh=new THREE.Mesh(geo,skinMaterial);
  mesh.name=name;
  mesh.castShadow=true;
  mesh.receiveShadow=true;
  parent.add(mesh);
  return mesh;
}
function fineFur(parent,name,surface,colorFn,count,length){
  const positions=[],colors=[],normals=[];
  for(let i=0;i<count;i++){
    const t=.025+random()*.95,theta=random()*Math.PI*2;
    const p=surface(t,theta);
    const vertical=surface(Math.min(.999,t+.001),theta)
      .sub(surface(Math.max(.001,t-.001),theta)).normalize();
    const across=surface(t,theta+.002).sub(surface(t,theta-.002)).normalize();
    const normal=vertical.clone().cross(across).normalize();
    if(normal.dot(v(Math.sin(theta),0,Math.cos(theta)))<0)normal.negate();
    const base=p.clone().addScaledVector(normal,.002);
    const width=.002+random()*.0025;
    const end=base.clone().addScaledVector(normal,length*(.45+random()*.55));
    end.y-=length*.20;
    const a=base.clone().addScaledVector(across,width);
    const b=base.clone().addScaledVector(across,-width);
    const c=colorFn(p,theta,t).multiplyScalar(.96+random()*.08);
    for(const q of [a,b,end]){
      positions.push(q.x,q.y,q.z);
      colors.push(c.r,c.g,c.b);
      normals.push(normal.x,normal.y,normal.z);
    }
  }
  const geo=new THREE.BufferGeometry();
  geo.setAttribute('position',new THREE.Float32BufferAttribute(positions,3));
  geo.setAttribute('color',new THREE.Float32BufferAttribute(colors,3));
  geo.setAttribute('normal',new THREE.Float32BufferAttribute(normals,3));
  const mesh=new THREE.Mesh(geo,new THREE.MeshStandardMaterial({
    vertexColors:true,roughness:1,side:THREE.DoubleSide,
    transparent:true,opacity:.28,depthWrite:false
  }));
  mesh.name=name;
  parent.add(mesh);
}

function poseArm(rig,p){
  const elbow=v(...p.e),palm=v(...p.h);
  const handQ=new THREE.Quaternion().setFromEuler(new THREE.Euler(...p.r,'XYZ'));
  const wrist=palm.clone().add(v(0,.17*p.sz,-.015*p.sz).applyQuaternion(handQ));
  const shoulder=rig.shoulder.clone();
  const upperDirection=elbow.clone().sub(shoulder).normalize();
  const start=shoulder.clone().addScaledVector(upperDirection,-.19);
  const points=[
    start,
    shoulder.clone(),
    shoulder.clone().lerp(elbow,.65),
    elbow.clone(),
    elbow.clone().lerp(wrist,.34),
    wrist.clone()
  ];
  const curve=new THREE.CatmullRomCurve3(points,false,'centripetal');
  const firstLength=shoulder.distanceTo(elbow)+.19;
  const secondLength=elbow.distanceTo(wrist);
  const bend=firstLength/(firstLength+secondLength);
  const pos=rig.geometry.attributes.position;
  const nor=rig.geometry.attributes.normal;
  const col=rig.geometry.attributes.color;
  for(let i=0;i<=ARM_RINGS;i++){
    const u=i/ARM_RINGS;
    const center=curve.getPointAt(u);
    const tangent=curve.getTangentAt(u).normalize();
    let a=v(0,0,1).cross(tangent);
    if(a.lengthSq()<.001)a=v(1,0,0).cross(tangent);
    a.normalize();
    const b=tangent.clone().cross(a).normalize();
    let radius;
    if(u<bend){
      const f=u/Math.max(.001,bend);
      radius=.267*(1-f)+.187*f;
    }else{
      const f=(u-bend)/Math.max(.001,1-bend);
      radius=.187*(1-f)+.093*f;
    }
    radius*=Math.sin(Math.min(1,u/.085)*Math.PI*.5);
    radius*=Math.sin(Math.min(1,(1-u)/.035)*Math.PI*.5);
    const c=color(0xffc62c).lerp(color(0xe9b22c),u*.40);
    c.lerp(color(0xa3954d),clamp((u-.72)/.23,0,1)*.65);
    c.lerp(color(0x655e3e),clamp((u-.91)/.09,0,1)*.82);
    for(let j=0;j<=ARM_SIDES;j++){
      const theta=j/ARM_SIDES*Math.PI*2;
      const normal=a.clone().multiplyScalar(Math.cos(theta)).addScaledVector(b,Math.sin(theta));
      const point=center.clone().addScaledVector(normal,radius);
      const index=i*(ARM_SIDES+1)+j;
      pos.setXYZ(index,point.x,point.y,point.z);
      nor.setXYZ(index,normal.x,normal.y,normal.z);
      col.setXYZ(index,c.r,c.g,c.b);
    }
  }
  pos.needsUpdate=true;
  nor.needsUpdate=true;
  col.needsUpdate=true;
  rig.geometry.computeBoundingBox();
  rig.geometry.computeBoundingSphere();
  rig.elbow.position.copy(elbow);
  rig.forearm.position.copy(wrist);
  rig.hand.group.position.copy(palm).sub(wrist);
  rig.hand.group.quaternion.copy(handQ);
  rig.hand.group.scale.setScalar(p.sz);
  rig.hand.fingers.forEach((f,i)=>{
    f.base.rotation.set(-(.09+p.c*.78),0,rig.hand.side*(i-1.45)*.20*p.sp);
    f.middle.rotation.set(-(.10+p.c*1.05),0,0);
    f.tip.rotation.set(-(.10+p.c*.78),0,0);
  });
  rig.hand.thumb.rotation.set(-p.c*.65,0,-rig.hand.side*(.66+.21*p.sp));
  rig.hand.thumbTip.rotation.set(-p.c*.75,0,rig.hand.side*.36);
}

function mix(a,b,f){
  if(typeof a==='number')return a+(b-a)*f;
  if(Array.isArray(a))return a.map((value,i)=>mix(value,b[i],f));
  const result={};
  for(const key of Object.keys(a))result[key]=mix(a[key],b[key],f);
  return result;
}
function sample(t){
  let index=0;
  while(index<poses.length-2&&t>=poses[index+1].t)index++;
  const a=poses[index],b=poses[index+1];
  let f=(t-a.t)/(b.t-a.t);
  f=f*f*(3-2*f);
  return mix(a,b,f);
}
