    const t=i/28,s=1-t,p=point(a,t);
    const dx=3*s*s*(a[2]-a[0])+6*s*t*(a[4]-a[2])+3*t*t*(a[6]-a[4]);
    const dy=3*s*s*(a[3]-a[1])+6*s*t*(a[5]-a[3])+3*t*t*(a[7]-a[5]);
    const length=Math.hypot(dx,dy)||1;
    const r=a[8]*(1-t)+a[9]*t+Math.sin(Math.PI*t)*1.8;
    const nx=-dy/length*r,ny=dx/length*r;
    left.push([p[0]+nx,p[1]+ny]);
    right.push([p[0]-nx,p[1]-ny]);
  }
  const pairs=left.concat(right.reverse());
  return pairs.map((p,i)=>(i?"L":"M")+p[0].toFixed(3)+" "+p[1].toFixed(3)).join("")+"Z";
}
function centerPath(a){
  return `M${a[0]} ${a[1]}C${a[2]} ${a[3]} ${a[4]} ${a[5]} ${a[6]} ${a[7]}`;
}
function makeElement(name,attrs,parent){
  const e=document.createElementNS(NS,name);
  set(e,attrs);
  parent.appendChild(e);
  return e;
}
function makeHand(id){
  const g=$(id);
  const fingers=[],highlights=[];
  for(let i=0;i<4;i++){
    fingers.push(makeElement("path",{
      id:id+"-finger-"+i,fill:"none",stroke:"#535b43",
      "stroke-width":[3.4,3.7,3.6,3.1][i],
      "stroke-linecap":"round","stroke-linejoin":"round"
    },g));
    highlights.push(makeElement("path",{
      fill:"none",stroke:"#a39e70","stroke-width":".8",
      "stroke-linecap":"round",opacity:".43"
    },g));
  }
  const thumb=makeElement("path",{
    id:id+"-thumb",fill:"none",stroke:"#696c4c",
    "stroke-width":"4.6","stroke-linecap":"round"
  },g);
  makeElement("path",{
    id:id+"-palm",
    d:"M-4.8-1C-7 1-8.1 5-7.2 9.5L-5.9 15C-3 18 4.3 17.5 7 13C9 10 8.4 4 5.1 1L4-1Z",
    fill:"url(#hand-color)"
  },g);
  makeElement("path",{
    d:"M-3.6 2Q-5.3 7-3.2 12",fill:"none",stroke:"#a4a06b",
    "stroke-width":"1.3",opacity:".35","stroke-linecap":"round"
  },g);
  return {g,fingers,highlights,thumb};
}
const hands={near:makeHand("near-hand"),far:makeHand("far-hand")};

function renderHand(which,arm,p){
  const h=hands[which];
  const [angle,sx,sy,spread,curl,mirror]=p;
  set(h.g,{transform:`translate(${arm[6]} ${arm[7]}) rotate(${angle}) scale(${sx*mirror} ${sy})`});
  const base=[-5.8,-2,2,5.7],ends=[-8,-2.5,3.5,8],fan=[-8,-2,4,8],lengths=[24,29,27,22];
  for(let k=0;k<4;k++){
    const x=base[k],ex=ends[k]+fan[k]*spread;
    const endY=lengths[k]*(1-curl)+(15+k*.45)*curl;
    const bendY=lengths[k]+curl*3;
    const cx=x+(ex-x)*.6;
    const d=`M${x} 11Q${cx} ${bendY} ${ex} ${endY}`;
    set(h.fingers[k],{d});
    set(h.highlights[k],{d:`M${x-.7} 12Q${cx-.7} ${bendY-1} ${ex-.6} ${endY-1.1}`,opacity:.43*(1-.5*curl)});
  }
  const tx=11+13*spread,ty=17-20*spread;
  set(h.thumb,{d:`M6 6Q${13+6*spread} 8 ${tx} ${ty}`});
}
function renderArm(which,a,h){
  set(which+"-arm-outline",{d:tube(a)});
  set(which+"-arm-highlight",{d:centerPath(a)});
  set(which+"-arm-color",{x1:a[0],y1:a[1],x2:a[6],y2:a[7]});
  renderHand(which,a,h);
}
function ellipse(id,cx,cy,rx,ry,opacity){
  const a={cx,cy,rx,ry};
  if(opacity!==undefined)a.opacity=opacity;
  set(id,a);
}
function mixColor(a,b,u){
  return "rgb("+a.map((v,i)=>Math.round(v+(b[i]-v)*u)).join(",")+")";
}

function render(t){
  t=Number(t);
  if(!Number.isFinite(t))t=0;
  const time=((t%5540)+5540)%5540;
  let a=0;
  while(a<times.length-1&&time>=times[a+1])a++;
  const b=Math.min(a+1,times.length-1);
  let u=a===b?0:(time-times[a])/(times[b]-times[a]);
  u=u*u*(3-2*u);
  const p={};
  for(const name in frames[a]){
    p[name]=frames[a][name].map((v,i)=>v+(frames[b][name][i]-v)*u);
  }
  p.nh[0]=angleAt(frames[a].nh[0],frames[b].nh[0],u);
  p.fh[0]=angleAt(frames[a].fh[0],frames[b].fh[0],u);

  set("torso-outline",{d:pathAt(bodyData,a,b,u)});
  set("belly-outline",{d:pathAt(bellyData,a,b,u)});
  set("left-leg-outline",{d:pathAt(leftData,a,b,u)});
  set("right-leg-outline",{d:pathAt(rightData,a,b,u)});
  const l=p.legLight;
  set("left-leg-highlight",{d:`M${l[0]} ${l[1]}Q${l[2]} ${l[3]} ${l[4]} ${l[5]}`});
  set("right-leg-highlight",{d:`M${l[6]} ${l[7]}Q${l[8]} ${l[9]} ${l[10]} ${l[11]}`});

  const yaw=p.yaw[0],h=p.head,light=p.light;
  ellipse("chest-highlight",light[0],light[1],light[2],light[3]);
  ellipse("belly-blush",light[0]+10,light[1]+31,light[2]+1,light[3]-.5);
  ellipse("neck-shadow",h[0]+1,h[1]+29*h[3],21*h[2],14*h[3],.33);
  ellipse("back-highlight",h[0]+36,h[1]+53,30,50,yaw*.8);
  ellipse("back-blush",h[0]+52,h[1]+124,34,45,yaw*.85);
  ellipse("underarm-shadow",p.near[0]-8,p.near[1]+26,15,8,yaw*.35);
  set("body-right-color",{"stop-color":mixColor([209,179,70],[239,186,36],yaw)});
  set("far-arm-back-copy",{opacity:yaw});
  set("far-arm-front-copy",{opacity:1-yaw});
  set("floor-shadow",{cx:light[0]+8,rx:55+light[2]*.25});

  set("head",{transform:`translate(${h[0]} ${h[1]}) rotate(${h[4]}) scale(${h[2]} ${h[3]}) translate(-100 -49)`});
  set("head-outline",{d:pathAt(headData,0,1,yaw)});

  const f=p.face;
  let open=f[4];
  const blink1=Math.max(0,1-Math.abs(time-1860)/65);
  const blink2=Math.max(0,1-Math.abs(time-4390)/60);
  open*=1-.93*Math.max(blink1,blink2);
  set("left-eye",{transform:`translate(${f[0]} ${f[1]})`});
  set("right-eye",{transform:`translate(${f[2]} ${f[3]})`,opacity:f[5]});
  ["left","right"].forEach((side,index)=>{
    const r=f[12+index];
    set(side+"-eye-pupil",{
      rx:(1.35+1.85*open)*r,
      ry:1.0+3.3*open,
      fill:mixColor([125,132,57],[13,26,12],open)
    });
    set(side+"-eye-socket",{opacity:.24+.76*open});
    set(side+"-eye-glint",{opacity:.45*open});
    set(side+"-eye-lid",{opacity:1-open});
  });
  ellipse("face-shading",(f[0]+f[2])/2,(f[1]+f[3])/2+1,16-7*yaw,9,.18);
  const nx=f[10],ny=f[11];
  set("nose-outline",{
    d:`M${nx-4} ${ny-5}C${nx-3} ${ny-2} ${nx+2} ${ny-3} ${nx+5} ${ny-1}C${nx+7} ${ny+1} ${nx+3} ${ny+3} ${nx-3} ${ny+1}Z`,
    opacity:1-.25*yaw
  });
  set("nose-shadow",{d:`M${nx} ${ny+1}Q${nx+3} ${ny+2} ${nx+5} ${ny}`});
  const mx=f[6],my=f[7],mw=f[8],curve=f[9];
  ellipse("muzzle",mx,my,mw*.65,4.1,.4-.13*yaw);
  const md=`M${mx-mw/2} ${my}C${mx-mw*.2} ${my+curve} ${mx+mw*.22} ${my+curve} ${mx+mw/2} ${my-.2}`;
  set("mouth-line",{d:md});
  set("lower-lip",{d:`M${mx-mw*.34} ${my+3}Q${mx} ${my+4+curve*.4} ${mx+mw*.36} ${my+2.8}`,opacity:.7-.3*yaw});

  renderArm("far",p.far,p.fh);
  renderArm("near",p.near,p.nh);
}
