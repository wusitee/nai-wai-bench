    "left-cast-shadow","right-cast-shadow",
    "left-arm-highlight","left-arm-shade",
    "right-arm-highlight","right-arm-shade",
    "hand-left","hand-right"
  ];
  const el = {};
  ids.forEach(function(id){el[id]=document.getElementById(id);});

  function curve(a,closed){
    let d="M"+a[0]+" "+a[1];
    for(let i=2;i<a.length;i+=6){
      d+="C"+a.slice(i,i+6).join(" ");
    }
    return d+(closed?"Z":"");
  }

  function handTransform(a,cx,cy){
    return "translate("+a[0]+" "+a[1]+") rotate("+a[4]+") scale("+
      a[2]+" "+a[3]+") translate("+(-cx)+" "+(-cy)+")";
  }

  function render(time){
    let t=Number(time);
    if(!Number.isFinite(t)) t=0;
    t=((t%1200)+1200)%1200;

    let i=0;
    while(i<poses.length-2 && t>=poses[i+1].t) i++;

    const a=poses[i], b=poses[i+1];
    const u=(t-a.t)/(b.t-a.t);
    const w=u*u*(3-2*u);

    function mix(key){
      return a[key].map(function(v,j){return v+(b[key][j]-v)*w;});
    }

    const core=mix("core");
    const c=core[2];
    const transform="translate("+core[0]+" "+core[1]+") translate(124 0) scale("+core[4]+" 1) translate(-124 0)";

    const crownStart="M99 "+(-6+c)+
      "C90 "+(4+c*.75)+" 89 "+(20+c*.3)+" 88 37";
    const crownEnd="C157 "+(8+c*.55)+" 154 "+(-2+c)+" 145 "+(-7+c)+
      "C132 "+(-12+c)+" 111 "+(-13+c)+" 99 "+(-6+c)+"Z";

    const body=crownStart+
      "C87 50 78 71 65 94"+
      "C56 116 48 139 42 161"+
      "C36 183 32 205 36 224L42 244H211"+
      "C218 226 219 207 215 186"+
      "C212 165 206 146 199 124"+
      "C194 105 184 86 174 72"+
      "C164 58 164 42 160 24"+crownEnd;

    const head=crownStart+
      "C87 50 80 64 78 76"+
      "C86 91 108 104 133 101"+
      "C155 99 175 86 178 76"+
      "C164 56 164 42 160 24"+crownEnd;

    el["body-and-head"].setAttribute("transform",transform);
    el["body-shape"].setAttribute("d",body);
    el["body-clip-shape"].setAttribute("d",body);
    el["cast-shadow-clip-shape"].setAttribute("d",body);
    el["cast-shadow-clip-shape"].setAttribute("transform",transform);
    el["head-shape"].setAttribute("d",head);
    el["head-clip-shape"].setAttribute("d",head);
    el.face.setAttribute("transform","translate(-.7 0) rotate("+core[3]+" 124 25)");

    const left=curve(mix("l"),true);
    const right=curve(mix("r"),true);

    el["left-arm-shape"].setAttribute("d",left);
    el["left-cast-shadow"].setAttribute("d",left);
    el["right-arm-shape"].setAttribute("d",right);
    el["right-cast-shadow"].setAttribute("d",right);

    el["left-arm-highlight"].setAttribute("d",curve(mix("hl"),false));
    el["left-arm-shade"].setAttribute("d",curve(mix("sl"),false));
    el["right-arm-highlight"].setAttribute("d",curve(mix("hr"),false));
    el["right-arm-shade"].setAttribute("d",curve(mix("sr"),false));
    el["hand-left"].setAttribute("transform",handTransform(mix("lh"),70,93));
    el["hand-right"].setAttribute("transform",handTransform(mix("rh"),212,84));
  }

  window.nwb={render:render};
  render(0);
})();
