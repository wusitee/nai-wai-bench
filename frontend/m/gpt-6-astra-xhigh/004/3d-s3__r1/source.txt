function profileValue(y,k){
  if(y<=profile[0][0]) return profile[0][k];
  if(y>=profile[profile.length-1][0]) return profile[profile.length-1][k];
  let i=0;
  while(profile[i+1][0]<y) i++;
  const a=profile[i],b=profile[i+1];
  const previous=profile[Math.max(0,i-1)];
  const next=profile[Math.min(profile.length-1,i+2)];
  const span=b[0]-a[0],t=(y-a[0])/span;
  const m0=(b[k]-previous[k])/(b[0]-previous[0]);
  const m1=(next[k]-a[k])/(next[0]-a[0]);
  const t2=t*t,t3=t2*t;
  return Math.max(0.001,
    (2*t3-3*t2+1)*a[k]+(t3-2*t2+t)*span*m0+
    (-2*t3+3*t2)*b[k]+(t3-t2)*span*m1
  );
}

function skinColor(x,y,front){
  const c=goldLow.clone().lerp(
    goldMiddle,THREE.MathUtils.smoothstep(y,0.0,2.7)
  );
  c.lerp(goldHigh,THREE.MathUtils.smoothstep(y,3.15,4.48));
  const belly=bellyLow.clone().lerp(
    bellyHigh,THREE.MathUtils.smoothstep(y,0,2.6)
  );
  const oval=Math.exp(
    -Math.pow(x/1.10,4)-Math.pow((y-1.03)/1.61,4)
  );
  const facing=THREE.MathUtils.smoothstep(front,0.1,0.82);
  c.lerp(belly,0.95*oval*facing);
  return c;
}

function makeBody(){
  const y0=profile[0][0],y1=profile[profile.length-1][0];
  const rows=180,sides=96;
  const positions=[],colors=[],uvs=[],indices=[];
  for(let j=0;j<=rows;j++){
    const y=y0+(y1-y0)*j/rows;
    const r=profileValue(y,1),d=profileValue(y,2);
    for(let i=0;i<=sides;i++){
      const a=i/sides*Math.PI*2;
      const x=r*Math.sin(a),front=Math.cos(a);
      positions.push(x,y,d*front);
      const c=skinColor(x,y,front);
      colors.push(c.r,c.g,c.b);
      uvs.push(i/sides,(y-y0)/(y1-y0));
      if(j<rows&&i<sides){
        const p=j*(sides+1)+i,q=p+sides+1;
        indices.push(p,p+1,q,p+1,q+1,q);
      }
    }
  }
  for(const top of [false,true]){
    const row=top?rows:0;
    const y=top?y1:y0;
    const center=positions.length/3;
    positions.push(0,y,0);
    const c=skinColor(0,y,0);
    colors.push(c.r,c.g,c.b);
    uvs.push(0.5,0.5);
    for(let i=0;i<sides;i++){
      const a=row*(sides+1)+i;
      if(top) indices.push(center,a,a+1);
      else indices.push(center,a+1,a);
    }
  }
  const geometry=new THREE.BufferGeometry();
  geometry.setAttribute('position',new THREE.Float32BufferAttribute(positions,3));
  geometry.setAttribute('color',new THREE.Float32BufferAttribute(colors,3));
  geometry.setAttribute('uv',new THREE.Float32BufferAttribute(uvs,2));
  geometry.setIndex(indices);
  geometry.computeVertexNormals();

  const mesh=new THREE.Mesh(geometry,bodyMaterial);
  mesh.name='continuous-rounded-body-and-head';
  mesh.castShadow=true;
  mesh.receiveShadow=true;
  torso.add(mesh);
  return {mesh,geometry,rest:new Float32Array(positions)};
}

function poseBody(width,tilt){
  const positions=body.geometry.attributes.position;
  const rest=body.rest;
  for(let i=0;i<positions.count;i++){
    const x=rest[i*3],y=rest[i*3+1],z=rest[i*3+2];
    const weight=THREE.MathUtils.smoothstep(y,2.55,3.60);
    const a=tilt*weight,c=Math.cos(a),s=Math.sin(a);
    const w=1+(width-1)*(1-weight);
    const localY=y-3.25;
    positions.setXYZ(
      i,
      x*w*c-localY*s+0.9*Math.sin(tilt)*weight,
      x*w*s+localY*c+3.25,
      z
    );
  }
  positions.needsUpdate=true;
  body.geometry.computeVertexNormals();
  body.geometry.computeBoundingSphere();
}

const channels=Object.keys(frames[0]).filter(name=>name!=='t');
const count=frames.length;
const tangents=frames.map((frame,i)=>{
  const previous=(i+count-1)%count,next=(i+1)%count;
  const before=i===0?PERIOD-frames[previous].t:frame.t-frames[previous].t;
  const after=i===count-1?PERIOD-frame.t:frames[next].t-frame.t;
  const result={};
  for(const channel of channels){
    const a=(frame[channel]-frames[previous][channel])/before;
    const b=(frames[next][channel]-frame[channel])/after;
    if(a*b<=0){
      result[channel]=0;
    }else{
      const w1=2*after+before,w2=after+2*before;
      result[channel]=(w1+w2)/(w1/a+w2/b);
    }
  }
  return result;
});

function samplePose(time){
  let i=0;
  while(i<count-1&&time>=frames[i+1].t) i++;
  const next=(i+1)%count;
  const duration=next===0?PERIOD-frames[i].t:frames[next].t-frames[i].t;
  const u=(time-frames[i].t)/duration;
  const u2=u*u,u3=u2*u;
  const h00=2*u3-3*u2+1,h10=u3-2*u2+u;
  const h01=-2*u3+3*u2,h11=u3-u2;
  const result={};
  for(const channel of channels){
    result[channel]=
      h00*frames[i][channel]+
      h10*duration*tangents[i][channel]+
      h01*frames[next][channel]+
      h11*duration*tangents[next][channel];
  }
  return result;
}

const pixelsPerUnit=240/viewHeight;
const pitch=Math.atan2(0.46,10);
const pitchCos=Math.cos(pitch);
const pitchTan=Math.tan(pitch);

function displacedPoint(base,dx,dy,bob,depth=0){
  return new THREE.Vector3(
    base.x+dx/pixelsPerUnit,
    base.y+(bob-dy)/(pixelsPerUnit*pitchCos)+depth*pitchTan,
    base.z+depth
  );
}

function poseArm(rig,dx,dy,ex,ey,bob,angle,width,depth,turn){
  const elbow=displacedPoint(rig.baseElbow,ex,ey,bob,depth*0.25);
  const palm=displacedPoint(rig.baseHand,dx,dy,bob,depth);
  const localElbow=elbow.clone().sub(rig.arm.position);

  const upperTurn=new THREE.Quaternion().setFromUnitVectors(
    rig.restElbow.clone().normalize(),localElbow.clone().normalize()
  );
  rig.upperMesh.position.copy(rig.upperPosition).addScaledVector(
    localElbow.clone().sub(rig.restElbow),0.5
  );
  rig.upperMesh.quaternion.copy(upperTurn).multiply(rig.upperQuaternion);
  rig.upperMesh.scale.copy(rig.upperScale);
  rig.upperMesh.scale.y*=localElbow.length()/rig.restElbow.length();

  const reach=palm.clone().sub(elbow);
  const restAngle=Math.atan2(rig.restHand.y,rig.restHand.x);
  const forearmAngle=Math.atan2(reach.y,reach.x)-restAngle;
  const stretch=Math.hypot(reach.x,reach.y)/
    Math.hypot(rig.restHand.x,rig.restHand.y);

  rig.forearm.position.copy(localElbow);
  rig.forearm.rotation.set(0,0,forearmAngle);
  rig.forearm.scale.set(stretch,stretch,1);
  rig.hand.position.copy(rig.restHand);
  rig.hand.position.z=reach.z;
  rig.hand.rotation.set(0,turn,angle-forearmAngle);
  rig.hand.scale.set(width/stretch,1/stretch,0.96);
}

function render(t){
  const input=Number.isFinite(t)?t:0;
  const time=((input%PERIOD)+PERIOD)%PERIOD;
  const p=samplePose(time);

  character.position.set(0.035,-p.bob/(pixelsPerUnit*pitchCos),0);
  character.rotation.set(0,0,0);
  character.scale.set(1,1,1);

  poseBody(p.width,p.tilt);
  head.position.set(0.9*Math.sin(p.tilt),3.25,0);
  head.rotation.set(0,0,p.tilt);

  poseArm(
    leftRig,p.lx,p.ly,p.lex,p.ley,p.bob,
    p.la,p.lw,p.lz,0.14*Math.min(1,Math.abs(p.lx)/54)
  );
  poseArm(
    rightRig,p.rx,p.ry,p.rex,p.rey,p.bob,
    p.ra,p.rw,p.rz,-0.18*Math.min(1,Math.abs(p.rx)/56)
  );

  renderer.setRenderTarget(null);
  renderer.setViewport(0,0,244,240);
  renderer.setScissorTest(false);
  renderer.render(scene,ref);
}
