const sphereGeo = new THREE.SphereGeometry(1, 40, 28);

function ellipsoid(parent, name, position, scale, material) {
  const mesh = new THREE.Mesh(sphereGeo, material);
  mesh.name = name;
  mesh.position.set(...position);
  mesh.scale.set(...scale);
  mesh.castShadow = true;
  mesh.receiveShadow = true;
  parent.add(mesh);
  return mesh;
}

function capsuleBetween(parent, name, a, b, radius, material) {
  const start = new THREE.Vector3(...a);
  const end = new THREE.Vector3(...b);
  const delta = end.clone().sub(start);
  const length = Math.max(0.01, delta.length() - radius * 2);
  const mesh = new THREE.Mesh(new THREE.CapsuleGeometry(radius, length, 8, 20), material);
  mesh.name = name;
  mesh.position.copy(start).add(end).multiplyScalar(0.5);
  mesh.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), delta.normalize());
  mesh.castShadow = true;
  mesh.receiveShadow = true;
  parent.add(mesh);
  return mesh;
}

function conePart(parent, name, position, radius, height, direction, material) {
  const mesh = new THREE.Mesh(new THREE.ConeGeometry(radius, height, 20), material);
  mesh.name = name;
  mesh.position.set(...position);
  mesh.quaternion.setFromUnitVectors(
    new THREE.Vector3(0, 1, 0),
    new THREE.Vector3(...direction).normalize()
  );
  mesh.castShadow = true;
  parent.add(mesh);
  return mesh;
}

const tailCurve = new THREE.CatmullRomCurve3([
  new THREE.Vector3(0.45, 1.20, -0.62),
  new THREE.Vector3(1.05, 0.95, -1.00),
  new THREE.Vector3(1.72, 0.58, -0.72),
  new THREE.Vector3(2.00, 0.48, -0.05),
  new THREE.Vector3(1.83, 0.65, 0.48),
  new THREE.Vector3(1.55, 0.88, 0.62)
]);
const tailSegments = 48;
const tailRadial = 16;
const frames = tailCurve.computeFrenetFrames(tailSegments, false);
const tailPositions = [];
const tailIndices = [];

for (let i = 0; i <= tailSegments; i++) {
  const u = i / tailSegments;
  const p = tailCurve.getPointAt(u);
  const radius = 0.39 * Math.pow(1 - u, 0.82) + 0.012;
  for (let j = 0; j <= tailRadial; j++) {
    const angle = j / tailRadial * Math.PI * 2;
    const normal = frames.normals[i].clone().multiplyScalar(Math.cos(angle))
      .add(frames.binormals[i].clone().multiplyScalar(Math.sin(angle)));
    const v = p.clone().addScaledVector(normal, radius);
    tailPositions.push(v.x, v.y, v.z);
  }
}
for (let i = 0; i < tailSegments; i++) {
  for (let j = 0; j < tailRadial; j++) {
    const a = i * (tailRadial + 1) + j;
    const b = a + tailRadial + 1;
    tailIndices.push(a, b, a + 1, b, b + 1, a + 1);
  }
}

for (let j = 0; j < tailRadial; j++) {
  tailIndices.push(tailTipIndex, tailSegments * (tailRadial + 1) + j, tailSegments * (tailRadial + 1) + j + 1);
}

function laughingEye(name, x) {
  const group = new THREE.Group();
  group.name = name;
  group.position.set(x, 0.50, 1.005);

    new THREE.Vector3(-0.28, -0.025, 0),

  const mesh = new THREE.Mesh(new THREE.TubeGeometry(curve, 28, 0.038, 10, false), darkMat);
  mesh.castShadow = true;
  group.add(mesh);
  eyes.add(group);
}
