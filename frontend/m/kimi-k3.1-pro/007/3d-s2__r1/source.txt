function makeBackground() {
  const canvas = document.createElement('canvas');
  canvas.width = canvas.height = 64;
  const ctx = canvas.getContext('2d');
  const image = ctx.createImageData(64, 64);
  let seed = 918273;
  for (let y = 0; y < 64; y++) {
    for (let x = 0; x < 64; x++) {
      seed = (seed * 1664525 + 1013904223) >>> 0;
      const random = (seed / 4294967296 - 0.5) * 5;
      const nx = (x - 31.5) / 45;
      const ny = (y - 31.5) / 45;
      const edge = Math.min(1, Math.sqrt(nx * nx + ny * ny));
      const cloud = Math.sin(x * 0.31 + Math.sin(y * 0.17) * 2) *
                    Math.sin(y * 0.23 - x * 0.08) * 3;
      const value = 249 - edge * 17 - (1 - y / 63) * 4 + random + cloud;
      const i = (y * 64 + x) * 4;
      image.data[i] = value;
      image.data[i + 1] = value;
      image.data[i + 2] = value - 2;
      image.data[i + 3] = 255;
    }
  }
  ctx.putImageData(image, 0, 0);
  const texture = new THREE.CanvasTexture(canvas);
  texture.colorSpace = THREE.SRGBColorSpace;
  texture.minFilter = THREE.LinearFilter;
  texture.magFilter = THREE.LinearFilter;
  return texture;
}

  const v = THREE.MathUtils.clamp((x - a) / (b - a), 0, 1);
  return v * v * (3 - 2 * v);

function makeLimb(name, radius, material) {
  const mesh = new THREE.Mesh(
    new THREE.CapsuleGeometry(radius, 1, 8, 16),
    material
  );
  mesh.name = name;
  mesh.userData.radius = radius;
  return mesh;
}

const limbDirection = new THREE.Vector3();
const limbMidpoint = new THREE.Vector3();
const upAxis = new THREE.Vector3(0, 1, 0);

function placeLimb(mesh, a, b) {
  const start = new THREE.Vector3().fromArray(a);
  const end = new THREE.Vector3().fromArray(b);
  limbDirection.copy(end).sub(start);
  const length = Math.max(0.02, limbDirection.length());
  limbMidpoint.copy(start).add(end).multiplyScalar(0.5);
  mesh.position.copy(limbMidpoint);
  mesh.quaternion.setFromUnitVectors(upAxis, limbDirection.normalize());
  mesh.scale.set(1, length / (1 + mesh.userData.radius * 2), 1);
}

function buildHand(name, right = false) {
  const hand = new THREE.Group();
  hand.name = name;

  const palm = new THREE.Mesh(new THREE.SphereGeometry(1, 24, 18), handMaterial);
  palm.name = 'palm';
  palm.scale.set(right ? 0.155 : 0.18, right ? 0.23 : 0.255, 0.115);
  hand.add(palm);

  const lengths = right ? [0.25, 0.31, 0.30, 0.23] : [0.28, 0.36, 0.34, 0.26];
  const xs = [-0.125, -0.042, 0.045, 0.125];
  for (let i = 0; i < 4; i++) {
    const spread = (i - 1.5) * 0.035;
    const start = new THREE.Vector3(xs[i], -0.11, 0.005);
    const end = new THREE.Vector3(
      xs[i] + spread,
      -0.11 - lengths[i],
      0.015
    );
    const direction = end.clone().sub(start);
    const length = direction.length();
    const finger = new THREE.Mesh(
      new THREE.CapsuleGeometry(0.037, Math.max(0.01, length - 0.074), 6, 12),
      handMaterial
    );
    finger.name = 'finger-' + (i + 1);
    finger.position.copy(start).add(end).multiplyScalar(0.5);
    finger.quaternion.setFromUnitVectors(upAxis, direction.normalize());
    hand.add(finger);
  }

  const side = right ? -1 : 1;
  const start = new THREE.Vector3(side * 0.135, 0.055, 0.005);
  const end = new THREE.Vector3(side * 0.235, -0.18, 0.025);
  const direction = end.clone().sub(start);
  const thumb = new THREE.Mesh(
    new THREE.CapsuleGeometry(0.045, direction.length() - 0.09, 6, 12),
    handMaterial
  );
  thumb.name = 'thumb';
  thumb.position.copy(start).add(end).multiplyScalar(0.5);
  thumb.quaternion.setFromUnitVectors(upAxis, direction.normalize());
  hand.add(thumb);
  return hand;
}

}

function makeLeg(name, side) {
  const isLeft = side === 'left';
  const points = [
    new THREE.Vector2(0.001, -0.06),
    new THREE.Vector2(0.275, -0.045),
    new THREE.Vector2(0.315, 0.08),
    new THREE.Vector2(0.34, 0.48),
    new THREE.Vector2(0.395, 1.12),
    new THREE.Vector2(0.465, 1.78),
    new THREE.Vector2(isLeft ? 0.50 : 0.485, 2.34),
    new THREE.Vector2(0.001, 2.39)
  ];
  const geometry = new THREE.LatheGeometry(
    new THREE.SplineCurve(points).getPoints(48),
    48
  );
  const p = geometry.attributes.position;
  for (let i = 0; i < p.count; i++) {
    const x = p.getX(i);
    const y = p.getY(i);
    const z = p.getZ(i) * 0.88;
    const upperX = isLeft ? -0.49 : 0.38;
    const lowerX = isLeft ? -0.76 : 0.30;
    const blend = 1 - smoothstep(0.05, 2.25, y);
    const centerX = THREE.MathUtils.lerp(upperX, lowerX, blend);
    p.setXYZ(i, x + centerX, y, z);
  }
  geometry.computeVertexNormals();
  const leg = new THREE.Mesh(geometry, legMaterial);
  leg.name = name;
  return leg;
}

const scalarFields = ['x', 'zoom', 'cy', 'open', 'yaw', 'lean', 'size', 'mood'];

function mixArray(a, b, u) {
  return a.map((value, i) => Array.isArray(value)
    ? mixArray(value, b[i], u)
    : THREE.MathUtils.lerp(value, b[i], u));
}

function samplePose(phase) {
  let index = 0;
  while (index < keys.length - 2 && phase > keys[index + 1].t) index++;
  const a = keys[index];
  const b = keys[index + 1];
  let u = (phase - a.t) / (b.t - a.t);
  u = THREE.MathUtils.clamp(u, 0, 1);
  u = u * u * (3 - 2 * u);
  const pose = { l: mixArray(a.l, b.l, u), r: mixArray(a.r, b.r, u) };
  for (const field of scalarFields) {
    pose[field] = THREE.MathUtils.lerp(a[field], b[field], u);
  }
  return pose;
}

const wristOffset = new THREE.Vector3();
const wrist = new THREE.Vector3();
const handEuler = new THREE.Euler();

function poseArm(rig, data, shoulder) {
  const elbow = data[0];
  const handPosition = data[1];
  const rotation = data[2];
  rig.hand.position.fromArray(handPosition);
  rig.hand.rotation.set(rotation[0], rotation[1], rotation[2]);
  handEuler.set(rotation[0], rotation[1], rotation[2]);
  wristOffset.set(0, 0.235, 0).applyEuler(handEuler);
  wrist.fromArray(handPosition).add(wristOffset);
  placeLimb(rig.upper, shoulder, elbow);
  placeLimb(rig.lower, elbow, wrist.toArray());
}

function render(t) {
  const phase = ((t % 5540) + 5540) % 5540;
  const pose = samplePose(phase);

  character.position.set(pose.x, 0, 0);
  character.rotation.set(0, pose.yaw, pose.lean);
  character.scale.setScalar(pose.size);

  poseArm(leftArm, pose.l, [-0.99, 4.77, -0.01]);
  poseArm(rightArm, pose.r, [0.76, 4.76, -0.025]);

  for (const eye of openEyes) {
    eye.visible = pose.open > 0.025;
    eye.scale.set(
      0.055 * (0.45 + 0.55 * pose.open),
      0.008 + 0.077 * pose.open,
      0.027
    );
  }
  faceLineMaterial.opacity = 0.68 * (1 - pose.open);
  faceLineMaterial.visible = pose.open < 0.995;

  const smile = Math.max(0, pose.mood);
  const frown = Math.max(0, -pose.mood);
  const flat = Math.max(0, 1 - smile - frown);
  flatMouth.material.opacity = 0.78 * flat;
  smileMouth.material.opacity = 0.78 * smile;
  frownMouth.material.opacity = 0.78 * frown;
  flatMouth.visible = flat > 0.001;
  smileMouth.visible = smile > 0.001;
  frownMouth.visible = frown > 0.001;

  cameras.ref.position.set(0, pose.cy, 12);
  cameras.ref.lookAt(0, pose.cy, 0);
  cameras.ref.zoom = pose.zoom;
  cameras.ref.updateProjectionMatrix();

  renderer.render(scene, cameras.ref);
}
