function makeBackground() {
  const canvas = document.createElement('canvas');
  canvas.width = canvas.height = 64;
  const ctx = canvas.getContext('2d');
  const image = ctx.createImageData(64, 64);
  let seed = 918273;
  for (let y = 0; y < 64; y++) {
    for (let x = 0; x < 64; x++) {
      seed = (seed * 1664525 + 1013904223) >>> 0;
      const random = (seed / 4294967296 - 0.5) * 5;
      const nx = (x - 31.5) / 45;
      const ny = (y - 31.5) / 45;
      const edge = Math.min(1, Math.sqrt(nx * nx + ny * ny));
      const cloud = Math.sin(x * 0.31 + Math.sin(y * 0.17) * 2) *
                    Math.sin(y * 0.23 - x * 0.08) * 3;
      const value = 249 - edge * 17 - (1 - y / 63) * 4 + random + cloud;
      const i = (y * 64 + x) * 4;
      image.data[i] = value;
      image.data[i + 1] = value;
      image.data[i + 2] = value - 2;
      image.data[i + 3] = 255;
    }
  }
  ctx.putImageData(image, 0, 0);
  const texture = new THREE.CanvasTexture(canvas);
  texture.colorSpace = THREE.SRGBColorSpace;
  texture.minFilter = THREE.LinearFilter;
  texture.magFilter = THREE.LinearFilter;
  return texture;
}

function smoothstep(a, b, x) {
  const v = THREE.MathUtils.clamp((x - a) / (b - a), 0, 1);
  return v * v * (3 - 2 * v);
}

function addFaceCurve(parent, name, points, radius, material) {
  const curve = new THREE.CatmullRomCurve3(points);
  const mesh = new THREE.Mesh(new THREE.TubeGeometry(curve, 20, radius, 7, false), material);
  mesh.name = name;
  parent.add(mesh);
  return mesh;
}

function capsuleBetween(a, b, radius, material, name) {
  const start = new THREE.Vector3(...a);
  const end = new THREE.Vector3(...b);
  const direction = end.clone().sub(start);
  const length = direction.length();
  const geometry = new THREE.CapsuleGeometry(
    radius,
    Math.max(0.01, length - radius * 2),
    8,
    16
  );
  const mesh = new THREE.Mesh(geometry, material);
  mesh.name = name;
  mesh.position.copy(start).add(end).multiplyScalar(0.5);
  mesh.quaternion.setFromUnitVectors(
    new THREE.Vector3(0, 1, 0),
    direction.normalize()
  );
  return mesh;
}

function buildHand(name, position, rotation, right = false) {
  const hand = new THREE.Group();
  hand.name = name;
  hand.position.set(...position);
  hand.rotation.z = rotation;
  character.add(hand);

  const palm = new THREE.Mesh(new THREE.SphereGeometry(1, 24, 18), handMaterial);
  palm.name = 'palm';
  palm.scale.set(right ? 0.155 : 0.18, right ? 0.23 : 0.255, 0.115);
  hand.add(palm);

  const lengths = right ? [0.25, 0.31, 0.30, 0.23] : [0.28, 0.36, 0.34, 0.26];
  const xs = [-0.125, -0.042, 0.045, 0.125];
  for (let i = 0; i < 4; i++) {
    const spread = (i - 1.5) * 0.035;
    hand.add(capsuleBetween(
      [xs[i], -0.11, 0.005],
      [xs[i] + spread, -0.11 - lengths[i], 0.015],
      0.037,
      handMaterial,
      'finger-' + (i + 1)
    ));
  }

  const side = right ? -1 : 1;
  hand.add(capsuleBetween(
    [side * 0.135, 0.055, 0.005],
    [side * 0.235, -0.18, 0.025],
    0.045,
    handMaterial,
    'thumb'
  ));
  return hand;
}

function makeLeg(name, side) {
  const isLeft = side === 'left';
  const points = [
    new THREE.Vector2(0.001, -0.06),
    new THREE.Vector2(0.275, -0.045),
    new THREE.Vector2(0.315, 0.08),
    new THREE.Vector2(0.34, 0.48),
    new THREE.Vector2(0.395, 1.12),
    new THREE.Vector2(0.465, 1.78),
    new THREE.Vector2(isLeft ? 0.50 : 0.485, 2.34),
    new THREE.Vector2(0.001, 2.39)
  ];
  const geometry = new THREE.LatheGeometry(new THREE.SplineCurve(points).getPoints(48), 48);
  const p = geometry.attributes.position;
  for (let i = 0; i < p.count; i++) {
    const x = p.getX(i);
    const y = p.getY(i);
    const z = p.getZ(i) * 0.88;
    const upperX = isLeft ? -0.49 : 0.38;
    const lowerX = isLeft ? -0.76 : 0.30;
    const blend = 1 - smoothstep(0.05, 2.25, y);
    const centerX = THREE.MathUtils.lerp(upperX, lowerX, blend);
    p.setXYZ(i, x + centerX, y, z);
  }
  geometry.computeVertexNormals();
  const leg = new THREE.Mesh(geometry, legMaterial);
  leg.name = name;
  return leg;
}
