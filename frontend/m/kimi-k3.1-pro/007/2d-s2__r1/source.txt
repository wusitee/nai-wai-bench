    for (var i = 0; i < times.length - 1; i++) {
      if (t >= times[i] && t <= times[i + 1]) {
        index = i;
        break;
      }
    }

    var a = poses[index];
    var b = poses[index + 1];
    var raw = (t - times[index]) / (times[index + 1] - times[index]);
    var u = raw * raw * (3 - 2 * raw);

    var root = arrayLerp(a.root, b.root, u);
    var side = lerp(a.side, b.side, u);
    var eyes = arrayLerp(a.eyes, b.eyes, u);
    var hl = arrayLerp(a.hl, b.hl, u);
    var hr = arrayLerp(a.hr, b.hr, u);

    byId("yellow-character").setAttribute(
      "transform",
      "translate(" + root[1].toFixed(2) + " " + root[2].toFixed(2) + ") scale(" + root[0].toFixed(4) + ")"
    );

    setPath("torso-shape", mixPath(a.body, b.body, u));
    setPath("head-shape", mixPath(a.head, b.head, u));
    var bellyPath = mixPath(a.belly, b.belly, u);
    setPath("belly-patch", bellyPath);
    setPath("belly-highlight", bellyPath);
    setPath("leg-left-shape", mixPath(a.ll, b.ll, u));
    setPath("leg-right-shape", mixPath(a.rl, b.rl, u));
    setPath("arm-left-shape", mixPath(a.al, b.al, u));
    setPath("arm-right-shape", mixPath(a.ar, b.ar, u));
    setPath("mouth-line", mixPath(a.mouth, b.mouth, u));

    byId("hand-left-pose").setAttribute("transform", handTransform(hl));
    byId("hand-right-pose").setAttribute("transform", handTransform(hr));
    setOpacity("hand-left", hl[4]);
    setOpacity("hand-right", hr[4]);

    setOpacity("front-shading", 1 - side);
    setOpacity("side-shading", side);
    setOpacity("nose", (1 - side) * 0.9);
    setOpacity("side-nose", side * 0.4);
    byId("nose").setAttribute(
      "transform",
      "translate(" + ((eyes[0] - 109) * 0.3).toFixed(2) + " " + (eyes[1] - 42).toFixed(2) + ")"
    );

    var open = eyes[4];
    var rightVisible = eyes[5];
    byId("eye-left-open").setAttribute(
      "transform",
      "translate(" + eyes[0].toFixed(2) + " " + eyes[1].toFixed(2) + ") scale(1 " + (0.45 + 0.55 * open).toFixed(3) + ")"
    );
    byId("eye-right-open").setAttribute(
      "transform",
      "translate(" + eyes[2].toFixed(2) + " " + eyes[3].toFixed(2) + ") scale(1 " + (0.45 + 0.55 * open).toFixed(3) + ")"
    );
    setOpacity("eye-left-open", open);
    setOpacity("eye-right-open", open * rightVisible);
    setOpacity("eyes", 1 - open);
    setOpacity("eyelids", (1 - open) * 0.72);
    setOpacity("eye-right", rightVisible);
    setOpacity("eyelid-right", rightVisible);
    setOpacity("socket-right", rightVisible);

    setPath("eye-left",
      "M" + (eyes[0] - 5).toFixed(2) + " " + eyes[1].toFixed(2) +
      " Q" + eyes[0].toFixed(2) + " " + (eyes[1] + 1).toFixed(2) +
      " " + (eyes[0] + 5).toFixed(2) + " " + (eyes[1] - 0.2).toFixed(2));
    setPath("eye-right",
      "M" + (eyes[2] - 5).toFixed(2) + " " + eyes[3].toFixed(2) +
      " Q" + eyes[2].toFixed(2) + " " + (eyes[3] + 1).toFixed(2) +
      " " + (eyes[2] + 5).toFixed(2) + " " + (eyes[3] - 0.2).toFixed(2));
    setPath("eyelid-left",
      "M" + (eyes[0] - 6).toFixed(2) + " " + (eyes[1] - 1.2).toFixed(2) +
      " Q" + eyes[0].toFixed(2) + " " + (eyes[1] - 2).toFixed(2) +
      " " + (eyes[0] + 6).toFixed(2) + " " + (eyes[1] - 0.9).toFixed(2));
    setPath("eyelid-right",
      "M" + (eyes[2] - 6).toFixed(2) + " " + (eyes[3] - 1.2).toFixed(2) +
      " Q" + eyes[2].toFixed(2) + " " + (eyes[3] - 2).toFixed(2) +
      " " + (eyes[2] + 6).toFixed(2) + " " + (eyes[3] - 0.9).toFixed(2));
    setPath("socket-left",
      "M" + (eyes[0] - 7).toFixed(2) + " " + (eyes[1] - 3).toFixed(2) +
      " Q" + eyes[0].toFixed(2) + " " + (eyes[1] - 5).toFixed(2) +
      " " + (eyes[0] + 7).toFixed(2) + " " + (eyes[1] - 2.5).toFixed(2));
    setPath("socket-right",
      "M" + (eyes[2] - 7).toFixed(2) + " " + (eyes[3] - 3).toFixed(2) +
      " Q" + eyes[2].toFixed(2) + " " + (eyes[3] - 5).toFixed(2) +
      " " + (eyes[2] + 7).toFixed(2) + " " + (eyes[3] - 2.5).toFixed(2));

    var mouthCenter = (eyes[0] + eyes[2]) * 0.5;
    setPath("lower-lip",
      "M" + (mouthCenter - 11).toFixed(2) + " " + (eyes[1] + 12).toFixed(2) +
      " Q" + mouthCenter.toFixed(2) + " " + (eyes[1] + 14).toFixed(2) +
      " " + (mouthCenter + 11).toFixed(2) + " " + (eyes[1] + 11.5).toFixed(2));
    setOpacity("lower-lip", (1 - side) * open * 0.32);

    var shadow = byId("floor-shadow");
    shadow.setAttribute("cx", (lerp(117, 121, side) * root[0] + root[1]).toFixed(2));
    shadow.setAttribute("rx", (52 * root[0]).toFixed(2));
    shadow.setAttribute("opacity", Math.min(0.15, 0.12 * root[0]).toFixed(3));
  }

  window.nwb = { render: render };
  render(0);
})();
